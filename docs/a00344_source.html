125template <typename Range, typename Separator>
126typename std::enable_if<
127 lumex::core::utility::traits::range::has_elements_convertible_to<
128 Range, std::string const &>::value
129 && lumex::core::utility::traits::stream::is_streamable<
130 typename std::decay<Separator>::type>::value,
131 std::string>::type
132quote_each (Range const &range, Separator const &separator, char mark)
133{
134 std::ostringstream oss;
135 bool is_first = true;
136 for (auto const &element : range)
137 {
138 std::string const &text = element;
139 if (!is_first)
140 oss << separator;
141 is_first = false;
142 oss << mark << text << mark;
143 }
144 return oss.str ();
145}
