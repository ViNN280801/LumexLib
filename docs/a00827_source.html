108inline int
109builtin_order (std::memory_order order) LUMEX_NOEXCEPT
110{
111 switch (order)
112 {
113 case std::memory_order_relaxed:
114 return __ATOMIC_RELAXED;
115 case std::memory_order_consume:
116 case std::memory_order_acquire:
117 return __ATOMIC_ACQUIRE;
118 case std::memory_order_release:
119 return __ATOMIC_RELEASE;
120 case std::memory_order_acq_rel:
121 return __ATOMIC_ACQ_REL;
122 case std::memory_order_seq_cst:
123 return __ATOMIC_SEQ_CST;
124 }
125 return __ATOMIC_SEQ_CST;
126}
127
130inline int
131builtin_load_order (std::memory_order order) LUMEX_NOEXCEPT
132{
133 if (order == std::memory_order_release)
134 return __ATOMIC_RELAXED;
135 if (order == std::memory_order_acq_rel)
136 return __ATOMIC_ACQUIRE;
137 return builtin_order (order);
138}
139
142inline int
143builtin_store_order (std::memory_order order) LUMEX_NOEXCEPT
144{
145 if (order == std::memory_order_consume || order == std::memory_order_acquire
146 || order == std::memory_order_acq_rel)
147 return __ATOMIC_RELEASE;
148 return builtin_order (order);
149}
150
154inline int
155builtin_failure_order (std::memory_order success,
156 std::memory_order failure) LUMEX_NOEXCEPT
157{
158 int const wanted = builtin_load_order (failure);
159 int const allowed = builtin_load_order (success);
160 return wanted > allowed ? allowed : wanted;
161}
162
164inline u128_t
165pack (dwcas_value_t value) LUMEX_NOEXCEPT
166{
167 return (static_cast<u128_t> (value.hi) << 64) | value.lo;
168}
169
171inline dwcas_value_t
172unpack (u128_t value) LUMEX_NOEXCEPT
173{
174 dwcas_value_t halves = { static_cast<std::uint64_t> (value),
175 static_cast<std::uint64_t> (value >> 64) };
176 return halves;
177}

180struct builtin_operations_t
181{
183 static dwcas_value_t
184 load (std::uint64_t *words, std::memory_order order) LUMEX_NOEXCEPT
185 {
186 return unpack (__atomic_load_n (reinterpret_cast<u128_alias_t *> (words),
187 builtin_load_order (order)));
188 }
189
191 static dwcas_value_t
192 exchange (std::uint64_t *words, std::uint64_t const *, dwcas_value_t desired,
193 std::memory_order order) LUMEX_NOEXCEPT
194 {
195 return unpack (
196 __atomic_exchange_n (reinterpret_cast<u128_alias_t *> (words),
197 pack (desired), builtin_order (order)));
198 }
199
201 static void
202 store (std::uint64_t *words, std::uint64_t const *, dwcas_value_t desired,
203 std::memory_order order) LUMEX_NOEXCEPT
204 {
205 __atomic_store_n (reinterpret_cast<u128_alias_t *> (words), pack (desired),
206 builtin_store_order (order));
207 }
208
211 static dwcas_value_t
212 compare_exchange (std::uint64_t *words, dwcas_value_t expected,
213 dwcas_value_t desired, std::memory_order success,
214 std::memory_order failure) LUMEX_NOEXCEPT
215 {
216 u128_alias_t seen = pack (expected);
217 __atomic_compare_exchange_n (
218 reinterpret_cast<u128_alias_t *> (words), &seen, pack (desired), false,
219 builtin_order (success), builtin_failure_order (success, failure));
220 return unpack (seen);
221 }
222};
