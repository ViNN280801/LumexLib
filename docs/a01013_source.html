765template <typename Char>
766LUMEX_FORMAT_CONSTEXPR std::size_t
767code_point_length (Char const *it, Char const *end) LUMEX_NOEXCEPT
768{
769 std::size_t length = 1;
770 if (sizeof (Char) == 1)
771 {
772 unsigned char const lead = static_cast<unsigned char> (*it);
773 if (lead >= 0xF0 && lead < 0xF8)
774 length = 4;
775 else if (lead >= 0xE0)
776 length = lead < 0xF0 ? 3 : 1;
777 else if (lead >= 0xC0)
778 length = 2;
779 }
780 else if (sizeof (Char) == 2)
781 {
782 unsigned const unit = static_cast<unsigned> (*it) & 0xFFFFu;
783 if (unit >= 0xD800u && unit < 0xDC00u)
784 length = 2;
785 }
786 if (static_cast<std::size_t> (end - it) < length)
787 length = 1;
788 return length;
789}

792template <typename Char>
793LUMEX_FORMAT_CONSTEXPR int
794parse_nonnegative_int (Char const *&it, Char const *end)
795{
796 unsigned long long value = 0;
797 while (it != end && is_digit (*it))
798 {
799 value
800 = value * 10 + static_cast<unsigned> (*it - static_cast<Char> ('0'));
801 if (value > static_cast<unsigned long long> (INT_MAX))
802 throw format_error ("number is too big");
803 ++it;
804 }
805 return static_cast<int> (value);
806}

812template <typename Char>
813LUMEX_FORMAT_CONSTEXPR arg_ref_t<Char>
814parse_arg_id (Char const *&it, Char const *end)
815{
816 arg_ref_t<Char> ref;
817 if (it != end && is_digit (*it))
818 {
819 if (*it == static_cast<Char> ('0') && it + 1 != end && is_digit (it[1]))
820 throw format_error ("invalid format string");
821 // An index above INT_MAX cannot name an argument: saturate, so the
822 // lookup reports "argument not found" rather than a number error.
823 unsigned long long value = 0;
824 while (it != end && is_digit (*it))
825 {
826 if (value <= static_cast<unsigned long long> (INT_MAX))
827 value = value * 10
828 + static_cast<unsigned> (*it - static_cast<Char> ('0'));
829 ++it;
830 }
831 ref.kind = ArgRefKind::index;
832 ref.index = value > static_cast<unsigned long long> (INT_MAX)
833 ? INT_MAX
834 : static_cast<int> (value);
835 return ref;
836 }
837 if (it != end && is_name_start (*it))
838 {
839 Char const *const start = it;
840 while (it != end && (is_name_start (*it) || is_digit (*it)))
841 ++it;
842 ref.kind = ArgRefKind::name;
843 ref.name = start;
844 ref.name_size = static_cast<std::size_t> (it - start);
845 return ref;
846 }
847 throw format_error ("invalid format string");
848}

953template <typename Char>
954LUMEX_FORMAT_CONSTEXPR arg_ref_t<Char>
955parse_dynamic_ref (Char const *&it, Char const *end,
956 basic_format_parse_context<Char> &ctx)
957{
958 ++it; // '{'
959 arg_ref_t<Char> ref;
960 if (it != end && *it == static_cast<Char> ('}'))
961 {
962 ref.kind = ArgRefKind::index;
963 ref.index = ctx.next_arg_id ();

972 ctx.check_dynamic_spec (ref.index);
973 }
974 else
975 ctx.check_named_arg ();
976 }
977 if (it == end || *it != static_cast<Char> ('}'))
978 throw format_error ("invalid format string");
979 ++it;
980 return ref;
981}

1007template <typename Char>
1008LUMEX_FORMAT_CONSTEXPR void
1009validate_specs (format_specs_t<Char> const &specs, SpecKind kind)
1010{
1011 Char const type = specs.type;
1012 bool numeric = false;
1013 bool allows_precision = false;
1014 bool allows_localized = false;
1015 bool allows_zero = false;
1016 char const *const invalid = kind == SpecKind::character
1017 ? "invalid format specifier for char"
1018 : "invalid format specifier";
1019 switch (kind)
1020 {
1021 case SpecKind::integer:
1022 if (type != Char () && type != static_cast<Char> ('c')
1023 && !is_integer_presentation (type))
1024 throw format_error ("invalid format specifier");
1025 numeric = type != static_cast<Char> ('c');
1026 allows_localized = numeric;
1027 allows_zero = numeric;
1028 break;
1029 case SpecKind::character:
1030 if (type != Char () && type != static_cast<Char> ('c')
1031 && type != static_cast<Char> ('?')
1032 && !is_integer_presentation (type))
1033 throw format_error ("invalid format specifier");
1034 numeric = is_integer_presentation (type);
1035 allows_localized = numeric;
1036 allows_zero = numeric;
1037 break;
1038 case SpecKind::boolean:
1039 if (type != Char () && type != static_cast<Char> ('s')
1040 && !is_integer_presentation (type))
1041 throw format_error ("invalid format specifier");
1042 numeric = is_integer_presentation (type);
1043 allows_localized = true;
1044 allows_zero = numeric;
1045 break;
1046 case SpecKind::floating:
1047 if (type != Char () && !is_float_presentation (type))
1048 throw format_error ("invalid format specifier");
1049 numeric = true;
1050 allows_precision = true;
1051 allows_localized = true;
1052 allows_zero = true;
1053 break;
1054 case SpecKind::string:
1055 if (type != Char () && type != static_cast<Char> ('s')
1056 && type != static_cast<Char> ('?'))
1057 throw format_error ("invalid format specifier");
1058 allows_precision = true;
1059 break;
1060 case SpecKind::pointer:
1061 if (type != Char () && type != static_cast<Char> ('p')
1062 && type != static_cast<Char> ('P'))
1063 throw format_error ("invalid format specifier");
1064 // P2510 (a defect report against C++20): zeros pad after "0x".
1065 allows_zero = true;
1066 break;
1067 case SpecKind::enumeration:
1068 if (type != Char () && type != static_cast<Char> ('s')
1069 && !is_integer_presentation (type))
1070 throw format_error ("invalid format specifier");
1071 numeric = is_integer_presentation (type);
1072 allows_precision = !numeric;
1073 allows_localized = numeric;
1074 allows_zero = numeric;
1075 break;
1076 }
1077 if ((specs.sign != Sign::none || specs.alternate) && !numeric)
1078 throw format_error (invalid);
1079 if (specs.zero_pad && !allows_zero)
1080 throw format_error (kind == SpecKind::character
1081 ? invalid
1082 : "format specifier requires numeric argument");
1083 if ((specs.precision >= 0 || specs.precision_ref.kind != ArgRefKind::none)
1084 && !allows_precision)
1085 throw format_error ("invalid format specifier");
1086 if (specs.localized && !allows_localized)
1087 throw format_error ("invalid format specifier");
1088}

1095template <typename Char>
1096LUMEX_FORMAT_CONSTEXPR Char const *
1097parse_std_specs (basic_format_parse_context<Char> &ctx,
1098 format_specs_t<Char> &specs, SpecKind kind)
1099{
1100 Char const *it = ctx.begin ();
1101 Char const *const end = ctx.end ();
1102 if (it == end || *it == static_cast<Char> ('}'))
1103 {
1104 validate_specs (specs, kind);
1105 return it;
1106 }
1107
1108 // [[fill]align]
1109 std::size_t const fill_length = code_point_length (it, end);
1110 if (static_cast<std::size_t> (end - it) > fill_length
1111 && is_align (it[fill_length]))
1112 {
1113 if (*it == static_cast<Char> ('{'))
1114 throw format_error ("invalid fill character '{'");
1115 if (*it == static_cast<Char> ('}'))
1116 throw format_error ("invalid fill character '}'");
1117 for (std::size_t i = 0; i < fill_length; ++i)
1118 specs.fill[i] = it[i];
1119 specs.fill_size = static_cast<unsigned char> (fill_length);
1120 it += fill_length;
1121 }
1122 if (it != end && is_align (*it))
1123 {
1124 specs.align = *it == static_cast<Char> ('<') ? Align::left
1125 : *it == static_cast<Char> ('>') ? Align::right
1126 : Align::center;
1127 ++it;
1128 }
1129
1130 // [sign]
1131 if (it != end
1132 && (*it == static_cast<Char> ('+') || *it == static_cast<Char> ('-')
1133 || *it == static_cast<Char> (' ')))
1134 {
1135 specs.sign = *it == static_cast<Char> ('+') ? Sign::plus
1136 : *it == static_cast<Char> ('-') ? Sign::minus
1137 : Sign::space;
1138 ++it;
1139 }
1140
1141 // [#]
1142 if (it != end && *it == static_cast<Char> ('#'))
1143 {
1144 specs.alternate = true;
1145 ++it;
1146 }
1147
1148 // [0]
1149 if (it != end && *it == static_cast<Char> ('0'))
1150 {
1151 specs.zero_pad = true;
1152 ++it;
1153 // The width that may follow starts with a nonzero digit (`{:00}`
1154 // is an error, as in std::format).
1155 if (it != end && *it == static_cast<Char> ('0'))
1156 throw format_error ("invalid format specifier");
1157 }
1158
1159 // [width]
1160 if (it != end && is_digit (*it))
1161 specs.width = parse_nonnegative_int (it, end);
1162 else if (it != end && *it == static_cast<Char> ('{'))
1163 specs.width_ref = parse_dynamic_ref (it, end, ctx);
1164
1165 // [.precision]
1166 if (it != end && *it == static_cast<Char> ('.'))
1167 {
1168 ++it;
1169 if (it != end && is_digit (*it))
1170 specs.precision = parse_nonnegative_int (it, end);
1171 else if (it != end && *it == static_cast<Char> ('{'))
1172 specs.precision_ref = parse_dynamic_ref (it, end, ctx);
1173 else if (it == end)
1174 throw format_error ("invalid precision");
1175 else
1176 throw format_error ("invalid format string");
1177 }
1178
1179 // [L]
1180 if (it != end && *it == static_cast<Char> ('L'))
1181 {
1182 specs.localized = true;
1183 ++it;
1184 }
1185
1186 // [type]
1187 if (it != end && *it != static_cast<Char> ('}'))
1188 {
1189 specs.type = *it;
1190 ++it;
1191 }
1192
1193 if (it != end && *it != static_cast<Char> ('}'))
1194 throw format_error ("invalid format specifier");
1195 validate_specs (specs, kind);
1196 return it;
1197}

1224template <typename Char, typename Handler>
1225LUMEX_FORMAT_CONSTEXPR void
1226parse_format_string (Char const *begin, Char const *end, Handler &handler)
1227{
1228 Char const *text = begin;
1229 Char const *it = begin;
1230 while (it != end)
1231 {
1232 Char const value = *it;
1233 if (value == static_cast<Char> ('{'))
1234 {
1235 handler.on_text (text, it);
1236 handler.on_field_start (static_cast<std::size_t> (it - begin));
1237 ++it;
1238 if (it == end)
1239 throw format_error ("invalid format string");
1240 if (*it == static_cast<Char> ('{'))
1241 {
1242 handler.on_text (it, it + 1);
1243 ++it;
1244 text = it;
1245 continue;
1246 }
1247 if (*it == static_cast<Char> ('}') || *it == static_cast<Char> (':'))
1248 handler.on_auto_arg ();
1249 else
1250 {
1251 arg_ref_t<Char> const ref = parse_arg_id (it, end);
1252 if (ref.kind == ArgRefKind::index)
1253 handler.on_index_arg (ref.index);
1254 else
1255 handler.on_name_arg (ref.name, ref.name_size);
1256 }
1257 if (it == end)
1258 throw format_error ("invalid format string");
1259 if (*it == static_cast<Char> (':'))
1260 ++it;
1261 else if (*it != static_cast<Char> ('}'))
1262 throw format_error ("invalid format string");
1263 it = handler.on_format_specs (it, end);
1264 if (it == end)
1265 throw format_error ("missing '}' in format string");
1266 if (*it != static_cast<Char> ('}'))
1267 throw format_error ("unknown format specifier");
1268 ++it;
1269 text = it;
1270 }
1271 else if (value == static_cast<Char> ('}'))
1272 {
1273 handler.on_text (text, it);
1274 handler.on_field_start (static_cast<std::size_t> (it - begin));
1275 ++it;
1276 if (it == end || *it != static_cast<Char> ('}'))
1277 throw format_error ("unmatched '}' in format string");
1278 handler.on_text (it, it + 1);
1279 ++it;
1280 text = it;
1281 }
1282 else
1283 ++it;
1284 }
1285 handler.on_text (text, end);
1286}

1515template <typename Char>
1516std::uint32_t
1517decode (Char const *&it, Char const *end, bool &valid) LUMEX_NOEXCEPT
1518{
1519 valid = true;
1520 if (sizeof (Char) == 1)
1521 {
1522 unsigned char const lead = static_cast<unsigned char> (*it);
1523 std::size_t length = 0;
1524 std::uint32_t cp = 0;
1525 if (lead < 0x80)
1526 {
1527 ++it;
1528 return lead;
1529 }
1530 if (lead >= 0xC2 && lead < 0xE0)
1531 {
1532 length = 2;
1533 cp = lead & 0x1Fu;
1534 }
1535 else if (lead >= 0xE0 && lead < 0xF0)
1536 {
1537 length = 3;
1538 cp = lead & 0x0Fu;
1539 }
1540 else if (lead >= 0xF0 && lead < 0xF5)
1541 {
1542 length = 4;
1543 cp = lead & 0x07u;
1544 }
1545 if (length == 0 || static_cast<std::size_t> (end - it) < length)
1546 {
1547 valid = false;
1548 ++it;
1549 return lead;
1550 }
1551 for (std::size_t i = 1; i < length; ++i)
1552 {
1553 unsigned char const unit = static_cast<unsigned char> (it[i]);
1554 if ((unit & 0xC0u) != 0x80u)
1555 {
1556 valid = false;
1557 ++it;
1558 return lead;
1559 }
1560 cp = (cp << 6) | (unit & 0x3Fu);
1561 }
1562 it += length;
1563 return cp;
1564 }
1565 if (sizeof (Char) == 2)
1566 {
1567 std::uint32_t const unit = static_cast<std::uint32_t> (*it) & 0xFFFFu;
1568 if (unit >= 0xD800u && unit < 0xDC00u && end - it >= 2)
1569 {
1570 std::uint32_t const low
1571 = static_cast<std::uint32_t> (it[1]) & 0xFFFFu;
1572 if (low >= 0xDC00u && low < 0xE000u)
1573 {
1574 it += 2;
1575 return 0x10000u + ((unit - 0xD800u) << 10) + (low - 0xDC00u);
1576 }
1577 }
1578 if (unit >= 0xD800u && unit < 0xE000u)
1579 valid = false;
1580 ++it;
1581 return unit;
1582 }
1583 std::uint32_t const unit = static_cast<std::uint32_t> (*it);
1584 ++it;
1585 return unit;
1586}

1753template <typename Char>
1754std::size_t
1755next_grapheme (Char const *&it, Char const *end) LUMEX_NOEXCEPT
1756{
1757 if (*it == static_cast<Char> ('\r') && it + 1 != end
1758 && it[1] == static_cast<Char> ('\n'))
1759 {
1760 it += 2; // CR LF is one cluster
1761 return 1;
1762 }
1763 if (static_cast<std::uint32_t> (*it) < 0x80u
1764 && (it + 1 == end || static_cast<std::uint32_t> (it[1]) < 0x80u))
1765 {
1766 ++it; // ASCII followed by ASCII (or the end): the common fast path
1767 return 1;
1768 }
1769 bool valid = true;
1770 std::uint32_t const first = decode (it, end, valid);
1771 if (!valid)
1772 return 1;
1773 std::size_t const width = is_wide_code_point (first) ? 2 : 1;
1774 std::uint32_t previous = first;
1775 bool regional_pair_open = is_regional_indicator (first);
1776 while (it != end)
1777 {
1778 Char const *next = it;
1779 bool next_valid = true;
1780 std::uint32_t const cp = decode (next, end, next_valid);
1781 if (!next_valid)
1782 break;
1783 bool joins = is_grapheme_extend (cp) || cp == 0x200Du;
1784 if (!joins && previous == 0x200Du && is_extended_pictographic (cp))
1785 joins = true;
1786 if (!joins && regional_pair_open && is_regional_indicator (cp))
1787 {
1788 joins = true;
1789 regional_pair_open = false;
1790 }
1791 if (!joins)
1792 break;
1793 previous = cp;
1794 it = next;
1795 }
1796 return width;
1797}
