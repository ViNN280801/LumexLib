243 T
244 generate_number (T from_val, T to_val, DistributionType dist_type) const
245 {
246 switch (dist_type)
247 {
248 case DistributionType::UNIFORM:
249 return generate_uniform (from_val, to_val);
250
251 case DistributionType::NORMAL:
252 return generate_normal_safe (from_val, to_val);
253
254 case DistributionType::EXPONENTIAL:
255 return generate_exponential_safe (from_val);
256
257 case DistributionType::GAMMA:
258 return generate_gamma_safe (from_val, to_val);
259
260 case DistributionType::BERNOULLI:
261 return generate_bernoulli_safe (static_cast<double> (from_val));
262
263 case DistributionType::BINOMIAL:
264 return generate_binomial_safe (static_cast<int> (to_val),
265 static_cast<double> (from_val));
266
267 case DistributionType::GEOMETRIC:
268 return generate_geometric_safe (static_cast<double> (from_val));
269
270 case DistributionType::POISSON:
271 return generate_poisson_safe (static_cast<double> (from_val));
272
273 default:
274 return generate_uniform (from_val, to_val);
275 }
276 }

989 std::vector<T>
990 get_sequence (std::size_t count, T from_val = get_default_min (),
991 T to_val = get_default_max (),
992 DistributionType dist_type = DistributionType::UNIFORM) const
993 {
994 if (count == 0)
995 {
996 // Note: Returning empty sequence for zero count request
997 return {};
998 }
999
1000 try
1001 {
1002 std::vector<T> sequence;
1003 sequence.reserve (count); // Pre-allocate memory to avoid reallocations
1004
1005 for (std::size_t i = 0; i < count; ++i)
1006 sequence.push_back (generate_number (from_val, to_val, dist_type));
1007
1008 return sequence;
1009 }
1010 catch (std::exception const &e)
1011 {
1012 std::cerr << "Error generating sequence: " << e.what ()
1013 << ". Parameters: count=" << count << ", from=" << from_val
1014 << ", to=" << to_val << ". Returning empty sequence.\n";
1015 return {};
1016 }
1017 catch (...)
1018 {
1019 std::cerr << "Unknown exception during sequence generation. "
1020 << "Parameters: count=" << count << ", from=" << from_val
1021 << ", to=" << to_val << ". Returning empty sequence.\n";
1022 return {};
1023 }
1024 }
