118template <typename SuccessType, typename ErrorType> class Expected
119{
120public:
121 // ====================== Static assertions ====================== //
122 // References are forbidden because:
123 // 1. This class (and its specializations) is meant to
124 // own the stored success or error value. If SuccessType or ErrorType were
125 // references, for example `int&`, Expected would not own the object, only
126 // refer to it. That can dangle if the original object referred to by
127 // SuccessType (or ErrorType) is destroyed before Expected.
128 // 2. Union restriction: the C++ standard forbids reference members in a
129 // `union`,
130 // because references are neither movable nor copyable.
131 // 3. C++23 compatibility: std::expected also forbids reference types for
132 // the error parameter for the same reasons. Behavior stays consistent and
133 // predictable.
134 LUMEX_STATIC_ASSERT_MSG (!std::is_reference<SuccessType>::value,
135 "Expected<T,E>: T must not be a reference");
136 LUMEX_STATIC_ASSERT_MSG (!std::is_function<SuccessType>::value,
137 "Expected<T,E>: T must not be a function type");
138 LUMEX_STATIC_ASSERT_MSG (
139 !std::is_same<typename std::decay<SuccessType>::type,
140 in_place_tag>::value,
141 "Expected<T,E>: T must not be in_place_tag");
142 LUMEX_STATIC_ASSERT_MSG (
143 !std::is_same<typename std::decay<SuccessType>::type, unexpect_t>::value,
144 "Expected<T,E>: T must not be unexpect_t");
145
146 LUMEX_STATIC_ASSERT_MSG (!std::is_reference<ErrorType>::value,
147 "Expected<T,E>: E must not be a reference");
148 LUMEX_STATIC_ASSERT_MSG (!std::is_function<ErrorType>::value,
149 "Expected<T,E>: E must not be a function type");

545 LUMEX_ATTRIBUTE_NODISCARD (
546 "Return value is the contained error; should always be used.")
547 LUMEX_CONSTEXPR_CXX14 ErrorType &
548 error () &
549 {
550 // Precondition: !m_has_value.
551 // In std::expected, error() is an unchecked observer: calling it without
552 // an error is undefined behavior, and checked access is value(), which
553 // throws bad_expected_access<E>. Here the storage is a union, so reading
554 // m_storage.m_error while m_has_value is true would read an inactive
555 // member, which is undefined behavior in C++.
556 //
557 // This method checks the precondition with LUMEX_ASSERT instead of
558 // throwing. LUMEX_ASSERT is active in every build, NDEBUG included: a
559 // violation prints the condition, the file and the line, and aborts the
560 // program before the inactive member is read. The check costs one branch
561 // in every build. Because error() never throws, it stays usable where an
562 // exception must not escape (destructors, swap, emergency paths);
563 // checked, catchable access is value() or a prior has_value() check.
564 LUMEX_ASSERT (!m_has_value
565 && "error() called on an Expected that holds a value");
566 return m_storage.m_error;
567 }

908 LUMEX_CONSTEXPR_CXX14 void
909 swap (Expected &other)
910#if __cplusplus >= 201703L
911 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible_v<SuccessType>
912 &&std::is_nothrow_move_constructible_v<ErrorType>
913 &&std::is_nothrow_swappable_v<SuccessType>
914 &&std::is_nothrow_swappable_v<ErrorType>)
915#else
916 LUMEX_NOEXCEPT_IF (
917 std::is_nothrow_move_constructible<SuccessType>::value
918 &&std::is_nothrow_move_constructible<ErrorType>::value)
919#endif
920 {
921 if (this == &other)
922 return;
923
924 if (m_has_value && other.m_has_value)
925 {
926 std::swap (m_storage.m_value, other.m_storage.m_value);
927 }
928 else if (!m_has_value && !other.m_has_value)
929 { // Both hold an error
930 std::swap (m_storage.m_error, other.m_storage.m_error);
931 }
932 else
933 { // One holds a value, the other an error. Move is required.
934 if (m_has_value)
935 {
936 ErrorType temp_error (std::move (other.m_storage.m_error));
937 other.m_storage.m_error.~ErrorType ();
938 new (std::addressof (other.m_storage.m_value))
939 SuccessType (std::move (m_storage.m_value));
940 m_storage.m_value.~SuccessType ();
941 new (std::addressof (m_storage.m_error))
942 ErrorType (std::move (temp_error));
943 }
944 else
945 {
946 SuccessType temp_value (std::move (other.m_storage.m_value));
947 other.m_storage.m_value.~SuccessType ();
948 new (std::addressof (other.m_storage.m_error))
949 ErrorType (std::move (m_storage.m_error));
950 m_storage.m_error.~ErrorType ();
951 new (std::addressof (m_storage.m_value))
952 SuccessType (std::move (temp_value));
953 }
954 std::swap (m_has_value, other.m_has_value);
955 }
956 }
