494 optional &
495 operator= (optional const &other)
496 LUMEX_NOEXCEPT_IF (std::is_nothrow_copy_constructible<T>::value
497 &&std::is_nothrow_copy_assignable<T>::value)
498 {
499 if (this != &other) // Handle self-assignment
500 {
501 if (other.m_has_value) // 'other' has a value
502 {
503 if (m_has_value) // '*this' also has a value: assign or destroy
504 // old, construct new
505 {
506 // If T is trivially copy assignable, we could assign directly.
507 // For general case, destroy and reconstruct is safer for
508 // non-trivial types. This implementation chooses safety over
509 // potential micro-optimizations.
510 destroy ();
511 construct (*other.get_ptr ());
512 }
513 else // '*this' is empty, 'other' has value: construct new
514 {
515 construct (*other.get_ptr ());
516 }
517 }
518 else // 'other' is empty
519 {
520 if (m_has_value) // '*this' has a value, 'other' is empty: destroy
521 // '*this' value
522 destroy ();
523 // else: Both are empty, nothing to do.
524 }
525 }
526 return *this;
527 }

546 operator= (optional &&other)
547 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible<T>::value
548 &&std::is_nothrow_move_assignable<T>::value)
549 {
550 if (this != &other) // Handle self-assignment
551 {
552 if (other.m_has_value)
553 {
554 if (m_has_value)
555 // If T is move assignable, we could assign directly: *get_ptr()
556 // = std::move(*other.get_ptr()); For general case, destroy and
557 // reconstruct is safer for non-trivial types. This
558 // implementation chooses safety over potential
559 // micro-optimizations.

566 other.destroy (); // Ensure the source optional is empty after move
567 }

570 destroy ();
571 } // If other is empty, ensure *this* is empty
