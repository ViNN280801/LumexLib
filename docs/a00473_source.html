571// A character of an identifier. A byte of a multibyte UTF-8 sequence counts,
572// so that an identifier written in UTF-8 is read whole.
573inline bool

575{
576 return (ch >= 'a' && ch <= 'z') || (ch >= 'A' && ch <= 'Z')
577 || (ch >= '0' && ch <= '9') || ch == '_'
578 || static_cast<unsigned char> (ch) >= 0x80;

588// Where the name of the member begins in the pretty string (the
589// LUMEX_FUNCTION_NAME of nttp_pretty), or len if there is none.
590//
591// The pointer is the last template argument of nttp_pretty, and the member is
592// the last thing the pointer names, so the name is the identifier that follows
593// the last member access (`.`, `->`) or scope (`::`) operator of the string,
594// and nothing but closing delimiters follows it. What stands before the
595// operator does not matter, and that is the point: the compilers print the
596// path to the member differently (the data in the parser tests).
597// GCC 8 .. 12 `...Pointer = (&
598// fake_object_storage<T>.fake_object_wrapper_t<T>::value.id)]` GCC 13
599// `...Pointer = (&
600// fake_object_storage<T>.fake_object_wrapper_t<T>::value.ns::T::id)]`
601// (the member qualified by its class, which is also printed
602// after the last `.`)
603// Clang `...Pointer = &fake_object_storage.value.id]`
604// MSVC `...nttp_pretty<struct ns::T,&fake_object_storage<struct
605// ns::T>->value->id>(void) noexcept`
606// Taking the text after the last `.` or `->` alone yields `ns` for GCC 13.
607// The operator must be followed by an identifier start: a `.` inside a
608// floating-point template argument is followed by a digit, and an ellipsis by
609// a dot. Such text can only stand before the member, and the last operator
610// wins anyway.
611inline std::size_t
612member_name_begin (char const *pretty, std::size_t len) LUMEX_NOEXCEPT
613{
614 std::size_t begin = len;
615 for (std::size_t i = 0; i < len; ++i)
616 {
617 std::size_t op_len = 0;
618 if (pretty[i] == '.')
619 op_len = 1;
620 else if (i + 1 < len
621 && ((pretty[i] == ':' && pretty[i + 1] == ':')
622 || (pretty[i] == '-' && pretty[i + 1] == '>')))
623 op_len = 2;
624 if (op_len == 0)
625 continue;
626 if (i + op_len < len && is_ident_start (pretty[i + op_len]))
627 begin = i + op_len;
628 i += op_len - 1; // both characters of `::` and `->` are consumed
629 }
630 return begin;
631}

633// Copies the name of the member out of the pretty string into dest, which is
634// always NUL-terminated (dest_size characters at most, the name is cut if it
635// does not fit). A string without a member gives an empty name.
636inline void
637copy_parsed_name (char *dest, std::size_t dest_size,
638 char const *pretty) LUMEX_NOEXCEPT
639{
640 if (dest_size == 0)
641 return;
642
643 std::size_t len = 0;
644 while (pretty[len] != '\0')
645 ++len;
646
647 std::size_t start = member_name_begin (pretty, len);
648 std::size_t out = 0;
649 while (start < len && is_ident_char (pretty[start]) && out + 1 < dest_size)
650 dest[out++] = pretty[start++];
651 dest[out] = '\0';
652}

732// Phantom storage for name extraction only. Taking the address of a member is
733// enough for a pointer NTTP; constructing a real `inline T fake{}` fails on
734// MSVC (C2672) because a local reference into that object is not a constant
735// expression for `template <auto>`.
736//
737// On the Microsoft toolchain the object also has to exist: an instantiation
738// whose template argument has no linkage (a class local to a function, a type
739// of an unnamed namespace) is an internal-linkage variable, and MSVC reports
740// one that is declared but not defined as C7631 - the never-emitted address
741// is not enough for it (Clang, with the same toolchain, only warns
742// -Wundefined-internal). The union below carries a second member with a
743// default initializer, so `storage{}` initializes that one and T is never
744// constructed: every aggregate stays reflectable, trivial or not. The empty
745// destructor keeps the union destructible for a T with a non-trivial one -
746// `= default` would be deleted there - and destroys nothing.
747#if defined(_MSC_VER)
748template <typename T> union fake_object_wrapper_t
749{
750 T value;
751 char none = 0;
752 ~fake_object_wrapper_t () LUMEX_NOEXCEPT {}
753};

756fake_object_wrapper_t<T> const fake_object_storage{};
757#else

760 T const value;
761};

764extern fake_object_wrapper_t<T> const fake_object_storage;
765#endif

893// get of an aggregate with a registration, in every standard: the I-th
894// registered member. Agg may be a (const) reference or const. For an
895// unregistered aggregate, or an index past the last registered field, value_t
896// is unregistered_t: get then reports it (below C++14) or goes the automatic
897// way (from C++14).
898template <std::size_t I, typename Agg> struct registered_ref
899{
900 typedef typename std::remove_reference<Agg>::type plain_t;
901 typedef typename std::remove_const<plain_t>::type bare_t;
902 typedef typename registered_member<
903 bare_t, I, registered_usable<bare_t, I>::value>::value_t value_t;
904 typedef typename std::conditional<std::is_const<plain_t>::value,
905 value_t const, value_t>::type &type;
906};

908template <std::size_t I, typename Agg>
909typename registered_ref<I, Agg>::type
910get_field (Agg &value, std::true_type) LUMEX_NOEXCEPT
911{
912 typedef typename std::remove_const<Agg>::type bare_t;
913 return value.*(registered_member<bare_t, I, true>::pointer ());
914}
