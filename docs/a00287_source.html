273 Pointer
274 load (std::memory_order order) const LUMEX_NOEXCEPT
275 {
276 ::lumex::core::atomic::dwcas::require_dwcas ();
277 split_value_t const pinned = pin (rmw_order (order));
278 if (word::is_empty (pinned))
279 return traits_type::adopt (nullptr, raw_pointer (pinned));
280 block_type *const block = word::block_of (pinned);
281 block_type *owner = block;
282 element_type *stored = nullptr;
283 if (word::is_holder (pinned))
284 {
285 holder_type const *const holder
286 = static_cast<holder_type const *> (block);
287 owner = holder->owner ();
288 stored = static_cast<element_type *> (holder->pointer ());
289 }
290 else
291 stored = block_pointer (block, pinned);
292 count_and_settle (pinned, owner);
293 return traits_type::adopt (owner, stored);
294 }

424 static split_value_t
425 take_image (Pointer &desired) LUMEX_NOEXCEPT
426 {
427 block_type *const block = traits_type::control (desired);
428 element_type *const stored = traits_type::stored (desired);
429 if (block == nullptr)
430 return word::empty_word (word::address_of (stored));
431 std::uint64_t const offset
432 = word::address_of (stored)
433 - static_cast<std::uint64_t> (block->anchor ());
434 if (word::fits (offset))
435 {
436 traits_type::detach (desired);
437 return word::block_word (block, offset);
438 }
439 return word::holder_word (traits_type::make_holder (block, stored));
440 }

442 // -- The counter group of a word --
443 //
444 // The group G of a word is the ledger of the block for a block word and
445 // the strong group of the holder for a holder word. In the shared engine
446 // every unit that enters or leaves the ext of a holder also enters or
447 // leaves the ext of its owner (the mirror).
448
451 static void
452 deposit (split_value_t w, std::uint32_t n) LUMEX_NOEXCEPT
453 {
454 if (n == 0u)
455 return;
456 block_type *const block = word::block_of (w);
457 Policy::at (split_point_t::transfer);
458 if (word::is_holder (w))
459 {
460 holder_ledger_type::transfer (block, n);
461 if (traits_type::mirror_holder)
462 owner_of (block)->transfer_strong_ext (n);
463 }
464 else
465 ledger_type::transfer (block, n);
466 Policy::note (split_event_t::transfer, n);
467 }
468
471 static void
472 withdraw (split_value_t w, std::uint32_t n) LUMEX_NOEXCEPT
473 {
474 if (n == 0u)
475 return;
476 block_type *const block = word::block_of (w);
477 Policy::note (split_event_t::untransfer, n);
478 Policy::at (split_point_t::transfer);
479 if (word::is_holder (w))
480 {
481 if (traits_type::mirror_holder)
482 owner_of (block)->untransfer_strong_ext (n);
483 holder_ledger_type::untransfer (block, n);
484 }
485 else
486 ledger_type::untransfer (block, n);
487 }
488
491 static void
492 pay (split_value_t w, std::uint32_t n) LUMEX_NOEXCEPT
493 {
494 block_type *const block = word::block_of (w);
495 Policy::at (split_point_t::settle_ext);
496 if (word::is_holder (w))
497 {
498 if (traits_type::mirror_holder)
499 {
500 if (n == 1u)
501 owner_of (block)->settle_strong ();
502 else
503 owner_of (block)->settle_strong_n (n);
504 }
505 if (n == 1u)
506 holder_ledger_type::settle (block);
507 else
508 holder_ledger_type::settle_n (block, n);
509 }
510 else if (n == 1u)
511 ledger_type::settle (block);
512 else
513 ledger_type::settle_n (block, n);
514 }
515
518 static void
519 take_and_pay (split_value_t w, block_type *owner) LUMEX_NOEXCEPT
520 {
521 block_type *const block = word::block_of (w);
522 Policy::at (split_point_t::settle_ext);
523 if (word::is_holder (w))
524 {
525 if (traits_type::mirror_holder)
526 owner->take_and_settle_strong ();
527 else
528 ledger_type::add (owner);
529 holder_ledger_type::settle (block);
530 }
531 else
532 ledger_type::take_and_settle (block);
533 }

539 static void
540 drop_word (split_value_t w) LUMEX_NOEXCEPT
541 {
542 if (word::is_empty (w))
543 return;
544 Policy::at (split_point_t::release);
545 if (word::is_holder (w))
546 {
547 release_owner<traits_type::mirror_holder> (word::block_of (w), 0u);
548 holder_ledger_type::release (word::block_of (w));
549 }
550 else
551 ledger_type::release (word::block_of (w));
552 }

563 template <bool Mirror>
564 static typename std::enable_if<Mirror>::type
565 release_owner (block_type *holder, std::uint32_t surplus) LUMEX_NOEXCEPT
566 {
567 holder_type *const h = static_cast<holder_type *> (holder);
568 block_type *const owner = h->owner ();
569 h->disown ();
570 if (surplus == 0u)
571 owner->release_strong ();
572 else
573 owner->release_strong_with_ext (surplus);

604 static void
605 drop_word_with_ext (split_value_t w, std::uint32_t surplus) LUMEX_NOEXCEPT
606 {
607 if (surplus == 0u || word::is_empty (w))
608 {
609 drop_word (w);
610 return;
611 }
612 block_type *const block = word::block_of (w);
613 Policy::note (split_event_t::untransfer, surplus);
614 Policy::at (split_point_t::release);
615 if (word::is_holder (w))
616 {
617 release_owner<traits_type::mirror_holder> (block, surplus);
618 holder_ledger_type::release_with_ext (block, surplus);
619 }
620 else
621 ledger_type::release_with_ext (block, surplus);
622 }

627 static void
628 pay_back (split_value_t w, std::uint32_t deposited) LUMEX_NOEXCEPT
629 {
630 if (deposited != 0u)
631 Policy::note (split_event_t::untransfer, deposited);
632 Policy::note (split_event_t::settle_ext, 1u);
633 pay (w, deposited + 1u);

640 static Pointer
641 hand_over (split_value_t old) LUMEX_NOEXCEPT
642 {
643 if (word::is_empty (old))
644 return traits_type::adopt (nullptr, raw_pointer (old));
645 block_type *const block = word::block_of (old);
646 if (!word::is_holder (old))
647 return traits_type::adopt (block, block_pointer (block, old));
648 holder_type const *const holder = static_cast<holder_type const *> (block);
649 block_type *const owner = holder->owner ();
650 element_type *const stored
651 = static_cast<element_type *> (holder->pointer ());
652 Policy::at (split_point_t::load_count);
653 take_owner_reference<traits_type::mirror_holder> (block, owner);
654 Pointer result = traits_type::adopt (owner, stored);
655 Policy::at (split_point_t::release);
656 holder_ledger_type::release (block);
657 return result;
658 }

666 split_value_t
667 pin (std::memory_order order) const LUMEX_NOEXCEPT
668 {
669 Policy::at (split_point_t::load_read);
670 split_value_t seen = Policy::guess (word_);
671 for (;;)
672 {
673 if (word::is_empty (seen))
674 {
675 Policy::at (split_point_t::load_tick);
676 split_value_t const found = word_.compare_exchange_strong (
677 seen, seen, order, std::memory_order_relaxed);
678 if (found == seen)
679 return seen;
680 seen = found;
681 continue;
682 }
683 if (word::ticks_of (seen) >= Policy::tick_limit ())
684 {
685 Policy::back_off ();
686 Policy::at (split_point_t::load_read);
687 seen = word_.load (std::memory_order_relaxed);
688 continue;
689 }
690 Policy::at (split_point_t::load_tick);
691 split_value_t const found = word_.compare_exchange_strong (
692 seen, word::with_tick (seen), order, std::memory_order_relaxed);
693 if (found == seen)
694 {
695 Policy::note (split_event_t::tick, 1u);
696 Policy::ticked (word::ticks_of (seen) + 1u);
697 return seen;
698 }
699 seen = found;
700 }
701 }

710 void
711 count_and_settle (split_value_t pinned,
712 block_type *owner) const LUMEX_NOEXCEPT
713 {
714 Policy::at (split_point_t::load_count);
715 std::uint64_t const tag = pinned.lo;
716 split_value_t seen = Policy::guess (word_);
717 for (;;)
718 {
719 if (seen.lo == tag && word::ticks_of (seen) != 0u)
720 {
721 ledger_type::add (owner);
722 settle (pinned);
723 return;
724 }
725 seen = word_.load (std::memory_order_acquire);
726 if (seen.lo == tag && word::ticks_of (seen) != 0u)
727 continue;
728 Policy::note (split_event_t::settle_ext, 1u);
729 take_and_pay (pinned, owner);
730 return;
731 }
732 }

741 void
742 settle (split_value_t pinned) const LUMEX_NOEXCEPT
743 {
744 std::uint64_t const tag = pinned.lo;
745 Policy::at (split_point_t::settle_read);
746 split_value_t seen = Policy::guess (word_);
747 for (;;)
748 {
749 if (seen.lo == tag && word::ticks_of (seen) != 0u)
750 {
751 Policy::at (split_point_t::settle_untick);
752 split_value_t const found = word_.compare_exchange_strong (
753 seen, word::without_tick (seen), std::memory_order_release,
754 std::memory_order_relaxed);
755 if (found == seen)
756 {
757 Policy::note (split_event_t::untick, 1u);
758 return;
759 }
760 seen = found;
761 continue;
762 }
763 Policy::at (split_point_t::settle_read);
764 seen = word_.load (std::memory_order_acquire);
765 if (seen.lo == tag && word::ticks_of (seen) != 0u)
766 continue;
767 Policy::note (split_event_t::settle_ext, 1u);
768 pay (pinned, 1u);
769 return;
770 }

782 swapped_t
783 swap_out (split_value_t image, std::memory_order order) LUMEX_NOEXCEPT
784 {
785 Policy::at (split_point_t::swap);
786 split_value_t seen = Policy::guess (word_);
787 for (;;)
788 {
789 if (word::is_empty (seen) || word::ticks_of (seen) == 0u)
790 {
791 Policy::at (split_point_t::swap);
792 split_value_t const found = word_.compare_exchange_strong (
793 seen, image, order, std::memory_order_relaxed);
794 if (found == seen)
795 {
796 swapped_t const result = { seen, 0u };
797 return result;
798 }
799 seen = found;
800 continue;
801 }
802 if (word::ticks_of (seen) >= Policy::tick_limit ())
803 {
804 Policy::back_off ();
805 Policy::at (split_point_t::swap);
806 seen = word_.load (std::memory_order_relaxed);
807 continue;
808 }
809 Policy::at (split_point_t::load_tick);
810 split_value_t const ticked = word_.compare_exchange_strong (
811 seen, word::with_tick (seen), order, std::memory_order_relaxed);
812 if (!(ticked == seen))
813 {
814 seen = ticked;
815 continue;
816 }
817 Policy::note (split_event_t::tick, 1u);
818 Policy::ticked (word::ticks_of (seen) + 1u);
819 split_value_t const pinned = seen;
820 split_value_t current = word::with_tick (seen);
821 std::uint32_t moved = word::ticks_of (seen) + Policy::reserve ();
822 deposit (current, moved);
823 for (;;)
824 {
825 Policy::at (split_point_t::swap);
826 split_value_t const found = word_.compare_exchange_strong (
827 current, image, order, std::memory_order_acquire);
828 if (found == current)
829 {
830 // The writer's own tick left with the word: consumed.
831 Policy::note (split_event_t::consumed, 1u);
832 swapped_t const result
833 = { current, moved - (word::ticks_of (current) - 1u) };
834 return result;
835 }
836 if (found.lo != pinned.lo || word::ticks_of (found) == 0u)
837 {
838 // The block was swapped out by another writer, who deposited
839 // for the own tick: pay it and the own deposit at once.
840 pay_back (current, moved);
841 seen = found;
842 break;
843 }
844 // The same block with other ticks (any installation of it): top
845 // up the deposit when the ticks outgrew it, never lower it.
846 std::uint32_t const need = word::ticks_of (found) - 1u;
847 if (need > moved)
848 {
849 std::uint32_t const total = need + Policy::reserve ();
850 deposit (found, total - moved);
851 moved = total;
852 }
853 current = found;
854 }
855 }

858 bool
859 compare_exchange (Pointer &expected, Pointer &desired, bool weak,
860 std::memory_order success,
861 std::memory_order failure) LUMEX_NOEXCEPT
862 {
863 ::lumex::core::atomic::dwcas::require_dwcas ();
864 block_type *const expected_block = traits_type::control (expected);
865 element_type *const expected_stored = traits_type::stored (expected);
866 split_value_t want = word::empty_word (word::address_of (expected_stored));
867 if (expected_block != nullptr)
868 {
869 std::uint64_t const offset
870 = word::address_of (expected_stored)
871 - static_cast<std::uint64_t> (expected_block->anchor ());
872 if (!word::fits (offset))
873 return compare_exchange_holder (expected, desired, weak, success,
874 failure);
875 want = word::block_word (expected_block, offset);
876 }
877 bool built = false;
878 split_value_t image = word::empty_word (0u);
879 Policy::at (split_point_t::cas_read);
880 split_value_t seen = Policy::guess (word_);
881 for (;;)
882 {
883 if (word::same_value (seen, want))
884 {
885 if (!built)
886 {
887 image = take_image (desired);
888 built = true;
889 }
890 std::uint32_t const ticks
891 = word::is_empty (seen) ? 0u : word::ticks_of (seen);
892 // S9: no pin, `expected` keeps the block alive.
893 deposit (seen, ticks);
894 Policy::at (split_point_t::cas);
895 split_value_t const found = word_.compare_exchange_strong (
896 seen, image, rmw_order (success), std::memory_order_relaxed);
897 if (found == seen)
898 {
899 drop_word (seen);
900 return true;
901 }
902 withdraw (seen, ticks);
903 seen = found;
904 if (!weak)
905 continue;
906 }
907 // Not equivalent, or the one attempt of the weak form failed: the
908 // failure is this load, and a strong form whose value came back
909 // tries again.
910 Pointer current = load (failure);
911 if (!weak && equivalent (current, expected))
912 {
913 Policy::at (split_point_t::cas_read);
914 seen = Policy::guess (word_);
915 continue;
916 }
917 if (built)
918 drop_word (image);
919 expected = std::move (current);
920 return false;
921 }
922 }

932 bool
933 compare_exchange_holder (Pointer &expected, Pointer &desired, bool weak,
934 std::memory_order success,
935 std::memory_order failure) LUMEX_NOEXCEPT
936 {
937 block_type const *const expected_block = traits_type::control (expected);
938 void const *const expected_pointer
939 = split_erase (traits_type::stored (expected));
940 bool built = false;
941 split_value_t image = word::empty_word (0u);
942 Policy::at (split_point_t::cas_read);
943 split_value_t seen = Policy::guess (word_);
944 for (;;)
945 {
946 if (word::is_holder (seen))
947 {
948 if (word::ticks_of (seen) >= Policy::tick_limit ())
949 {
950 Policy::back_off ();
951 Policy::at (split_point_t::cas_read);
952 seen = word_.load (std::memory_order_relaxed);
953 continue;
954 }
955 Policy::at (split_point_t::cas_pin);
956 split_value_t const found = word_.compare_exchange_strong (
957 seen, word::with_tick (seen), rmw_order (success),
958 std::memory_order_relaxed);

1011 }

1013 }

1021 seen = Policy::guess (word_);
1022 continue;
1023 }
1024 if (built)
1025 drop_word (image);
1026 expected = std::move (current);
1027 return false;
1028 }
1029 }
