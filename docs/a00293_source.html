196// CWG 2118 friend-injection loophole (Boost.PFR / Alexandr Poltavsky shape).
197// MSVC may instantiate the converting operator for both a field type U and a
198// type convertible into U (e.g. int and std::optional<int>). Defining the
199// friend body on every hit yields C2084. Probe whether loophole_fn is already
200// declared for (T, N); only the first successful U gets a body. operator U&()
201// const&& (not operator U()) is required so optional-like fields resolve to
202// the wrapper type, not the contained type.
203template <typename T, std::size_t N> struct loophole_tag
204{
205 friend auto loophole_fn (loophole_tag<T, N>);
206};

251// Defines loophole_fn for every field of T. A static_assert of a class
252// template is instantiated with the class, so a base of this type injects
253// the fields before its derived class reads them. Without it only
254// aggregate_traits<T> (tuple_size<T>) injected them, and a get<I> before
255// the first tuple_size<T> of a translation unit did not compile.

257{

261 "field types of the aggregate are injected");

301template <typename T> struct aggregate_traits
302{
303 LUMEX_STATIC_ASSERT_MSG (std::is_class<T>::value && !std::is_union<T>::value,
304 "field reflection requires a non-union class type");
305 LUMEX_STATIC_ASSERT_MSG (
306 !std::is_polymorphic<T>::value,
307 "field reflection does not support polymorphic types");
308#if __cplusplus >= 201703L
309 LUMEX_STATIC_ASSERT_MSG (std::is_aggregate<T>::value,
310 "field reflection requires an aggregate type");
311#endif
312 LUMEX_STATIC_ASSERT_MSG (can_construct_n<T, 0>::value,
313 "field reflection requires brace initialization");
314 LUMEX_STATIC_ASSERT_MSG (
315 !can_construct_n<T, k_max_aggregate_fields + 1>::value,
316 "aggregate has more than 32 fields");
317

319 = count_fields_impl<T, 0, k_max_aggregate_fields>::value;
320
321#if __cplusplus >= 201402L && !LUMEX_AGGREGATE_FIELDS_USE_SB
322 typedef inject_fields<T, count> injected_t;
323 static const std::size_t injected = injected_t::trigger;

325 LUMEX_STATIC_ASSERT_MSG (
326 count == 0
327 || (injected > 0 && computed_sizeof<T, count>::value == sizeof (T)),
328 "field reflection layout mismatch (no bit-fields, no "
329 "reference members, no base classes)");
330#endif

335{

340inline void
341copy_parsed_name (char *dest, std::size_t dest_size,
342 char const *pretty) LUMEX_NOEXCEPT
343{
344 std::size_t len = 0;
345 while (pretty[len] != '\0')
346 ++len;
347
348 // GCC/Clang pretty strings use `.member`; MSVC __FUNCSIG__ uses
349 // `->member` (`...storage<T>->value->id`). Prefer the last arrow, then
350 // the last dot. Falling back to `:` alone picks the aggregate type name
351 // on MSVC (`...::Plain,&...->value->id`) and is wrong.
352 std::size_t arrow = static_cast<std::size_t> (-1);
353 std::size_t dot = static_cast<std::size_t> (-1);
354 std::size_t colon = static_cast<std::size_t> (-1);
355 for (std::size_t i = 0; i < len; ++i)
356 {
357 if (pretty[i] == '.')
358 dot = i;
359 if (pretty[i] == ':')
360 colon = i;
361 if (i + 1 < len && pretty[i] == '-' && pretty[i + 1] == '>')
362 arrow = i;
363 }
364
365 std::size_t start = 0;
366 if (arrow != static_cast<std::size_t> (-1))
367 start = arrow + 2;
368 else if (dot != static_cast<std::size_t> (-1))
369 start = dot + 1;
370 else if (colon != static_cast<std::size_t> (-1))
371 start = colon + 1;
372
373 std::size_t out = 0;
374 while (start < len && is_ident_char (pretty[start]) && out + 1 < dest_size)
375 dest[out++] = pretty[start++];
376 dest[out] = '\0';
377}
