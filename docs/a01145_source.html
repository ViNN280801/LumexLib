85// The record a retired object carries, as a member (not a base class) so the
86// names of node_t do not leak into the protected class. Copying or moving an
87// object makes a fresh, not retired node; assigning one object to another
88// changes nothing. The class is standard-layout with the node as its only
89// member, so the two pointers convert by reinterpret_cast.
90class node_base
91{
92protected:
93 node_base () LUMEX_NOEXCEPT { engine::init_node (node_); }
94 node_base (node_base const &) LUMEX_NOEXCEPT { engine::init_node (node_); }
95 node_base (node_base &&) LUMEX_NOEXCEPT { engine::init_node (node_); }
96 node_base &
97 operator= (node_base const &) LUMEX_NOEXCEPT
98 {
99 return *this;
100 }
101 node_base &
102 operator= (node_base &&) LUMEX_NOEXCEPT
103 {
104 return *this;
105 }
106 ~node_base () = default;
107
108 engine::node_t *
109 node () LUMEX_NOEXCEPT
110 {
111 return &node_;
112 }
113
114 engine::node_t const *
115 node () const LUMEX_NOEXCEPT
116 {
117 return &node_;
118 }
119
120 static node_base *
121 from_node (engine::node_t *node) LUMEX_NOEXCEPT
122 {
123 return reinterpret_cast<node_base *> (node);
124 }
125
126private:
127 engine::node_t node_;
128};

189// Declared, never defined. The first overload deduces the object and deleter
190// type of the only base of the form hazard_pointer_obj_base<U, D> (a second
191// one or none make the deduction fail) and takes part only when T is that U,
192// the base is accessible (is_convertible honors access, which a call in
193// decltype does not on every compiler: a private or protected base must make
194// the overload vanish, not the program ill-formed) and the conversion back
195// from the base exists (a virtual base has none).
196template <class T, class U, class D,
197 class = decltype (static_cast<U *> (
198 std::declval<hazard_pointer_obj_base<U, D> *> ()))>
199typename std::enable_if<
200 std::is_same<U, typename std::remove_cv<T>::type>::value
201 && std::is_convertible<
202 T *, hazard_pointer_obj_base<U, D> const volatile *>::value,
203 std::true_type>::type
204protectable_probe (hazard_pointer_obj_base<U, D> const volatile *);
205
206template <class T> std::false_type protectable_probe (...);
207
208// The object and deleter type of the base of a protectable T.
209template <class U, class D>
210base_pair<U, D> base_of_probe (hazard_pointer_obj_base<U, D> const volatile *);

212template <class T,

214struct protectable_base

216 typedef void type;

219template <class T> struct protectable_base<T, true>
