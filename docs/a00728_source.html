430 template <class Comp>
431 static bool
432 compare_eq (
433 XPathAstNode *lhs, // NOLINT(readability-function-cognitive-complexity)
434 XPathAstNode *rhs, XPathContext const &ctx, XPathStack const &stack,
435 Comp const &comp)
436 {
437 xpath_value_type lt_ = lhs->rettype ();
438 xpath_value_type rt_ = rhs->rettype ();
439
440 if (lt_ != xpath_type_node_set && rt_ != xpath_type_node_set)
441 {
442 if (lt_ == xpath_type_boolean || rt_ == xpath_type_boolean)
443 return comp (lhs->eval_boolean (ctx, stack),
444 rhs->eval_boolean (ctx, stack));
445 if (lt_ == xpath_type_number || rt_ == xpath_type_number)
446 return comp (lhs->eval_number (ctx, stack),
447 rhs->eval_number (ctx, stack));
448 if (lt_ == xpath_type_string || rt_ == xpath_type_string)
449 {
450 XPathAllocatorCapture capture (stack.result);
451
452 XPathString ls_ = lhs->eval_string (ctx, stack);
453 XPathString rs_ = rhs->eval_string (ctx, stack);
454
455 return comp (ls_, rs_);
456 }
457 }
458 else if (lt_ == xpath_type_node_set && rt_ == xpath_type_node_set)
459 {
460 XPathAllocatorCapture capture (stack.result);
461
462 XPathNodeSetRaw ls_
463 = lhs->eval_node_set (ctx, stack, nodeset_eval_all);
464 XPathNodeSetRaw rs_
465 = rhs->eval_node_set (ctx, stack, nodeset_eval_all);
466
467 for (XPathNode const *li = ls_.begin (); li != ls_.end (); ++li)
468 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)
469 {
470 XPathAllocatorCapture cri (stack.result);
471
472 if (comp (string_value (*li, stack.result),
473 string_value (*ri, stack.result)))
474 return true;
475 }
476
477 return false;
478 }
479 else
480 {
481 if (lt_ == xpath_type_node_set)
482 {
483 std::swap (lhs, rhs);
484 std::swap (lt_, rt_);
485 }
486
487 if (lt_ == xpath_type_boolean)
488 return comp (lhs->eval_boolean (ctx, stack),
489 rhs->eval_boolean (ctx, stack));
490 if (lt_ == xpath_type_number)
491 {
492 XPathAllocatorCapture capture (stack.result);
493
494 double tmp = lhs->eval_number (ctx, stack);
495 XPathNodeSetRaw rs_
496 = rhs->eval_node_set (ctx, stack, nodeset_eval_all);
497
498 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)
499 {
500 XPathAllocatorCapture cri (stack.result);
501
502 if (comp (tmp, convert_string_to_number (
503 string_value (*ri, stack.result).c_str ())))
504 return true;
505 }
506
507 return false;
508 }
509 if (lt_ == xpath_type_string)
510 {
511 XPathAllocatorCapture capture (stack.result);
512
513 XPathString tmp = lhs->eval_string (ctx, stack);
514 XPathNodeSetRaw rs_
515 = rhs->eval_node_set (ctx, stack, nodeset_eval_all);
516
517 for (XPathNode const *ri = rs_.begin (); ri != rs_.end (); ++ri)
518 {
519 XPathAllocatorCapture cri (stack.result);
520 if (comp (tmp, string_value (*ri, stack.result)))
521 return true;
522 }
523
524 return false;
525 }
526 }
527
528 LUMEX_ASSERT (false && "Wrong types"); // unreachable
529 return false;
530 }

803 template <class T>
804 void
805 step_fill (XPathNodeSetRaw
806 &nsr, // NOLINT(readability-function-cognitive-complexity)
807 XmlNodeBase *node, XPathAllocator *alloc, bool once, T /*val*/)
808 {
809 axis_t const axis = T::axis;
810
811 switch (axis)
812 {
813 case axis_attribute:
814 {
815 for (XmlAttributeBase *attr = node->first_attribute; attr;
816 attr = attr->next_attribute)
817 if (step_push (nsr, attr, node, alloc) & once)
818 return;
819
820 break;
821 }
822
823 case axis_child:
824 {
825 for (XmlNodeBase *ctx = node->first_child; ctx;
826 ctx = ctx->next_sibling)
827 if (step_push (nsr, ctx, alloc) & once)
828 return;
829
830 break;
831 }
832
833 case axis_descendant:
834 case axis_descendant_or_self:
835 {
836 if (axis == axis_descendant_or_self)
837 if (step_push (nsr, node, alloc) & once)
838 return;
839
840 XmlNodeBase *cur = node->first_child;
841
842 while (cur)
843 {
844 if (step_push (nsr, cur, alloc) & once)
845 return;
846
847 if (cur->first_child)
848 cur = cur->first_child;
849 else
850 {
851 while (!cur->next_sibling)
852 {
853 cur = cur->parent;
854
855 if (cur == node)
856 return;
857 }
858
859 cur = cur->next_sibling;
860 }
861 }
862
863 break;
864 }
865
866 case axis_following_sibling:
867 {
868 for (XmlNodeBase *ctx = node->next_sibling; ctx;
869 ctx = ctx->next_sibling)
870 if (step_push (nsr, ctx, alloc) & once)
871 return;
872
873 break;
874 }
875
876 case axis_preceding_sibling:
877 {
878 for (XmlNodeBase *ctx = node->prev_sibling_c; ctx->next_sibling;
879 ctx = ctx->prev_sibling_c)
880 if (step_push (nsr, ctx, alloc) & once)
881 return;
882
883 break;
884 }
885
886 case axis_following:
887 {
888 XmlNodeBase *cur = node;
889
890 // exit from this node so that we don't include descendants
891 while (!cur->next_sibling)
892 {
893 cur = cur->parent;
894
895 if (!cur)
896 return;
897 }
898
899 cur = cur->next_sibling;
900
901 while (cur)
902 {
903 if (step_push (nsr, cur, alloc) & once)
904 return;
905
906 if (cur->first_child)
907 cur = cur->first_child;
908 else
909 {
910 while (!cur->next_sibling)
911 {
912 cur = cur->parent;
913
914 if (!cur)
915 return;
916 }
917
918 cur = cur->next_sibling;
919 }
920 }
921
922 break;
923 }
924
925 case axis_preceding:
926 {
927 XmlNodeBase *cur = node;
928
929 // exit from this node so that we don't include descendants
930 while (!cur->prev_sibling_c->next_sibling)
931 {
932 cur = cur->parent;
933
934 if (!cur)
935 return;
936 }
937
938 cur = cur->prev_sibling_c;
939
940 while (cur)
941 {
942 if (cur->first_child)
943 cur = cur->first_child->prev_sibling_c;
944 else
945 {
946 // leaf node, can't be ancestor
947 if (step_push (nsr, cur, alloc) & once)
948 return;
949
950 while (!cur->prev_sibling_c->next_sibling)
951 {
952 cur = cur->parent;
953
954 if (!cur)
955 return;
956
957 if (!node_is_ancestor (cur, node))
958 if (step_push (nsr, cur, alloc) & once)
959 return;
960 }
961
962 cur = cur->prev_sibling_c;
963 }
964 }
965
966 break;
967 }
968
969 case axis_ancestor:
970 case axis_ancestor_or_self:
971 {
972 if (axis == axis_ancestor_or_self)
973 if (step_push (nsr, node, alloc) & once)
974 return;
975
976 XmlNodeBase *cur = node->parent;
977
978 while (cur)
979 {
980 if (step_push (nsr, cur, alloc) & once)
981 return;
982
983 cur = cur->parent;
984 }
985
986 break;
987 }
988
989 case axis_self:
990 {
991 step_push (nsr, node, alloc);
992
993 break;
994 }
995
996 case axis_parent:
997 {
998 if (node->parent)
999 step_push (nsr, node->parent, alloc);
1000
1001 break;
1002 }
1003
1004 default:
1005 LUMEX_ASSERT (false && "Unimplemented axis"); // unreachable
1006 }
1007 }

1030 template <class T>
1031 void
1032 step_fill (XPathNodeSetRaw
1033 &nsr, // NOLINT(readability-function-cognitive-complexity)
1034 XmlAttributeBase *attr, XmlNodeBase *node, XPathAllocator *alloc,
1035 bool once, T val)
1036 {
1037 axis_t const axis = T::axis;
1038
1039 switch (axis)
1040 {
1041 case axis_ancestor:
1042 case axis_ancestor_or_self:
1043 {
1044 if (axis == axis_ancestor_or_self
1045 && m_test == nodetest_type_node) // reject attributes based on
1046 // principal node type test
1047 if (step_push (nsr, attr, node, alloc) & once)
1048 return;
1049
1050 XmlNodeBase *cur = node;
1051
1052 while (cur)
1053 {
1054 if (step_push (nsr, cur, alloc) & once)
1055 return;
1056
1057 cur = cur->parent;
1058 }
1059
1060 break;
1061 }
1062
1063 case axis_descendant_or_self:
1064 case axis_self:
1065 {
1066 if (m_test == nodetest_type_node) // reject attributes based on
1067 // principal node type test
1068 step_push (nsr, attr, node, alloc);
1069
1070 break;
1071 }
1072
1073 case axis_following:
1074 {
1075 XmlNodeBase *cur = node;
1076
1077 while (cur)
1078 {
1079 if (cur->first_child)
1080 cur = cur->first_child;
1081 else
1082 {
1083 while (!cur->next_sibling)
1084 {
1085 cur = cur->parent;
1086
1087 if (!cur)
1088 return;
1089 }
1090
1091 cur = cur->next_sibling;
1092 }
1093
1094 if (step_push (nsr, cur, alloc) & once)
1095 return;
1096 }
1097
1098 break;
1099 }
1100
1101 case axis_parent:
1102 {
1103 step_push (nsr, node, alloc);
1104
1105 break;
1106 }
1107
1108 case axis_preceding:
1109 {
1110 // preceding:: axis does not include attribute nodes and attribute
1111 // ancestors (they are the same as parent's ancestors), so we can
1112 // reuse node preceding
1113 step_fill (nsr, node, alloc, once, val);
1114 break;
1115 }
1116
1117 default:
1118 LUMEX_ASSERT (false && "Unimplemented axis"); // unreachable
1119 }
1120 }

1143 {

1174 template <class T>
1175 XPathNodeSetRaw
1176 step_do (XPathContext const &ctx, XPathStack const &stack,
1177 Types::nodeset_eval_t eval, T val)
1178 {
1179 axis_t const axis = T::axis;
1180 bool const axis_reverse
1181 = (axis == axis_ancestor || axis == axis_ancestor_or_self
1182 || axis == axis_preceding || axis == axis_preceding_sibling);
1183 XPathNodeSet::type_t const axis_type
1184 = axis_reverse ? XPathNodeSet::type_sorted_reverse
1185 : XPathNodeSet::type_sorted;
1186
1187 bool once = (axis == axis_attribute && m_test == nodetest_name)
1188 || (!m_right && eval_once (axis_type, eval)) ||
1189 // coverity[mixed_enums]
1190 (m_right && !m_right->m_next
1191 && m_right->m_test == predicate_constant_one);
1192
1193 XPathNodeSetRaw nsr;
1194 nsr.set_type (axis_type);
1195
1196 if (m_left)
1197 {
1198 XPathNodeSetRaw tmp
1199 = m_left->eval_node_set (ctx, stack, nodeset_eval_all);
1200
1201 // self axis preserves the original order
1202 if (axis == axis_self)
1203 nsr.set_type (tmp.type ());
1204
1205 for (XPathNode const *it = tmp.begin (); it != tmp.end (); ++it)
1206 {
1207 std::size_t size = nsr.size ();
1208
1209 // in general, all axes generate elements in a particular order,
1210 // but there is no order guarantee if axis is applied to two nodes
1211 if (axis != axis_self && size != 0)
1212 nsr.set_type (XPathNodeSet::type_unsorted);
1213
1214 step_fill (nsr, *it, stack.result, once, val);
1215 if (m_right)
1216 apply_predicates (nsr, size, stack, eval);
1217 }
1218 }
1219 else
1220 {
1221 step_fill (nsr, ctx.node, stack.result, once, val);
1222 if (m_right)
1223 apply_predicates (nsr, 0, stack, eval);
1224 }
1225
1226 // child, attribute and self axes always generate unique set of nodes
1227 // for other axis, if the set stayed sorted, it stayed unique because the
1228 // traversal algorithms do not visit the same node twice
1229 if (axis != axis_child && axis != axis_attribute && axis != axis_self
1230 && nsr.type () == XPathNodeSet::type_unsorted)
1231 nsr.remove_duplicates (stack.temp);
1232
1233 return nsr;
1234 }
