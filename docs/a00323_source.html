317template <typename Target, typename Source>
318bool
319fits (Source value, std::true_type, std::true_type)
320{
321 return !cmp_less (value, std::numeric_limits<Target>::lowest ())
322 && !cmp_less ((std::numeric_limits<Target>::max) (), value);
323}

330template <typename Target, typename Source>
331bool
332fits (Source value, std::false_type, std::true_type)
333{
334 long double const v = static_cast<long double> (value);
335 long double const lowest
336 = static_cast<long double> (std::numeric_limits<Target>::lowest ());
337 long double const highest
338 = static_cast<long double> ((std::numeric_limits<Target>::max) ());
339 long double const bound
340 = std::ldexp (1.0L, std::numeric_limits<Target>::digits);
341 return !(v < lowest) && !(v > highest) && v < bound;
342}

352 && !(v > static_cast<long double> (
353 (std::numeric_limits<Target>::max) ()));

378template <typename It1, typename End1, typename It2, typename End2>
379void
380require_same_size (char const *function, It1 &it1, End1 const &last1, It2 &it2,
381 End2 const &last2, std::size_t count)
382{
383 if (!(it1 != last1) && !(it2 != last2))
384 return;
385
386 std::size_t size1 = count;
387 std::size_t size2 = count;
388 for (; it1 != last1; ++it1)
389 ++size1;
390 for (; it2 != last2; ++it2)
391 ++size2;
392 throw LumexMathSizeMismatchException (
393 std::string (function) + ": size of both ranges must be equal ("
394 + std::to_string (size1) + " vs " + std::to_string (size2) + ")");
395}

425template <typename SourceType, typename TargetType, typename Name>
426LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
427typename std::enable_if<traits::is_numeric<SourceType>::value
428 && traits::is_numeric<TargetType>::value,
429 TargetType>::type
430 checked_narrow_cast (SourceType sourceValue, Name const &fieldName)
431{
432 Detail::throw_if_not_finite (
433 sourceValue, fieldName,
434 std::integral_constant<bool,
435 std::is_floating_point<SourceType>::value> ());
436
437 bool const inRange = Detail::fits<TargetType> (
438 sourceValue,
439 std::integral_constant<bool, std::is_integral<SourceType>::value> (),
440 std::integral_constant<bool, std::is_integral<TargetType>::value> ());
441 if (!inRange)
442 {
443 std::ostringstream oss;
444 oss << "Field '" << fieldName << "' value ";
445 Detail::print_number (oss, sourceValue);
446 oss << " is out of range [";
447 Detail::print_number (oss, std::numeric_limits<TargetType>::lowest ());
448 oss << "; ";
449 Detail::print_number (oss, (std::numeric_limits<TargetType>::max) ());
450 oss << "]";
451 throw std::out_of_range (oss.str ());
452 }
453
454 return static_cast<TargetType> (sourceValue);
455}

498template <typename Range>
499typename Detail::numeric_range_value<Range>::type
500avg (Range &&range)
501{
502 typedef typename Detail::numeric_range_value<Range>::type ValueType;
503
504 ValueType sum = ValueType (0);
505 std::size_t count = 0;
506 auto it = Detail::adl_begin (range);
507 auto const last = Detail::adl_end (range);
508 for (; it != last; ++it)
509 {
510 sum += *it;
511 ++count;
512 }
513
514 return (count == 0)
515 ? ValueType (0)
516 : static_cast<ValueType> (sum / static_cast<ValueType> (count));
517}

584template <typename Range1, typename Range2>
585typename Detail::range_pair_result<Range1, Range2>::type
586rms (Range1 &&first, Range2 &&second)
587{
588 typedef
589 typename Detail::range_pair_result<Range1, Range2>::common CommonType;
590 typedef typename Detail::range_pair_result<Range1, Range2>::type ResultType;
591
592 auto it1 = Detail::adl_begin (first);
593 auto const last1 = Detail::adl_end (first);
594 auto it2 = Detail::adl_begin (second);
595 auto const last2 = Detail::adl_end (second);
596 CommonType sumOfProducts = CommonType (0);
597 std::size_t count = 0;
598 for (; it1 != last1 && it2 != last2; ++it1, ++it2)
599 {
600 sumOfProducts
601 += static_cast<CommonType> (*it1) * static_cast<CommonType> (*it2);
602 ++count;
603 }
604 Detail::require_same_size ("rms", it1, last1, it2, last2, count);
605
606 if (count == 0)
607 return ResultType (0);
608 return static_cast<ResultType> (
609 std::sqrt (sumOfProducts / static_cast<ResultType> (count)));
610}
