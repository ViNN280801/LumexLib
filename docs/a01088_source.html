197// Packing control macros for MSVC compatibility
198// @details These macros provide portable packing control across GCC/Clang and
199// MSVC
200// @note GCC/Clang support #pragma pack for MSVC compatibility, so these macros
201// work on all platforms
202// @usage:
203// LUMEX_PACK_BEGIN(1)
204// struct MyStruct {
205// char c;
206// int i;
207// };
208// LUMEX_PACK_END
209#if defined(_MSC_VER)
210#define LUMEX_PACK_BEGIN(alignment) __pragma (pack (push, alignment))
211#define LUMEX_PACK_END __pragma (pack (pop))
212#elif defined(__GNUC__) || defined(__clang__)
213// GCC/Clang support MSVC-style #pragma pack for compatibility
214// GCC 4.9+ and Clang support __pragma (same as MSVC)
215#if (defined(__GNUC__) \
216 && (__GNUC__ >= 5 || (__GNUC__ == 4 && __GNUC_MINOR__ >= 9))) \
217 || defined(__clang__)
218#define LUMEX_PACK_BEGIN(alignment) __pragma (pack (push, alignment))
219#define LUMEX_PACK_END __pragma (pack (pop))
220#else
221// Fallback to _Pragma for older GCC versions
222// Define common alignment values directly
223#define LUMEX_PACK_BEGIN_1 _Pragma ("pack(push, 1)")
224#define LUMEX_PACK_BEGIN_2 _Pragma ("pack(push, 2)")
225#define LUMEX_PACK_BEGIN_4 _Pragma ("pack(push, 4)")
226#define LUMEX_PACK_BEGIN_8 _Pragma ("pack(push, 8)")
227#define LUMEX_PACK_BEGIN(alignment) LUMEX_PACK_BEGIN_##alignment
228#define LUMEX_PACK_END _Pragma ("pack(pop)")
229#endif
230#else
231#define LUMEX_PACK_BEGIN(alignment)
232#define LUMEX_PACK_END
233#endif

271// [[maybe_unused]]
272// Variadic so commas inside template args / call args stay one argument.
273//
274// A `(void)` cast discards a value on purpose, and Clang and the C++17
275// [[nodiscard]] accept it for a result they would otherwise report. Below
276// C++17 LUMEX_ATTRIBUTE_NODISCARD is __attribute__ ((warn_unused_result)),
277// and GCC still reports (-Wunused-result) a result that is only cast to void.
278// There the expression becomes the left operand of a comma whose right operand
279// is a sink: the overloaded operator takes the value as an argument, which
280// counts as using it, and does nothing with it. The sink works for any
281// expression, a void one (the built-in comma) and a name, an array or a
282// function too, and evaluates it once.
283#if __cplusplus < 201703L && defined(__GNUC__) && !defined(__clang__)
284#define LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR(...) \
285 (void)((__VA_ARGS__), \
286 ::lumex::core::utility::attr::detail::unused_value_sink_t ())
287#else
288#define LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR(...) (void)(__VA_ARGS__)
289#endif
290#if __cplusplus >= 201703L
291#define LUMEX_ATTRIBUTE_MAYBE_UNUSED [[maybe_unused]]
292#elif defined(__GNUC__) || defined(__clang__)
293#define LUMEX_ATTRIBUTE_MAYBE_UNUSED __attribute__ ((unused))
294#else
295#define LUMEX_ATTRIBUTE_MAYBE_UNUSED
296#endif

298// [[likely]] / [[unlikely]]
299// LUMEX_ATTRIBUTE_LIKELY_COND(cond) and LUMEX_ATTRIBUTE_UNLIKELY_COND(cond)
300// wrap a condition and exist in every standard. Below C++20 they pass the hint
301// through __builtin_expect on GCC and Clang (the condition is converted to
302// bool first, so a pointer works too) and are the plain condition elsewhere.
303// The statement attributes LUMEX_ATTRIBUTE_LIKELY and LUMEX_ATTRIBUTE_UNLIKELY
304// are empty below C++20: no compiler has a GNU spelling for them (GCC rejects
305// __attribute__ ((likely)) before a statement, Clang ignores it with a
306// warning).
307#if __cplusplus >= 202002L
308#define LUMEX_ATTRIBUTE_LIKELY [[likely]]
309#define LUMEX_ATTRIBUTE_UNLIKELY [[unlikely]]
310#define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (cond)
311#define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (cond)
312#else
313#define LUMEX_ATTRIBUTE_LIKELY
314#define LUMEX_ATTRIBUTE_UNLIKELY
315#if defined(__GNUC__) || defined(__clang__)
316#define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (__builtin_expect (!!(cond), 1))
317#define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (__builtin_expect (!!(cond), 0))
318#else
319#define LUMEX_ATTRIBUTE_LIKELY_COND(cond) (cond)
320#define LUMEX_ATTRIBUTE_UNLIKELY_COND(cond) (cond)
321#endif
322#endif
