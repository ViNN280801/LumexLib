1225 {
1226 if (full ())
1227 {
1228 // same as pop_front(): drop the oldest, make the buffer non-full
1229 alloc_traits::destroy (m_allocator, std::addressof (at_slot (0)));
1230 m_head = next_index (m_head);
1231 --m_size;
1232 }
1233 size_type const pos = physical_index (m_size); // buffer is no longer full
1234 alloc_traits::construct (m_allocator,
1235 reinterpret_cast<T *> (slot_ptr (pos)),
1236 std::forward<Args> (args)...);

1239 }

1257 reference

1259 {
1260 size_type const pos = (m_size == 0) ? m_head : prev_index (m_head);
1261
1262 if (full ())
1263 {
1264 // free the target slot: prev_index(m_head)
1265 alloc_traits::destroy (
1266 m_allocator,
1267 std::addressof (at_slot (0 + m_size - 1))); // former newest
1268 --m_size; // now capacity-1; head is not moved
1269 }
1270
1271 // construct at pos; invariants hold if it throws
1272 alloc_traits::construct (m_allocator,
1273 reinterpret_cast<T *> (slot_ptr (pos)),
1274 std::forward<Args> (args)...);
1275 m_head = (m_size == 0) ? m_head : pos;
1276 ++m_size;
1277 return *reinterpret_cast<T *> (slot_ptr (m_head));

1849 auto end1 = lhs.end ();

1855 return cmp;

1858 if (it1 == end1 && it2 == end2)
1859 return std::strong_ordering::equal;
1860 if (it1 == end1)
1861 { // lhs is shorter than rhs
1862 return std::strong_ordering::less;
1863 }
1864 // Otherwise (it2 == end2), rhs is shorter than lhs
