174template <typename T> struct default_return
175{
176 LUMEX_STATIC_ASSERT_MSG (
177 std::is_default_constructible<T>::value,
178 "Type must be default-constructible. For "
179 "non-default-constructible types, consider using "
180 "std::optional or providing an explicit default value.");
181
182 LUMEX_STATIC_ASSERT_MSG (
183 !std::is_reference<T>::value,
184 "Reference types cannot have default values. Use a pointer "
185 "or value type instead.");
186
187 LUMEX_STATIC_ASSERT_MSG (
188 !std::is_abstract<T>::value,
189 "Abstract classes cannot be instantiated. Use a pointer or a "
190 "concrete derived type instead.");
191
192 static LUMEX_CONSTEXPR T
193 value () LUMEX_NOEXCEPT_IF (std::is_nothrow_default_constructible<T>::value)
194 {
195 return T{};
196 }
197};
198

298template <typename T> struct invoke_impl
299{
300 template <typename Func, typename... Args>
301 static auto call (Func &&func, Args &&...args)
302 -> decltype (std::forward<Func> (func) (std::forward<Args> (args)...));
303};
304
305template <typename B, typename MT> struct invoke_impl<MT B::*>
306{
307 template <typename T, typename Td = typename std::decay<T>::type,
308 typename
309 = typename std::enable_if<std::is_base_of<B, Td>::value>::type>
310 static auto get (T &&arg) -> T &&;
311
312 template <typename T, typename Td = typename std::decay<T>::type,
313 typename
314 = typename std::enable_if<is_reference_wrapper<Td>::value>::type>
315 static auto get (T &&arg) -> decltype (arg.get ());
316
317 template <typename T, typename Td = typename std::decay<T>::type,
318 typename
319 = typename std::enable_if<!std::is_base_of<B, Td>::value>::type,
320 typename
321 = typename std::enable_if<!is_reference_wrapper<Td>::value>::type>
322 static auto get (T &&arg) -> decltype (*std::forward<T> (arg));
323
324 template <typename T, typename... Args, typename MT1,
325 typename
326 = typename std::enable_if<std::is_function<MT1>::value>::type>
327 static auto call (MT1 B::*pmf, T &&arg, Args &&...args)
328 -> decltype ((invoke_impl::get (std::forward<T> (arg))
329 .*pmf) (std::forward<Args> (args)...));
330
331 template <typename T>
332 static auto call (MT B::*pmd, T &&arg)
333 -> decltype (invoke_impl::get (std::forward<T> (arg)).*pmd);
334};
335
336template <typename Func, typename... Args,
337 typename FuncDecayed = typename std::decay<Func>::type>
338auto INVOKE (Func &&func, Args &&...args)
339 -> decltype (invoke_impl<FuncDecayed>::call (
340 std::forward<Func> (func), std::forward<Args> (args)...));
341
342// SFINAE core: if INVOKE(...) is well-formed, expose ::type =
343// decltype(INVOKE(...))
344template <typename AlwaysVoid, typename /*F*/, typename... /*Args*/>
345struct invoke_result_impl
346{ /* no ::type when ill-formed */
347};
348
349template <typename F, typename... Args>
350struct invoke_result_impl<meta::void_t<decltype (INVOKE (
351 std::declval<F> (), std::declval<Args> ()...))>,
352 F, Args...>
353{
354 using type
355 = decltype (INVOKE (std::declval<F> (), std::declval<Args> ()...));
356};
357} // namespace detail

788// `T::npos` is tested before `data ()`: substitution stops at the first
789// failure in lexical order, and libstdc++ declares `data ()` of

798 : std::integral_constant<
799 bool, std::is_same<typename T::value_type, Char>::value
800 && std::is_convertible<
801 decltype (std::declval<T const &> ().data ()),
802 Char const *>::value>
803{
804};
