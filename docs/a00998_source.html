137LUMEX_CONSTEXPR_FUNCTION char const *
138to_string (assertion_kind kind) LUMEX_NOEXCEPT
139{
140 return kind == assertion_kind::pre
141 ? "pre"
142 : (kind == assertion_kind::post
143 ? "post"
144 : (kind == assertion_kind::assert ? "assert" : "unknown"));
145}

148LUMEX_CONSTEXPR_FUNCTION char const *
149to_string (evaluation_semantic semantic) LUMEX_NOEXCEPT
150{
151 return semantic == evaluation_semantic::ignore
152 ? "ignore"
153 : (semantic == evaluation_semantic::observe
154 ? "observe"
155 : (semantic == evaluation_semantic::enforce
156 ? "enforce"
157 : (semantic == evaluation_semantic::quick_enforce
158 ? "quick_enforce"
159 : "unknown")));
160}
161
163LUMEX_CONSTEXPR_FUNCTION char const *
164to_string (detection_mode mode) LUMEX_NOEXCEPT
165{
166 return mode == detection_mode::predicate_false
167 ? "predicate_false"
168 : (mode == detection_mode::evaluation_exception
169 ? "evaluation_exception"
170 : "unknown");
171}

178LUMEX_CONSTEXPR_FUNCTION bool
179is_terminating (evaluation_semantic semantic) LUMEX_NOEXCEPT
180{
181 return semantic == evaluation_semantic::enforce
182 || semantic == evaluation_semantic::quick_enforce;
