431template <typename T> struct invoke_impl
432{
433 template <typename Func, typename... Args>
434 static auto call (Func &&func, Args &&...args)
435 -> decltype (std::forward<Func> (func) (std::forward<Args> (args)...));
436};
437
438template <typename B, typename MT> struct invoke_impl<MT B::*>
439{
440 template <typename T, typename Td = typename std::decay<T>::type,
441 typename
442 = typename std::enable_if<std::is_base_of<B, Td>::value>::type>
443 static auto get (T &&arg) -> T &&;
444
445 template <typename T, typename Td = typename std::decay<T>::type,
446 typename
447 = typename std::enable_if<is_reference_wrapper<Td>::value>::type>
448 static auto get (T &&arg) -> decltype (arg.get ());
449
450 template <typename T, typename Td = typename std::decay<T>::type,
451 typename
452 = typename std::enable_if<!std::is_base_of<B, Td>::value>::type,
453 typename
454 = typename std::enable_if<!is_reference_wrapper<Td>::value>::type>
455 static auto get (T &&arg) -> decltype (*std::forward<T> (arg));
456
457 template <typename T, typename... Args, typename MT1,
458 typename
459 = typename std::enable_if<std::is_function<MT1>::value>::type>
460 static auto call (MT1 B::*pmf, T &&arg, Args &&...args)
461 -> decltype ((invoke_impl::get (std::forward<T> (arg))
462 .*pmf) (std::forward<Args> (args)...));
463
464 // Pointers to data members only: for a pointer to a member function
465 // `obj.*pmd` is not a value, yet MinGW GCC 8 accepts `decltype` of it, so
466 // the overload would also match there and give the function type as the
467 // call result.
468 template <typename T, typename Member = MT,
469 typename
470 = typename std::enable_if<!std::is_function<Member>::value>::type>
471 static auto call (MT B::*pmd, T &&arg)
472 -> decltype (invoke_impl::get (std::forward<T> (arg)).*pmd);
473};
474
475template <typename Func, typename... Args,
476 typename FuncDecayed = typename std::decay<Func>::type>
477// The exposition-only name of the standard INVOKE; a lower-case invoke would
478// be found by ADL together with std::invoke and make calls ambiguous.
479// NOLINTNEXTLINE(readability-identifier-naming)
480auto INVOKE (Func &&func, Args &&...args)
481 -> decltype (invoke_impl<FuncDecayed>::call (
482 std::forward<Func> (func), std::forward<Args> (args)...));
483
484// SFINAE core: if INVOKE(...) is well-formed, expose ::type =
485// decltype(INVOKE(...))
486template <typename AlwaysVoid, typename /*F*/, typename... /*Args*/>
487struct invoke_result_impl
488{ /* no ::type when ill-formed */
489};
490
491template <typename F, typename... Args>
492struct invoke_result_impl<meta::void_t<decltype (INVOKE (
493 std::declval<F> (), std::declval<Args> ()...))>,
494 F, Args...>
495{
496 using type
497 = decltype (INVOKE (std::declval<F> (), std::declval<Args> ()...));
498};
499} // namespace detail
