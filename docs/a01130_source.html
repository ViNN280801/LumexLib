195template <typename Derived, typename Base,
196 typename std::enable_if<
197 std::is_pointer<Derived>::value
198 && Detail::is_valid_down_cast<Base, Derived>::value,
199 int>::type
200 = 0>
201LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
202Derived downcast (Base base)
203{
204 if (base == nullptr)
205 return nullptr; // p6: null in, null out - not a cast failure.
206
207 Derived result = dynamic_cast<Derived> (base);
208 if (result == nullptr)
209 throw bad_down_cast (typeid (*base),
210 typeid (typename std::remove_pointer<Derived>::type));
211 return result;
212}

225template <typename Derived, typename Base,
226 typename std::enable_if<
227 std::is_lvalue_reference<Derived>::value
228 && Detail::is_valid_down_cast<Base &, Derived>::value,
229 int>::type
230 = 0>
231LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
232Derived downcast (Base &base)
233{
234 // According to p9, std::bad_cast may be thrown.
235 try
236 {
237 return dynamic_cast<Derived> (base);
238 }
239 catch (std::bad_cast const &bc_exc)
240 {
241 throw bad_down_cast (
242 typeid (base),
243 typeid (typename std::remove_reference<Derived>::type),
244 bc_exc.what ());
245 }
246}
