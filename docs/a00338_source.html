156LUMEX_CONSTEXPR_FUNCTION inline bool
157equal_texts (char const *left, char const *right) LUMEX_NOEXCEPT
158{
159 // No shortcut for equal pointers: comparing the addresses of string
160 // literals is unspecified in a constant expression.
161 return *left == *right
162 && (*left == '\0' || equal_texts (left + 1, right + 1));
163}

262 && detail::equal_texts (left.m_file, right.m_file)
