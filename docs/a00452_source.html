299 // The protocol: announce, fence, reload through `load`, compare.
300 template <class T, class Load>
301 bool
302 announce_and_validate (T *&ptr, Load const &load) LUMEX_NOEXCEPT
303 {
304 static_assert (is_hazard_protectable<T>::value,
305 "T must have exactly one public, non-virtual base "
306 "hazard_pointer_obj_base<T, D>");
307 LUMEX_HAZARD_POINTER_DEBUG_ASSERT (slot_ != nullptr);
308 T *old = ptr;
309 announce (old == nullptr ? static_cast<void const *> (nullptr)
310 : node_address (old));
311 engine::reader_fence ();
312 LUMEX_HAZARD_POINTER_TEST_POINT (reader_announced);
313 ptr = load ();
314 if (old != ptr)
315 {
316 announce (nullptr);
317 return false;
318 }
319 return true;
320 }

322 void
323 announce (void const *node) LUMEX_NOEXCEPT

326 }
