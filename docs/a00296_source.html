679 LUMEX_CONSTEXPR_CXX14 void
680 swap (Expected &other)
681#if __cplusplus >= 201703L
682 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible<ErrorType>::value
683 &&std::is_nothrow_swappable<ErrorType>::value)
684#else
685 LUMEX_NOEXCEPT_IF (std::is_nothrow_move_constructible<ErrorType>::value)
686#endif
687 {
688 if (this == &other)
689 return;
690
691 if (m_has_value && other.m_has_value)
692 {
693 // Both hold void success; nothing to do
694 }
695 else if (!m_has_value && !other.m_has_value)
696 { // Both hold an error
697 std::swap (m_storage.m_error, other.m_storage.m_error);
698 }
699 else
700 { // One holds void success, the other an error. Move is required.
701 if (m_has_value)
702 { // this holds void success, other holds an error
703 new (std::addressof (m_storage.m_error))
704 ErrorType (std::move (other.m_storage.m_error));
705 other.m_storage.m_error.~ErrorType ();
706 // other now holds void success
707 }
708 else
709 { // this holds an error, other holds void success
710 new (std::addressof (other.m_storage.m_error))
711 ErrorType (std::move (m_storage.m_error));
712 m_storage.m_error.~ErrorType ();
713 // this now holds void success
714 }
715 std::swap (m_has_value, other.m_has_value);
716 }
717 }
