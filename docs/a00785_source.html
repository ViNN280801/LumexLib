91template <class T, class A, class... Args>
92shared_ptr<T>
93allocate_shared (A const &alloc, Args &&...args)
94{
95 static_assert (!std::is_array<T>::value,
96 "smart_ptr: make_shared / allocate_shared of an array "
97 "(C++20) is not provided");
98 typedef typename std::remove_cv<T>::type value_type;
99 detail::check_std_enable_shared_from_this<value_type> ();
100 typedef detail::ctl_inplace<T, A> block_type;
101 block_type *const block
102 = block_type::create (alloc, std::forward<Args> (args)...);
103 value_type *const object = block->object ();
104 block->set_anchor (reinterpret_cast<std::uintptr_t> (object));
105 shared_ptr<T> result = detail::access::adopt<T> (block, object);
106 detail::esft_dispatch<value_type>::apply (result, object);
107 return result;
108}

115template <class T, class... Args>
116shared_ptr<T>
117make_shared (Args &&...args)
118{
119 return smart_ptr::allocate_shared<T> (
120 detail::block_allocator<typename std::remove_cv<T>::type> (),
121 std::forward<Args> (args)...);
122}
