582template <typename IteratorType, typename Sentinel, typename ValueType,
583 typename Projection = functional::identity,
584 typename Predicate = functional::less,
585 typename std::enable_if<
586 Detail::is_nearest_call<IteratorType, Sentinel, ValueType,
587 Predicate, Projection>::value,
588 int>::type
589 = 0>
590LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
591LUMEX_CONSTEXPR_CXX14 IteratorType
592 get_nearest_to (IteratorType first, Sentinel last, ValueType const &value,
593 Predicate pred = Predicate (),
594 Projection proj = Projection ())
595{
596 IteratorType const end_position = Detail::iterator_at (first, last);
597 if (first == end_position)
598 return end_position;
599
600 IteratorType iter
601 = Detail::lower_bound_of (first, end_position, value, pred, proj);
602 if (iter == end_position)
603 {
604 --iter;
605 return iter;
606 }
607 if (iter == first)
608 return iter;
609
610 IteratorType previous = iter;
611 --previous;
612
613 auto const distBetweenFoundAndSpecified
614 = Detail::distance (Detail::invoke_projection (proj, *iter), value);
615 auto const distBetweenPrevAndSpecified
616 = Detail::distance (Detail::invoke_projection (proj, *previous), value);
617
618 return (distBetweenFoundAndSpecified < distBetweenPrevAndSpecified)
619 ? iter
620 : previous;
621}
