151inline void
152write_to_stderr (std::string const &prefix, char const *whatMsg) LUMEX_NOEXCEPT
153{
154 std::cerr.write (prefix.data (),
155 static_cast<std::streamsize> (prefix.size ()));
156
157 static char const kReason[] = ". Reason: ";
158 std::cerr.write (kReason,
159 static_cast<std::streamsize> (sizeof (kReason) - 1));
160
161 if (whatMsg != nullptr)
162 {
163 std::cerr.write (whatMsg,
164 static_cast<std::streamsize> (std::strlen (whatMsg)));
165 }
166 else
167 {
168 static char const kUnknownWhat[] = "<unknown exception message>";
169 std::cerr.write (kUnknownWhat, static_cast<std::streamsize> (
170 sizeof (kUnknownWhat) - 1));
171 }
172
173 std::cerr.put ('\n');
174 std::cerr.flush ();
175}

217inline void
218report_exception (std::string const &prefix,
219 char const *whatMsg) LUMEX_NOEXCEPT
220{
221 safe_call_report_fn const reporter = get_safe_call_reporter ();
222 if (reporter != nullptr)
223 {
224 try
225 {
226 std::string line = prefix;
227 line += ". Reason: ";
228 line += (whatMsg != nullptr) ? whatMsg
229 : "<unknown exception message>";
230 reporter (line.c_str ());
231 return;
232 }
233 catch (...)
234 {
235 }
236 }
237 write_to_stderr (prefix, whatMsg);

282template <typename Function, typename... Args>
283auto
284ExceptionWrapper (
285 std::string const
286 &excMessage, // NOLINT(bugprone-easily-swappable-parameters)
287 std::string const &unknownExcMessage, Function &&func, Args &&...args)
288 // std::declval, not the parameters: GCC 8 evaluates a noexcept operand
289 // that names the function parameters as noexcept even when it throws.
290 LUMEX_NOEXCEPT_IF (
291 noexcept (std::declval<Function> () (std::declval<Args> ()...)))
292 -> lumex::core::utility::traits::invoke::invoke_result_t<Function,
293 Args...>
294{
295 using ReturnType
296 = lumex::core::utility::traits::invoke::invoke_result_t<Function,
297 Args...>;
298
299 LUMEX_STATIC_ASSERT_MSG (
300 lumex::core::utility::traits::invoke::is_callable_v<Function, Args...>,
301 "Function must be callable with the supplied arguments. Check:\n"
302 "1) Function type is correct\n"
303 "2) Number of arguments matches\n"
304 "3) Argument types are compatible");
305
306 LUMEX_STATIC_ASSERT_MSG (
307 std::is_default_constructible<ReturnType>::value
308 || std::is_void<ReturnType>::value,
309 "ReturnType must be default-constructible or void. "
310 "For non-default-constructible types, consider using std::optional.");
311
312 try
313 {
314 return std::forward<Function> (func) (std::forward<Args> (args)...);
315 }
316 catch (std::exception const &exc)
317 {
318 detail::report_exception (excMessage, exc.what ());
319 }
320 catch (...)
321 {
322 detail::report_line (unknownExcMessage);
323 }
324
325 // static_assert above guarantees ReturnType is default-constructible or
326 // void, so this cannot throw.
327 return lumex::core::utility::traits::meta::default_return<
328 ReturnType>::value ();
329}
