223template <typename CharT, typename Traits>
224void
225print_serial_ports_info (std::basic_ostream<CharT, Traits> &os,
226 std::vector<serial_port_info_t> const &infos)
227{
228 std::vector<serial_port_info_t const *> by_free;
229 std::vector<serial_port_info_t const *> by_busy;
230 std::vector<serial_port_info_t const *> by_available;
231 std::vector<serial_port_info_t const *> by_unresponsive;
232
233 for (std::vector<serial_port_info_t>::const_iterator it = infos.begin ();
234 it != infos.end (); ++it)
235 {
236 if (it->state == serial_port_state::free)
237 by_free.push_back (&(*it));
238 else if (it->state == serial_port_state::busy)
239 by_busy.push_back (&(*it));
240 else if (it->state == serial_port_state::unresponsive)
241 by_unresponsive.push_back (&(*it));
242 else
243 by_available.push_back (&(*it));
244 }
245
246 os << "\n=== Ports: available (device connected, port free) ===\n";
247 for (std::size_t i = 0; i < by_available.size (); ++i)
248 {
249 serial_port_info_t const &info = *by_available[i];
250 os << " " << info.path.c_str ();
251 if (!info.friendly_name.empty ())
252 os << " [" << info.friendly_name.c_str () << "]";
253 if (!info.vid.empty () || !info.pid.empty ())
254 os << " VID:" << info.vid.c_str () << " PID:" << info.pid.c_str ();
255 if (!info.manufacturer.empty ())
256 os << " Mfg:" << info.manufacturer.c_str ();
257 if (!info.product.empty ())
258 os << " Product:" << info.product.c_str ();
259 if (!info.serial_number.empty ())
260 os << " SN:" << info.serial_number.c_str ();
261 if (!info.driver.empty ())
262 os << " Driver:" << info.driver.c_str ();
263 if (!info.bus_info.empty ())
264 os << " Bus:" << info.bus_info.c_str ();
265 if (info.vid.empty () && info.manufacturer.empty ()
266 && !info.device_details.empty ())
267 os << " " << info.device_details.c_str ();
268 if (!info.system_error.empty ())
269 os << " Error:" << info.system_error.c_str ();
270 os << "\n";
271 }
272
273 os << "\n=== Ports: busy (used by another process) ===\n";
274 for (std::size_t i = 0; i < by_busy.size (); ++i)
275 {
276 serial_port_info_t const &info = *by_busy[i];
277 os << " " << info.path.c_str ();
278 if (!info.friendly_name.empty ())
279 os << " [" << info.friendly_name.c_str () << "]";
280 if (!info.holder_process_info.empty ())
281 os << " | " << info.holder_process_info.c_str ();
282 os << "\n";
283 }
284
285 os << "\n=== Ports: unresponsive (open timed out) ===\n";
286 for (std::size_t i = 0; i < by_unresponsive.size (); ++i)
287 {
288 serial_port_info_t const &info = *by_unresponsive[i];
289 os << " " << info.path.c_str ();
290 if (!info.friendly_name.empty ())
291 os << " [" << info.friendly_name.c_str () << "]";
292 if (!info.system_error.empty ())
293 os << " " << info.system_error.c_str ();
294 os << "\n";
295 }
296
297 os << "\n=== Ports: free (no physical connection) ===\n";
298 for (std::size_t i = 0; i < by_free.size (); ++i)
299 {
300 serial_port_info_t const &info = *by_free[i];
301 os << " " << info.path.c_str ();
302 os << "\n";
303 }
304
305 os << "(total: " << infos.size () << ")\n" << std::flush;
306}
