128template <traits::meta::Extractible T>
129LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
130std::optional<T> As (void const *data, std::size_t size) LUMEX_NOEXCEPT
131{
132 if ((data == nullptr) || (size < sizeof (T)))
133 return std::nullopt;
134
135 T res{};
136 std::memcpy (std::addressof (res), data, sizeof (T));
137 return res;
138}

146template <traits::meta::Extractible T, Detail::DataSource TSource>
147LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
148std::optional<T> As (TSource const &source) LUMEX_NOEXCEPT
149{
150 int const rawSize{ source.GetDataSize () };
151 if (rawSize < 0)
152 return std::nullopt;
153 return As<T> (source.GetData (), static_cast<std::size_t> (rawSize));
154}

159template <traits::meta::Extractible T, traits::meta::ByteLike ByteType>
160LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
161std::optional<T> As (std::span<ByteType const> span) LUMEX_NOEXCEPT
162{
163 return As<T> (span.data (), span.size ());
164}
