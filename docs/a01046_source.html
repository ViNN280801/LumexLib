599 template <std::size_t Offset, std::size_t Count = dynamic_extent>
600 LUMEX_ATTRIBUTE_NODISCARD ("the subview is the result")
601 LUMEX_CONSTEXPR
602 span<element_type,
603 detail::subspan_extent<Extent, Offset, Count>::value> subspan ()
604 const LUMEX_NOEXCEPT
605 {
606 static_assert (Offset <= Extent,
607 "span::subspan<Offset, Count>: Offset out of range");
608 static_assert (
609 Count == dynamic_extent
610 || (Offset <= Extent && Count <= Extent - Offset),
611 "span::subspan<Offset, Count>: Offset + Count out of range");
612 return LUMEX_SPAN_CHECKED (
613 Offset <= m_storage.size ()
614 && (Count == dynamic_extent
615 || Count <= m_storage.size () - Offset),
616 span<element_type,
617 detail::subspan_extent<Extent, Offset, Count>::value> (
618 m_storage.data () + Offset,
619 Count == dynamic_extent ? m_storage.size () - Offset : Count));
620 }
