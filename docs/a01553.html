in_place_t lumex::core::optional::opt::in_place { in_place_t::init_tag{} }
constexpr

A constant of type in_place_t used for in-place construction.

This constant is used as the first argument in optional constructors when direct construction of the contained value is desired. It has no global alias: write lumex::core::optional::opt::in_place.

using lumex::core::optional::opt::in_place;
// Constructs MyClass (arg1, arg2) in the optional's storage.
optional<MyClass> opt1 (in_place, arg1, arg2);
// Constructs std::vector<int> ({1, 2, 3}).
optional<std::vector<int>> opt2 (in_place, { 1, 2, 3 });

nullopt_t lumex::core::optional::opt::nullopt { nullopt_t::init_tag{} }
constexpr

A constant of type nullopt_t used to represent an empty optional.

This global constant is the canonical way to initialize or assign an empty state to a optional object.

optional<int> opt1 = lumex::nullopt;
optional<std::string> opt2; // Also initializes as empty
opt2 = lumex::nullopt;
