281inline void
282append_crc_least_significant_byte_first (crc_params_t const &params,
283 std::vector<std::uint8_t> &buffer)
284 LUMEX_NOEXCEPT
285{
286 std::uint64_t const value = compute_crc_with_rev_eng_params (
287 params, buffer.data (), buffer.size ());
288 int const numBytes = (params.widthBits + 7) / 8;
289 for (int i = 0; i < numBytes; ++i)
290 buffer.push_back (static_cast<std::uint8_t> ((value >> (8 * i)) & 0xFFU));
291}
