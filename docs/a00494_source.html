105// Declared, never defined. The first overload deduces U from the (single)
106// base enable_shared_from_this<U> of Y and takes part only when that base is
107// accessible and unambiguous: the third parameter fails the substitution
108// otherwise, which removes the overload instead of making the program
109// ill-formed (an access violation in a call is not a SFINAE error, but a
110// failed substitution is). The ellipsis overload answers "none".
111template <class U, class Y>
112esft_found_t<U>
113esft_probe (enable_shared_from_this<U> const volatile *, Y *,
114 typename std::enable_if<
115 std::is_convertible<
116 Y *, enable_shared_from_this<U> const volatile *>::value,
117 int>::type
118 = 0);
119esft_none_t esft_probe (...);

132template <class Y> struct esft_probe_result
