101template <class Block, class Alloc, class Make>
102Block *
103create_block (Alloc const &alloc, Make make)
104{
105 typedef typename std::allocator_traits<Alloc>::template rebind_alloc<Block>
106 block_alloc;
107 typedef std::allocator_traits<block_alloc> block_traits;
108 static_assert (std::is_pointer<typename block_traits::pointer>::value,
109 "smart_ptr: allocators with fancy pointers are not "
110 "supported");
111 block_alloc block_allocator_copy (alloc);
112 Block *memory = block_traits::allocate (block_allocator_copy, 1);
113 LUMEX_SMART_PTR_TRY { make (memory); }
114 LUMEX_SMART_PTR_CATCH_ALL
115 {
116 block_traits::deallocate (block_allocator_copy, memory, 1);
117 LUMEX_SMART_PTR_RETHROW;
118 }
119 return memory;
120}

128template <class Block, class Alloc>
129void
130destroy_block (Block *block, Alloc const &alloc) LUMEX_NOEXCEPT
131{
132 typedef typename std::allocator_traits<Alloc>::template rebind_alloc<Block>
133 block_alloc;
134 block_alloc block_allocator_copy (alloc);
135 block->~Block ();
136 std::allocator_traits<block_alloc>::deallocate (block_allocator_copy, block,
137 1);
138}

146template <class P, class D, class A>
147class ctl_ptr final : public ctl_base,
148 private ebo_member<D, 0>,
149 private ebo_member<A, 1>
150{
151public:
158 static ctl_ptr *
159 create (P pointer, D &&deleter, A const &alloc)
160 {
161 return create_block<ctl_ptr> (
162 alloc, [&] (void *memory)
163 { ::new (memory) ctl_ptr (pointer, std::move (deleter), alloc); });
164 }
165
166 void *
167 query (void const *key) LUMEX_NOEXCEPT override
168 {
169 return !is_implicit_deleter<D>::value && key == type_id<D> ()
170 ? static_cast<void *> (std::addressof (deleter_ref ()))
171 : nullptr;
172 }
173
174private:
175 ctl_ptr (P pointer, D &&deleter, A const &alloc)
176 : ctl_base (), ebo_member<D, 0> (std::move (deleter)),
177 ebo_member<A, 1> (alloc), pointer_ (pointer)
178 {
179 }
180
181 D &
182 deleter_ref () LUMEX_NOEXCEPT
183 {
184 return ebo_member<D, 0>::get ();
185 }
186
187 A &
188 allocator_ref () LUMEX_NOEXCEPT
189 {
190 return ebo_member<A, 1>::get ();
191 }
192
193 void
194 dispose () LUMEX_NOEXCEPT override
195 {
196 deleter_ref () (pointer_);
197 }
198
199 void
200 destroy () LUMEX_NOEXCEPT override
201 {
202 A alloc_copy (allocator_ref ());
203 destroy_block (this, alloc_copy);
204 }
205
206 P pointer_;
207};

215template <class T, class A>
216class ctl_inplace final : public ctl_base, private ebo_member<A, 1>
217{
218public:
219 typedef typename std::remove_cv<T>::type value_type;
220
227 template <class... Args>
228 static ctl_inplace *
229 create (A const &alloc, Args &&...args)
230 {
231 ctl_inplace *created = nullptr;
232 create_block<ctl_inplace> (alloc,
233 [&] (void *memory)
234 {
235 created = ::new (memory) ctl_inplace (
236 alloc, std::forward<Args> (args)...);
237 });
238 return created;
239 }
240
242 value_type *
243 object () LUMEX_NOEXCEPT
244 {
245 return std::addressof (storage_.value);
246 }
247
248private:
249 typedef typename std::allocator_traits<A>::template rebind_alloc<value_type>
250 value_alloc;
251 typedef std::allocator_traits<value_alloc> value_traits;
252
253 // A union keeps the storage aligned for value_type and the object
254 // unconstructed until the block constructor builds it.
255 union storage_t
256 {
257 storage_t () {}
258 ~storage_t () {}
259 value_type value;
260 };
261
262 template <class... Args>
263 explicit ctl_inplace (A const &alloc, Args &&...args)
264 : ctl_base (), ebo_member<A, 1> (alloc)
265 {
266 value_alloc object_alloc (alloc);
267 value_traits::construct (object_alloc, object (),
268 std::forward<Args> (args)...);
269 }
270
271 A &
272 allocator_ref () LUMEX_NOEXCEPT
273 {
274 return ebo_member<A, 1>::get ();
275 }
276
277 void
278 dispose () LUMEX_NOEXCEPT override
279 {
280 value_alloc object_alloc (allocator_ref ());
281 value_traits::destroy (object_alloc, object ());
282 }
283
284 void
285 destroy () LUMEX_NOEXCEPT override
286 {
287 A alloc_copy (allocator_ref ());
288 destroy_block (this, alloc_copy);
289 }
290
291 storage_t storage_;
292};

298class ctl_holder final : public ctl_base
299{
300public:
307 static ctl_holder *
308 create (ctl_base *owner, void *pointer)
309 {
310 void *memory
311 = allocate_aligned (sizeof (ctl_holder), alignof (ctl_holder));
312 LUMEX_SMART_PTR_TRY
313 {
314 ctl_holder *holder = ::new (memory) ctl_holder (owner, pointer);
315 owner->add_strong ();
316 return holder;
317 }
318 LUMEX_SMART_PTR_CATCH_ALL
319 {
320 deallocate_aligned (memory, alignof (ctl_holder));
321 LUMEX_SMART_PTR_RETHROW;
322 }
323 return nullptr;
324 }
325

328 owner () const LUMEX_NOEXCEPT
329 {
330 return owner_;
331 }

335 pointer () const LUMEX_NOEXCEPT
336 {
337 return pointer_;
338 }
339
347 void
348 disown () LUMEX_NOEXCEPT
349 {
350 owns_ = false;
351 }
352
353private:
354 ctl_holder (ctl_base *owner, void *pointer) LUMEX_NOEXCEPT
355 : ctl_base (),
356 owner_ (owner),
357 pointer_ (pointer),
358 owns_ (true)
359 {
360 }
361
362 void
363 dispose () LUMEX_NOEXCEPT override
364 {
365 if (owns_)
366 owner_->release_strong ();
367 }
368

370 destroy () LUMEX_NOEXCEPT override

373 deallocate_aligned (this, alignof (ctl_holder));
374 }
