194template <typename Callable, typename... Args>
195long long
196measure_execution_time (Callable &&callable, Args &&...args)
197{
198 LUMEX_STATIC_ASSERT_MSG (
199 lumex::core::utility::traits::invoke::is_callable_v<Callable, Args...>,
200 "Callable must be invocable with Args");
201
202 LumexTimer timer;
203 timer.start_timer ();
204 std::forward<Callable> (callable) (std::forward<Args> (args)...);
205 timer.stop_timer ();
206 return timer.elapsed_time_ms ();
207}

225template <typename Callable>
226bool
227measure_time (Callable &&callable, std::string const &message,
228 std::ostream &out = std::clog, bool need_to_gate_via_env = true,
229 LUMEX_ATTRIBUTE_MAYBE_UNUSED char const *env_name
230 = default_measure_time_env_name ())
231{
232 LUMEX_STATIC_ASSERT_MSG (
233 lumex::core::utility::traits::invoke::is_callable_v<Callable>,
234 "Callable must be nullary-invocable");
235
236 if (need_to_gate_via_env)
237 {
238 if (env_name == nullptr
239 || !lumex::core::environment::env::is_env_set (env_name))
240 {
241 std::forward<Callable> (callable) ();
242 return false;
243 }
244 }
245 else
246 {
247 LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR (env_name);
248 }
249
250 long long const elapsed_ms
251 = measure_execution_time (std::forward<Callable> (callable));
252 write_measure_time_report (out, message, elapsed_ms);
253 return true;
254}
