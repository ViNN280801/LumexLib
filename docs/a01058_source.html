199template <typename Callable, typename... Args>
200long long
201measure_execution_time (Callable &&callable, Args &&...args)
202{
203 LUMEX_STATIC_ASSERT_MSG (
204 lumex::core::utility::traits::invoke::is_callable<Callable,
205 Args...>::value,
206 "Callable must be invocable with Args");
207
208 lumex_timer timer;
209 timer.start_timer ();
210 std::forward<Callable> (callable) (std::forward<Args> (args)...);
211 timer.stop_timer ();
212 return timer.elapsed_time_ms ();
213}

231template <typename Callable>
232bool
233measure_time (Callable &&callable, std::string const &message,
234 std::ostream &out = std::clog, bool need_to_gate_via_env = true,
235 LUMEX_ATTRIBUTE_MAYBE_UNUSED char const *env_name
236 = default_measure_time_env_name ())
237{
238 LUMEX_STATIC_ASSERT_MSG (
239 lumex::core::utility::traits::invoke::is_callable<Callable>::value,
240 "Callable must be nullary-invocable");
241
242 if (need_to_gate_via_env)
243 {
244 if (env_name == nullptr
245 || !lumex::core::environment::env::is_env_set (env_name))
246 {
247 std::forward<Callable> (callable) ();
248 return false;
249 }
250 }
251 else
252 {
253 LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR (env_name);
254 }
255
256 long long const elapsed_ms
257 = measure_execution_time (std::forward<Callable> (callable));
258 write_measure_time_report (out, message, elapsed_ms);
259 return true;
260}

287// The picker below takes a trailing `...`, filled with one dummy argument (0):
288// a macro whose `...` gets no argument is an extension before C++20
289// (-Wpedantic on GCC, -Wvariadic-macro-arguments-omitted on Clang), and the
290// one-argument form would leave it empty.
291#define LUMEX_MEASURE_TIME_DISPATCH(...) \
292 LUMEX_MEASURE_TIME_GET_OVERLOAD (__VA_ARGS__, LUMEX_MEASURE_TIME_2, \
293 LUMEX_MEASURE_TIME_1, 0) \
294 (__VA_ARGS__)
