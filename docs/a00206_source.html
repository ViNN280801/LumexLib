151inline bool
152_is_base64_char_impl (Types::byte_type chr) LUMEX_NOEXCEPT
153{
154 // A character is a valid Base64 alphabet character if its decoded value is
155 // not kBase64DecodeInvalidChar. This function specifically excludes '=' as a
156 // valid alphabet character, as '=' is padding.
157 return _decode_table.at (chr) != Constants::kBase64DecodeInvalidChar;
158}

168template <typename T>
169std::string
170_encode_impl (T const &data_view)
171{
172 LUMEX_STATIC_ASSERT_MSG (
173 lumex::core::utility::traits::range::has_convertible_size<T>::value,
174 "Encoding data requires a type with a .size() method whose "
175 "return type is convertible to std::size_t.");
176 LUMEX_STATIC_ASSERT_MSG (
177 lumex::core::utility::traits::range::has_convertible_indexed_access<
178 T, Types::byte_type>::value,
179 "Encoding data requires a type with an operator[] that "
180 "accepts std::size_t as an index and returns "
181 "a type convertible to byte_type.");
182
183 if (data_view.empty ())
184 return {};
185
186 // Calculate output length and reserve space
187 std::string result;
188 std::size_t out_length{ 4 * ((data_view.size () + 2) / 3) };
189 result.reserve (out_length);
190
191 // 1. Process 3-byte chunks
192 std::size_t pos{};
193 while (pos + 2 < data_view.size ())
194 {
195 // 1.1. Get the first 6 bits of the first byte
196 result += detail::_base64_chars.at ((data_view[pos] >> 2)
197 & Constants::kBase64MaskSixBits);
198
199 // 1.2. Get the last 2 bits of the first byte and the first 4 bits of the
200 // second byte
201 result += detail::_base64_chars.at (static_cast<std::size_t> (
202 ((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
203 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits)));
204
205 // 1.3. Get the last 4 bits of the second byte and the first 2 bits of
206 // the third byte
207 result += detail::_base64_chars.at (static_cast<std::size_t> (
208 ((data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2)
209 | ((data_view[pos + 2] >> Constants::kBase64RightShiftSixBits)
210 & Constants::kBase64MaskTwoBits)));
211
212 // 1.4. Get the last 6 bits of the third byte
213 result += detail::_base64_chars.at (data_view[pos + 2]
214 & Constants::kBase64MaskSixBits);
215
216 // 1.5. Move to the next 3 bytes to process
217 pos += 3;
218 }
219
220 // 2. Handle remaining bytes
221 if (pos < data_view.size ())
222 {
223 // 2.1. Get the first 6 bits of the first byte
224 result += detail::_base64_chars.at ((data_view[pos] >> 2)
225 & Constants::kBase64MaskSixBits);
226
227 if (pos + 1 < data_view.size ())
228 {
229 // 2.1.1. Get the last 2 bits of the first byte and the first 4 bits
230 // of the second byte
231 result += detail::_base64_chars.at (static_cast<std::size_t> (
232 ((data_view[pos] & Constants::kBase64MaskTwoBits) << 4)
233 | ((data_view[pos + 1] >> 4) & Constants::kBase64MaskFourBits)));
234
235 // 2.1.2. Get the last 4 bits of the second byte
236 result += detail::_base64_chars.at (static_cast<std::size_t> (
237 (data_view[pos + 1] & Constants::kBase64MaskFourBits) << 2));
238
239 // 2.1.3. Add padding
240 result += '=';
241 }
242 else
243 {
244 // 2.2.1. Get the last 2 bits of the first byte and the first 4 bits
245 // of the second byte
246 result += detail::_base64_chars.at (static_cast<std::size_t> (
247 (data_view[pos] & Constants::kBase64MaskTwoBits) << 4));
248
249 // 2.2.2. Add double padding, because there is only one byte left
250 result += "==";
251 }
252 }
253
254 // 3. Return the result
255 return result;
256}
