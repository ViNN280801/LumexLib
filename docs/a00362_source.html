71// --- dynamic_cast operand forms (pointer / lvalue reference to class) --- //
72template <typename T>
73concept DynCastForm
74 = traits::meta::PointerToClass<T> || traits::meta::LvalueRefToClass<T>;
75
76template <typename Base, typename Derived>
77concept ValidDownCast
78 = DynCastForm<Base> && DynCastForm<Derived>
79 && (std::is_pointer_v<Base>
80 == std::is_pointer_v<Derived>) // matching ptr "shape"
81 &&traits::meta::CompleteType<traits::meta::indirection_of_t<Base>>
82 && traits::meta::CompleteType<traits::meta::indirection_of_t<Derived>> &&
83 // p5: v must be pointer-to / glvalue-of a polymorphic type
84 std::is_polymorphic_v<
85 std::remove_cv_t<traits::meta::indirection_of_t<Base>>>
86 &&
87 // narrowed to an actual downcast - not an arbitrary cross-cast
88 std::derived_from<
89 std::remove_cv_t<traits::meta::indirection_of_t<Derived>>,
90 std::remove_cv_t<traits::meta::indirection_of_t<Base>>>
91 // p2
92 && traits::meta::PreserveCV<traits::meta::indirection_of_t<Base>,
93 traits::meta::indirection_of_t<Derived>>;
94} // namespace Detail

100class BadDownCast : public std::bad_cast

102public:

104 : m_message (std::string ("downcast failed: dynamic type '")
105 + from.name () + "' is not a '" + to.name () + "'")

146template <typename Derived, typename Base>
147 requires std::is_pointer_v<Derived> && Detail::ValidDownCast<Base, Derived>
148LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
149Derived downcast (Base base)
150{
151 if (base == nullptr)
152 return nullptr; // p6: null in, null out - not a cast failure.
153
154 Derived result = dynamic_cast<Derived> (base);
155 if (result == nullptr)
156 throw BadDownCast (typeid (*base),
157 typeid (std::remove_pointer_t<Derived>));
158 return result;
159}

172template <typename Derived, typename Base>
173 requires std::is_lvalue_reference_v<Derived>
174 && Detail::ValidDownCast<Base &, Derived>
175LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
176Derived downcast (Base &base)
177{
178 // According to p9, std::bad_cast may be thrown.
179 try
180 {
181 return dynamic_cast<Derived> (base);
182 }
183 catch (std::bad_cast const &bc_exc)
184 {
185 throw BadDownCast (typeid (base),
186 typeid (std::remove_reference_t<Derived>),
187 bc_exc.what ());
188 }
189}
