844template <typename Allocator>
845std::string
846to_string (lumex_basic_stacktrace<Allocator> const &stacktrace)
847{
848 std::string result;
849 result.reserve (stacktrace.size ()
850 * detail::kDefaultMaxFrames); // Pre-allocate for efficiency
851
852 for (std::size_t i = 0; i < stacktrace.size (); ++i)
853 {
854 result += std::to_string (i);
855 result += "# ";
856 result += stacktrace[i].description ();
857 result += '\n';
858 }
859
860 return result;
861}

895template <typename Allocator> struct hash<lumex_basic_stacktrace<Allocator>>
896{
903 std::size_t
904 operator() (lumex_basic_stacktrace<Allocator> const &stacktrace) const
905 LUMEX_NOEXCEPT
906 {
907 std::size_t seed = 0;
908 for (auto const &entry : stacktrace)
909 {
910 seed ^= std::hash<void *> () (entry.native_handle ())
911 + detail::kHashGoldenRatio + (seed << detail::kHashLeftShift)
912 + (seed >> detail::kHashRightShift);
913 }
914 return seed;
915 }
