137stringify (Args &&...args)
138{
139 std::ostringstream oss;
140#if LUMEX_HAS_FOLD_EXPRESSIONS
141 ((oss << std::forward<Args> (args)), ...);
142#else
143 int expanded[] = { 0, ((oss << std::forward<Args> (args)), 0)... };
144 LUMEX_ATTRIBUTE_MAYBE_UNUSED_VAR (expanded);
145#endif
146 return oss.str ();
147}
