206// C++20+ concepts version
207template <lumex::core::utility::traits::stream::AllStreamable... Args>
208inline std::string
209logger_stringify (Args &&...args)
210{
211 LUMEX_CONSTEXPR_IF (sizeof...(args) == 0) { return ""; }
212 else
213 {
214 std::ostringstream oss;
215 ((oss << std::forward<Args> (args)), ...);
216 return oss.str ();
217 }
218}

220#elif __cplusplus >= 201703L
221
222// C++17 version with fold expressions and if constexpr
223template <typename... Args>
224inline std::string
225logger_stringify (Args &&...args)
226{
227 LUMEX_STATIC_ASSERT_MSG (
228 lumex::core::utility::traits::stream::all_ostreamable_v<Args...>,
229 "All arguments must be streamable");
230
231 LUMEX_CONSTEXPR_IF (sizeof...(args) == 0) { return ""; }
232 else
233 {
234 std::ostringstream oss;
235 ((oss << std::forward<Args> (args)), ...);
236 return oss.str ();
237 }
238}
239
240#elif __cplusplus >= 201402L
241
242// C++14 version with variable templates
243template <typename... Args>
244inline std::string
245logger_stringify (Args &&...args)
246{
247 LUMEX_STATIC_ASSERT_MSG (
248 lumex::core::utility::traits::stream::all_ostreamable_v<Args...>,
249 "All arguments must be streamable");
250
251 if ((sizeof...(args) == 0) ? true : false)
252 return "";
253
254 std::ostringstream oss;
255 (void)std::initializer_list<int>{ (oss << std::forward<Args> (args), 0)... };
256 return oss.str ();
257}

268 "All arguments must be streamable");

271 return "";
272
273 std::ostringstream oss;
274 (void)std::initializer_list<int>{ (oss << std::forward<Args> (args), 0)... };
275 return oss.str ();

317inline std::string
318logger_format_function_name_clean (char const *prettyFunc) LUMEX_NOEXCEPT
319{
320 if (prettyFunc == nullptr)
321 return "";
322
323 std::string func (prettyFunc);
324 if (func.empty ())
325 return "";
326
327 // Drop arguments (everything after the first '(')
328 std::size_t openParen = func.find ('(');
329 if (openParen != std::string::npos)
330 func = func.substr (0, openParen);
331
332 // Last space ends the return type (and, on MSVC, the calling convention):
333 // MSVC "bool __cdecl Class::method" / "int __cdecl compute", GCC "void
334 // Class::method" / "void compute". Searching the whole string (not just up
335 // to the last "::") also fixes free functions (no "::" at all), which
336 // previously short-circuited and returned the return type too.
337 std::size_t lastSpace = func.rfind (' ');
338 if (lastSpace != std::string::npos)
339 func = func.substr (lastSpace + 1);
340
341 return func + "()";
342}
