175inline void *
176allocate_aligned (std::size_t bytes, std::size_t alignment)
177{
178 if (alignment <= LUMEX_SMART_PTR_DEFAULT_NEW_ALIGNMENT)
179 return ::operator new (bytes);
180#if defined(__cpp_aligned_new)
181 return ::operator new (bytes, std::align_val_t (alignment));
182#else
183 // Over-allocate, align inside the block and keep the pointer that
184 // operator new returned in the word in front of the aligned address.
185 std::size_t const extra = alignment + sizeof (void *);
186 if (bytes > (std::numeric_limits<std::size_t>::max) () - extra)
187 LUMEX_SMART_PTR_THROW (std::bad_alloc ());
188 void *const raw = ::operator new (bytes + extra);
189 std::uintptr_t address
190 = reinterpret_cast<std::uintptr_t> (raw) + sizeof (void *);
191 address = (address + alignment - 1)
192 & ~(static_cast<std::uintptr_t> (alignment) - 1);
193 reinterpret_cast<void **> (address)[-1] = raw;
194 return reinterpret_cast<void *> (address);
195#endif
196}

199inline void
200deallocate_aligned (void *block, std::size_t alignment) LUMEX_NOEXCEPT
201{
202 if (alignment <= LUMEX_SMART_PTR_DEFAULT_NEW_ALIGNMENT)
203 {
204 ::operator delete (block);
205 return;
206 }
207#if defined(__cpp_aligned_new)
208 ::operator delete (block, std::align_val_t (alignment));
209#else
210 ::operator delete (reinterpret_cast<void **> (block)[-1]);
211#endif
212}
