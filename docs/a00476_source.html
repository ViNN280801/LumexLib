97// How a value is written (see value_kind).
98LUMEX_CONSTEXPR int k_kind_direct = 0;
99LUMEX_CONSTEXPR int k_kind_optional = 1;
100LUMEX_CONSTEXPR int k_kind_map = 2;
101LUMEX_CONSTEXPR int k_kind_array = 3;
102LUMEX_CONSTEXPR int k_kind_aggregate = 4;

104// nlohmann converts U: a builtin type, a string, an enum, nlohmann::json, a
105// standard container of such types, or a type with a `to_json (json &, U
106// const &)` that ADL finds. nlohmann's constructor from a compatible type is
107// constrained by exactly that, so this is the test of whether it can.
108template <typename U>
109struct is_json_convertible : std::is_constructible<nlohmann::json, U const &>
110{
111};

113// A map: iterable, with a key type and a mapped type.
114template <typename U>
115struct is_map_like
116 : std::integral_constant<
117 bool,
118 lumex::core::utility::traits::range::is_iterable<U>::value
119 && lumex::core::utility::traits::range::has_key_type<U>::value
120 && lumex::core::utility::traits::range::has_mapped_type<
121 U>::value>
122{
123};

125// A class that is none of the above is taken for a nested aggregate.
126template <typename U>
127struct is_aggregate_candidate
128 : std::integral_constant<bool, std::is_class<U>::value
129 && !std::is_union<U>::value>
130{
131};

133// The rule that applies to a value of type U, the first that fits. Every
134// condition is a SFINAE-friendly trait, none looks into U as an aggregate (a
135// non-aggregate class must not reach aggregate_traits).
136template <typename U>
137struct value_kind
138 : std::integral_constant<
139 int,
140 is_json_convertible<U>::value ? k_kind_direct
141 : lumex::core::utility::traits::value::is_optional_like<U>::value
142 ? k_kind_optional
143 : is_map_like<U>::value ? k_kind_map
144 : lumex::core::utility::traits::range::is_iterable<U>::value
145 ? k_kind_array
146 : is_aggregate_candidate<U>::value ? k_kind_aggregate
147 : k_kind_direct>
148{
149};

151template <typename U> nlohmann::json to_value (U const &value);
152
153template <typename U, int Kind = value_kind<U>::value> struct value_writer;
154
155// nlohmann converts it (or it is left to nlohmann to report that it cannot).
156template <typename U> struct value_writer<U, k_kind_direct>
157{
158 static nlohmann::json
159 write (U const &value)
160 {
161 return nlohmann::json (value);
162 }
163};
164
165// An optional-like value inside a container or another optional: null when
166// empty. (As a field of an aggregate an empty one is omitted instead.)
167template <typename U> struct value_writer<U, k_kind_optional>
168{
169 static nlohmann::json
170 write (U const &value)
171 {
172 if (!value.has_value ())
173 return nlohmann::json (nullptr);
174 return to_value (*value);
175 }
176};
177
178// A map with string keys: an object, each mapped value by the same rule.
179template <typename U> struct value_writer<U, k_kind_map>
180{
181 LUMEX_STATIC_ASSERT_MSG (
182 (std::is_constructible<std::string,
183 typename U::key_type const &>::value),
184 "to_json writes a map of aggregates (or of other values nlohmann cannot "
185 "convert) only with keys that std::string can be made from");
186
187 static nlohmann::json
188 write (U const &value)
189 {
190 nlohmann::json j = nlohmann::json::object ();
191 for (auto const &entry : value)
192 j[std::string (entry.first)] = to_value (entry.second);
193 return j;
194 }
195};
196
197// Any other container, std::array included: an array, each element by the
198// same rule.
199template <typename U> struct value_writer<U, k_kind_array>
200{
201 static nlohmann::json
202 write (U const &value)
203 {
204 nlohmann::json j = nlohmann::json::array ();
205 for (auto const &element : value)
206 j.push_back (to_value (element));
207 return j;
208 }
209};
210
211// A nested aggregate: an object, with its own names.
212template <typename U> struct value_writer<U, k_kind_aggregate>
213{
214 static nlohmann::json
215 write (U const &value)
216 {
217 return ::lumex::core::reflection::field_reflection::to_json (value);
218 }
219};
220
221template <typename U>
222nlohmann::json
223to_value (U const &value)
224{
225 return value_writer<U>::write (value);
226}

269// Aggregates whose names are available (registered, or from the compiler at
270// C++20, or without fields).
271template <typename Struct>
272inline nlohmann::json
273to_json_fields (Struct const &obj, std::true_type)
274{
275 nlohmann::json j = nlohmann::json::object ();
276 std::array<char const *, tuple_size<Struct>::value> const names
277 = names_as_array<Struct> ();
278 append_all (
279 j, obj, names,
280 typename make_index_sequence<tuple_size<Struct>::value>::type ());
281 return j;
282}

284// No names: the static_assert is the only diagnostic.
285template <typename Struct>
286inline nlohmann::json
287to_json_fields (Struct const &, std::false_type)
288{
289 LUMEX_STATIC_ASSERT_MSG (
290 sizeof (Struct) == 0,
291 "to_json needs the field names of the aggregate: register them with "
292 "LUMEX_DEFINE_FIELD_NAMES (Type, field, ...) next to the definition of "
293 "the type, or build for C++20 (names then come from the compiler)");
294 return nlohmann::json::object ();
295}
