976 or_else (Fn &&fn) const &
977 {
978 // Both arms are prvalues of the result type: with the lvalue `*this` as
979 // one arm, MSVC materializes it into a temporary and copies the temporary
980 // into the result (two payload copies instead of the one a copy of the
