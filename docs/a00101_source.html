246 static bool
247 create_section (nlohmann::json &config,
248 std::string const &section) LUMEX_NOEXCEPT
249 {
250 if (section.empty ())
251 return true;
252 if (!config.is_object ())
253 {
254 _report (level_type::error, LUMEX_FUNCTION_NAME,
255 "Configuration is not a JSON object, cannot create section '",
256 section, "', returning false.");
257 return false;
258 }
259 try
260 {
261 nlohmann::json::iterator const it = config.find (section);
262 if (it == config.end ())
263 {
264 config[section] = nlohmann::json::object ();
265 _report (level_type::info, LUMEX_FUNCTION_NAME, "Section '",
266 section, "' created successfully.");
267 }
268 else if (!it->is_object ())
269 {
270 _report (level_type::warning, LUMEX_FUNCTION_NAME, "Key '",
271 section,
272 "' exists but is not a JSON object, cannot create "
273 "section, returning false.");
274 return false;
275 }
276 return true;
277 }
278 catch (std::exception const &exc)
279 {
280 _report (level_type::error, LUMEX_FUNCTION_NAME,
281 "Exception creating section '", section, "': ", exc.what (),
282 ", returning false.");
283 return false;
284 }

374 template <typename T, typename KeyType>
375 static bool
376 write_value (std::string const &filename, KeyType const &key, T const &value,
377 std::string const &section = std::string ()) LUMEX_NOEXCEPT
378 {
379 std::lock_guard<std::recursive_mutex> const lock (get_file_mutex ());
380 if (filename.empty ())
381 {
382 _report (level_type::error, LUMEX_FUNCTION_NAME,
383 "Config filename is empty, returning false.");
384 return false;
385 }
386 nlohmann::json config = _load_config_impl (filename);
387 if (config.is_null ())
388 config = nlohmann::json::object ();
389 if (!set_value (config, key, value, section))
390 return false;
391 return _save_config_impl (config, filename);
392 }

407 "Config filename is empty, returning false.");

413 _report (level_type::warning, LUMEX_FUNCTION_NAME,
414 "Config file is empty or invalid, returning false.");
415 return false;

418 {
419 _report (level_type::warning, LUMEX_FUNCTION_NAME, "Key '", key,
420 "' not found in config, nothing to edit, returning false.");
421 return false;
422 }
423 if (!set_value (config, key, value, section))
424 return false;
425 return _save_config_impl (config, filename);

438 static nlohmann::json
439 _load_config_impl (std::string const &filename) LUMEX_NOEXCEPT
440 {
441 if (filename.empty ())
442 {
443 _report (level_type::error, LUMEX_FUNCTION_NAME,
444 "Config filename is empty, returning empty JSON object.");
445 return nlohmann::json ();
446 }
447 try
448 {
449 std::ifstream json_file (filename.c_str ());
450 if (!json_file.is_open ())
451 {
452 _report (level_type::error, LUMEX_FUNCTION_NAME,
453 "Failed to open config file: ", filename, ".");
454 return nlohmann::json ();
455 }
456 nlohmann::json config;
457 json_file >> config;
458 return config;
459 }
460 catch (nlohmann::json::parse_error const &exc)
461 {
462 _report (level_type::error, LUMEX_FUNCTION_NAME,
463 "Failed to parse config file ", filename, ": ", exc.what (),
464 ".");
465 }
466 catch (std::exception const &exc)
467 {
468 _report (level_type::error, LUMEX_FUNCTION_NAME,
469 "Unexpected error reading config file ", filename, ": ",
470 exc.what (), ".");
471 }
472 return nlohmann::json ();
473 }

475 static bool
476 _save_config_impl (nlohmann::json const &config,
477 std::string const &filename) LUMEX_NOEXCEPT
478 {
479 if (filename.empty ())
480 {
481 _report (level_type::error, LUMEX_FUNCTION_NAME,
482 "Config filename is empty, returning false.");
483 return false;
484 }
485 try
486 {
487 std::ofstream json_file (filename.c_str ());
488 if (!json_file.is_open ())
489 {
490 _report (level_type::error, LUMEX_FUNCTION_NAME,
491 "Failed to open config file for writing: ", filename,
492 ", returning false.");
493 return false;

495 json_file << std::setw (4) << config << '\n';
496 json_file.close ();
497 if (json_file.fail ())
498 {
499 _report (level_type::error, LUMEX_FUNCTION_NAME,
500 "Error writing config to file ", filename,
501 ", returning false.");
502 return false;
503 }
504 return true;
505 }
506 catch (std::exception const &exc)
507 {
508 _report (level_type::error, LUMEX_FUNCTION_NAME,
509 "Error writing config to file ", filename, ": ", exc.what (),
510 ", returning false.");
511 return false;

515 static bool
516 _is_json_file_ok_impl (std::string const &filename) LUMEX_NOEXCEPT
517 {
518 if (filename.empty ())
519 {
520 _report (level_type::error, LUMEX_FUNCTION_NAME,
521 "Config filename is empty, returning false.");
522 return false;
523 }
524 try
525 {
526 std::ifstream json_file (filename.c_str ());
527 if (!json_file.is_open ())
528 {
529 _report (level_type::error, LUMEX_FUNCTION_NAME,
530 "Failed to open config file: ", filename, ".");
531 return false;
532 }
533 try
534 {
535 nlohmann::json config;
536 json_file >> config;
537 _report (level_type::debug, LUMEX_FUNCTION_NAME, "Config file ",
538 filename, " parsed successfully.");
539 return true;
540 }
541 catch (nlohmann::json::parse_error const &exc)
542 {
543 json_file.clear ();
544 json_file.seekg (0);
545 std::string const content (
546 (std::istreambuf_iterator<char> (json_file)),
547 std::istreambuf_iterator<char> ());
548 std::size_t line = 1;
549 std::size_t column = 1;
550 for (std::size_t i = 0; i < exc.byte && i < content.size (); ++i)
551 {
552 if (content[i] == '\n')
553 {
554 ++line;
555 column = 1;
556 }
557 else
558 ++column;
559 }
560 _report (level_type::error, LUMEX_FUNCTION_NAME,
561 "Failed to parse config file ", filename, " at line ",
562 line, ", column ", column, " (byte ", exc.byte,
563 "): ", exc.what (), ".");
564 return false;
565 }
566 }
567 catch (std::exception const &exc)
568 {
569 _report (level_type::error, LUMEX_FUNCTION_NAME,
570 "Error reading config file ", filename, ": ", exc.what (),
571 ".");
572 return false;
573 }
574 catch (...)
575 {
576 _report (level_type::error, LUMEX_FUNCTION_NAME,
577 "Unknown error reading config file ", filename, ".");
578 return false;
579 }

606 template <typename Visitor>
607 static bool
608 _walk_section_path (std::string const &section, Visitor visit)
609 {
610 std::size_t position = 0;
611 while (position < section.size ())
612 {
613 std::size_t const dot = section.find ('.', position);
614 std::size_t const end
615 = dot == std::string::npos ? section.size () : dot;
616 std::string const component
617 = section.substr (position, end - position);
618 position = end + 1;
619 if (component.empty ())
620 continue;
621 if (!visit (component))
622 return false;
623 }
624 return true;
625 }

631 static nlohmann::json &
632 _get_target_section (nlohmann::json &config, std::string const &section)
633 {
634 if (!config.is_object ())
635 config = nlohmann::json::object ();
636 nlohmann::json *current = &config;
637 _walk_section_path (
638 section,
639 [&current] (std::string const &component) -> bool
640 {
641 nlohmann::json::iterator const it = current->find (component);
642 if (it == current->end () || !it->is_object ())
643 {
644 (*current)[component] = nlohmann::json::object ();
645 _report (level_type::warning,
646 "LumexJsonHelper::_get_target_section",
647 "Creating or overwriting key '", component,
648 "' to an object in section traversal.");
649 }
650 current = &(*current)[component];
651 return true;
652 });
653 return *current;
654 }
655
661 static nlohmann::json const &
662 _get_target_section (nlohmann::json const &config,
663 std::string const &section)
664 {
665 static nlohmann::json const empty_object = nlohmann::json::object ();
666 if (!config.is_object ())
667 return empty_object;
668 nlohmann::json const *current = &config;
669 bool const found = _walk_section_path (
670 section,
671 [&current] (std::string const &component) -> bool
672 {
673 nlohmann::json::const_iterator const it
674 = current->find (component);
675 if (it == current->cend () || !it->is_object ())
676 return false;
677 current = &(*it);
678 return true;
679 });
680 return found ? *current : empty_object;
681 }
