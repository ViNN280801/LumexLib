172template <class T>
173shared_ptr<T>
174from_std (std::shared_ptr<T> const &p)
175{
176 typedef typename shared_ptr<T>::element_type element_type;
177 element_type *const stored = p.get ();
178 if (p.use_count () == 0)
179 return shared_ptr<T> (shared_ptr<void> (), stored);
180 if (detail::lumex_owner_deleter *const wrapper
181 = std::get_deleter<detail::lumex_owner_deleter> (p))
182 return shared_ptr<T> (wrapper->owner (), stored);
183 detail::ctl_std *const block = detail::ctl_std::create (
184 std::shared_ptr<void> (p, static_cast<void *> (nullptr)));
185 block->set_anchor (reinterpret_cast<std::uintptr_t> (stored));
186 return detail::access::adopt<T> (block, stored);
187}

190template <class T>
191shared_ptr<T>
192from_std (std::shared_ptr<T> &&p)
193{
194 typedef typename shared_ptr<T>::element_type element_type;
195 element_type *const stored = p.get ();
196 if (p.use_count () == 0)
197 return shared_ptr<T> (shared_ptr<void> (), stored);
198 if (detail::lumex_owner_deleter *const wrapper
199 = std::get_deleter<detail::lumex_owner_deleter> (p))
200 return shared_ptr<T> (wrapper->owner (), stored);
201 // (The rvalue aliasing constructor of std is C++20.) Take the owner part,
202 // then give up the argument, so no reference of it stays behind.
203 std::shared_ptr<void> owner (p, static_cast<void *> (nullptr));
204 p.reset ();
205 detail::ctl_std *const block = detail::ctl_std::create (std::move (owner));
206 block->set_anchor (reinterpret_cast<std::uintptr_t> (stored));
207 return detail::access::adopt<T> (block, stored);
208}

215template <class T>
216std::shared_ptr<T>
217to_std (shared_ptr<T> const &p)
218{
219 typedef typename shared_ptr<T>::element_type element_type;
220 element_type *const stored = p.get ();
221 detail::ctl_base *const block = detail::access::control (p);
222 if (block == nullptr)
223 return std::shared_ptr<T> (std::shared_ptr<void> (), stored);
224 if (void *const owner
225 = block->query (detail::type_id<detail::std_owner_key> ()))
226 return std::shared_ptr<T> (*static_cast<std::shared_ptr<void> *> (owner),
227 stored);
228 return std::shared_ptr<T> (
229 stored, detail::lumex_owner_deleter (
230 shared_ptr<void> (p, static_cast<void *> (nullptr))));
231}
