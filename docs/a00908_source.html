126template <typename Movable, typename N, typename O, typename... Args>
127LUMEX_EXPECTED_CONSTEXPR_CXX20 void
128reinit_nothrow (std::true_type /* nothrow constructible */, Movable, N &newval,
129 O &oldval, Args &&...args)
130{
131 detail::destroy_at (detail::address_of (oldval));
132 detail::construct_at (detail::address_of (newval),
133 detail::fwd<Args> (args)...);
134}

138template <typename N, typename O, typename... Args>
139LUMEX_EXPECTED_CONSTEXPR_CXX20 void
140reinit_nothrow (std::false_type, std::true_type /* nothrow movable */,
141 N &newval, O &oldval, Args &&...args)
142{
143 N tmp (detail::fwd<Args> (args)...);
144 detail::destroy_at (detail::address_of (oldval));
145 detail::construct_at (detail::address_of (newval), detail::mv (tmp));
146}

150template <typename N, typename O, typename... Args>
151LUMEX_EXPECTED_CONSTEXPR_CXX20 void
152reinit_nothrow (std::false_type, std::false_type, N &newval, O &oldval,
153 Args &&...args)
154{
155#if LUMEX_EXPECTED_COMPILER_HAS_EXCEPTIONS
156 O tmp (detail::mv (oldval));
157 detail::destroy_at (detail::address_of (oldval));
158 try
159 {
160 detail::construct_at (detail::address_of (newval),
161 detail::fwd<Args> (args)...);
162 }
163 catch (...)
164 {
165 detail::construct_at (detail::address_of (oldval), detail::mv (tmp));
166 throw;
167 }
168#else
169 // Nothing can throw: there is no old object to put back.
170 detail::destroy_at (detail::address_of (oldval));
171 detail::construct_at (detail::address_of (newval),
172 detail::fwd<Args> (args)...);
173#endif
174}

241 "Expected<T,E>: T must not be cv void; "

251 "Expected<T,E>: T must not be a specialization of unexpected");

254 "Expected<T,E>: E must not be a reference");

734 template <typename Rhs>

773 LUMEX_EXPECTED_CONSTEXPR_CXX20 void
774 swap_storage (expected_storage &rhs)
775 {
776 if (m_has_value)
777 {
778 if (rhs.m_has_value)
779 {
780 using std::swap;
781 swap (m_storage.m_value, rhs.m_storage.m_value);
782 }
783 else
784 swap_value_with_error (swap_mode (), rhs);
785 }
786 else
787 {
788 if (rhs.m_has_value)
789 rhs.swap_value_with_error (swap_mode (), *this);
790 else
791 {
792 using std::swap;
793 swap (m_storage.m_error, rhs.m_storage.m_error);
794 }
795 }
796 }
