576 using clean_U = traits::meta::CleanType<U>;

907 static bool
908 compare (T current, U other) LUMEX_NOEXCEPT
909 {
910 // Check ranges to prevent overflow
911 // Signedness is taken into account so the comparison is correct
912 if ((std::is_signed<clean_T>::value && std::is_unsigned<clean_U>::value)
913 ? true
914 : false)
915 {
916 // T signed, U unsigned: if current < 0 then current >= other is never
917 // false
918 if (current < 0)
919 return false;
920 // other is always >= 0; only the upper bound is checked
921#ifdef LUMEXLUMEX_SFNC_ARCH_X86
922 using UnsignedT = typename std::make_unsigned<clean_T>::type;
923 if (static_cast<UnsignedT> (other)
924 > static_cast<UnsignedT> ((std::numeric_limits<clean_T>::max) ()))
925 return false;
926#else
927 if (other > static_cast<typename std::make_unsigned<clean_T>::type> (
928 (std::numeric_limits<clean_T>::max) ()))
929 return false;
930#endif
931 }
932 else if ((std::is_unsigned<clean_T>::value
933 && std::is_signed<clean_U>::value)
934 ? true
935 : false)
936 {
937 // T unsigned, U signed: if other < 0 then current >= other is always
938 // true true
939 if (other < 0)
940 return true;
941 // Otherwise check the upper bound
942#ifdef LUMEXLUMEX_SFNC_ARCH_X86
943 using UnsignedU = typename std::make_unsigned<clean_U>::type;
944 if (static_cast<UnsignedU> (other)
945 > static_cast<UnsignedU> ((std::numeric_limits<clean_T>::max) ()))
946 return false;
947#else
948 using UnsignedU = typename std::make_unsigned<clean_U>::type;
949 if (static_cast<UnsignedU> (other)
950 > static_cast<UnsignedU> ((std::numeric_limits<clean_T>::max) ()))
951 return false;
952#endif
953 }
954 else
955 {
956 // Both signed or both unsigned - safe comparison
957#ifdef LUMEXLUMEX_SFNC_ARCH_X86
958 // On 32-bit architectures use explicit casts to silence C4018
959 if (static_cast<CommonType> (other)
960 > static_cast<CommonType> ((std::numeric_limits<clean_T>::max) ()))
961 return false;
962 if (static_cast<CommonType> (other)
963 < static_cast<CommonType> ((std::numeric_limits<clean_T>::min) ()))
964 return true;
965#else
966 if (other > (std::numeric_limits<clean_T>::max) ())
967 return false;
968 if (other < (std::numeric_limits<clean_T>::min) ())
969 return true;
970#endif
971 }
972 return static_cast<CommonType> (current)
973 >= static_cast<CommonType> (other);
974 }
