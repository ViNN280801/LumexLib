1517 // Find attribute using predicate. Returns first attribute for which
1518 // predicate returned true.
1531 template <typename Predicate>
1532 XmlAttribute
1533 find_attribute (Predicate pred) const
1534 {
1535 if (!m_root)
1536 return {};
1537
1538 for (XmlAttribute attrib = first_attribute (); attrib;
1539 attrib = attrib.next_attribute ())
1540 if (pred (attrib))
1541 return attrib;
1542
1543 return {};
1544 }

1546 // Find child node using predicate. Returns first child for which predicate
1547 // returned true.
1560 template <typename Predicate>
1561 XmlNode
1562 find_child (Predicate pred) const
1563 {
1564 if (!m_root)
1565 return {};
1566
1567 for (XmlNode node = first_child (); node; node = node.next_sibling ())
1568 if (pred (node))
1569 return node;
1570
1571 return {};
1572 }

1574 // Find node from subtree using predicate. Returns first node from subtree
1575 // (depth-first), for which predicate returned true.
1591 template <typename Predicate>
1592 XmlNode
1593 find_node (Predicate pred) const
1594 {
1595 if (!m_root)
1596 return {};
1597
1598 XmlNode cur = first_child ();
1599
1600 while (cur.m_root && cur.m_root != m_root)
1601 {
1602 if (pred (cur))
1603 return cur;
1604
1605 if (cur.first_child ())
1606 cur = cur.first_child ();
1607 else if (cur.next_sibling ())
1608 cur = cur.next_sibling ();
1609 else
1610 {
1611 while (!cur.next_sibling () && cur.m_root != m_root)
1612 cur = cur.parent ();
1613
1614 if (cur.m_root != m_root)
1615 cur = cur.next_sibling ();
1616 }
1617 }
1618
1619 return {};
1620 }
