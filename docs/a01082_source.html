129template <typename NumericType>
130typename std::enable_if<
131 std::is_integral<NumericType>::value
132 && std::is_unsigned<NumericType>::value
133 && !std::is_same<bool, NumericType>::value
134 && (sizeof (NumericType) == 1 || sizeof (NumericType) == 2
135 || sizeof (NumericType) == 4 || sizeof (NumericType) == 8),
136 std::uint8_t>::type
137count_leading_zeros (NumericType value) LUMEX_NOEXCEPT
138{
139 // __builtin_clz(0) is undefined, and _BitScanReverse(0) reports no bit.
140 if (value == 0)
141 return static_cast<std::uint8_t> (sizeof (NumericType) * 8);
142
143#if defined(_MSC_VER)
144 // clang-cl defines __clang__ as well as _MSC_VER. This branch is first
145 // so this compiler keeps the MSVC intrinsic.
146 unsigned long index = 0;
147
148 if (sizeof (NumericType) <= 4)
149 {
150 _BitScanReverse (std::addressof (index),
151 static_cast<std::uint32_t> (value));
152 // Index of the highest set bit (0 is the least significant bit).
153 return static_cast<std::uint8_t> ((sizeof (NumericType) * 8) - 1
154 - index);
155 }
156 else
157 {
158#if defined(_M_X64) || defined(_M_ARM64)
159 _BitScanReverse64 (std::addressof (index),
160 static_cast<std::uint64_t> (value));
161 return static_cast<std::uint8_t> (64 - 1 - index);
162#else
163 // 32-bit MSVC has no _BitScanReverse64. Split the value in half.
164 std::uint32_t high = static_cast<std::uint32_t> (
165 static_cast<std::uint64_t> (value) >> 32);
166 if (high != 0)
167 {
168 _BitScanReverse (std::addressof (index), high);
169 return static_cast<std::uint8_t> (32 - 1 - index);
170 }
171 else
172 {
173 _BitScanReverse (std::addressof (index),
174 static_cast<std::uint32_t> (value));
175 return static_cast<std::uint8_t> (64 - 1 - index);
176 }
177#endif
178 }
179
180#elif defined(__GNUC__) || defined(__clang__)
181 if (sizeof (NumericType) <= 4)
182 {
183 // __builtin_clz counts zeros of a 32-bit unsigned int, so a narrower
184 // value is widened and its extra high bits are subtracted. The
185 // arithmetic is done in int (the builtin returns int) and narrowed once.
186 return static_cast<std::uint8_t> (
187 __builtin_clz (static_cast<std::uint32_t> (value))
188 - (32 - static_cast<int> (sizeof (NumericType) * 8)));
189 }
190 else
191 {
192 return static_cast<std::uint8_t> (
193 __builtin_clzll (static_cast<std::uint64_t> (value)));
194 }
195
196#else
197 std::uint8_t bits = static_cast<std::uint8_t> (sizeof (NumericType) * 8);
198 std::uint8_t count = 0;
199
200 // Halve the window: 32, then 16, 8, 4, 2, 1 for a 64-bit value.
201 for (int shift = bits >> 1; shift > 0; shift >>= 1)
202 {
203 if ((value >> shift) != 0)
204 {
205 value = static_cast<NumericType> (value >> shift);
206 }
207 else
208 {
209 count = static_cast<std::uint8_t> (count + shift);
210 }
211 }
212 return count;
213#endif
214}

252template <typename T>
253 requires std::is_integral_v<T> && std::has_unique_object_representations_v<T>
254LUMEX_CONSTEXPR T
255byte_swap (T value) LUMEX_NOEXCEPT
256{
257 using U = std::make_unsigned_t<T>;
258 U u = static_cast<U> (value);
259
260 LUMEX_CONSTEXPR_IF (sizeof (T) == 1) { return value; }
261 else
262 {
263 if (std::is_constant_evaluated ())
264 {
265 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
266 std::ranges::reverse (bytes);
267 U swapped = std::bit_cast<U> (bytes);
268 return static_cast<T> (swapped);
269 }
270
271#if defined(__clang__) || defined(__GNUC__)
272 LUMEX_CONSTEXPR_IF (sizeof (T) == 2)
273 {
274 u = static_cast<U> (__builtin_bswap16 (static_cast<std::uint16_t> (u)));
275 }
276 else LUMEX_CONSTEXPR_IF (sizeof (T) == 4)
277 {
278 u = static_cast<U> (__builtin_bswap32 (static_cast<std::uint32_t> (u)));
279 }
280 else LUMEX_CONSTEXPR_IF (sizeof (T) == 8)
281 {
282 u = static_cast<U> (__builtin_bswap64 (static_cast<std::uint64_t> (u)));
283 }
284 else
285 {
286 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
287 std::ranges::reverse (bytes);
288 u = std::bit_cast<U> (bytes);
289 }
290 return static_cast<T> (u);
291#elif defined(_MSC_VER)
292 LUMEX_CONSTEXPR_IF (sizeof (T) == 2)
293 {
294 u = static_cast<U> (_byteswap_ushort (static_cast<std::uint16_t> (u)));
295 }
296 else LUMEX_CONSTEXPR_IF (sizeof (T) == 4)
297 {
298 u = static_cast<U> (_byteswap_ulong (static_cast<std::uint32_t> (u)));
299 }
300 else LUMEX_CONSTEXPR_IF (sizeof (T) == 8)
301 {
302 u = static_cast<U> (_byteswap_uint64 (static_cast<std::uint64_t> (u)));
303 }
304 else
305 {
306 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
307 std::ranges::reverse (bytes);
308 u = std::bit_cast<U> (bytes);
309 }
310 return static_cast<T> (u);
311#else
312 // Portable fallback for compilers without a recognized bswap builtin.
313 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
314 std::ranges::reverse (bytes);
315 u = std::bit_cast<U> (bytes);
316 return static_cast<T> (u);
317#endif
318 }

335template <typename Unsigned>
336LUMEX_CONSTEXPR Unsigned
337swap_bytes (Unsigned value,
338 std::integral_constant<std::size_t, 2> /* size */) LUMEX_NOEXCEPT
339{
340 return static_cast<Unsigned> (
341 static_cast<Unsigned> (static_cast<Unsigned> (value & 0xFFU) << 8)
342 | static_cast<Unsigned> (static_cast<Unsigned> (value >> 8) & 0xFFU));
343}

346template <typename Unsigned>
347LUMEX_CONSTEXPR Unsigned
348swap_bytes (Unsigned value,
349 std::integral_constant<std::size_t, 4> /* size */) LUMEX_NOEXCEPT
350{
351 return static_cast<Unsigned> (
352 static_cast<Unsigned> (static_cast<Unsigned> (value & 0xFFU) << 24)
353 | static_cast<Unsigned> (static_cast<Unsigned> (value & 0xFF00U) << 8)
354 | static_cast<Unsigned> (static_cast<Unsigned> (value >> 8) & 0xFF00U)
355 | static_cast<Unsigned> (value >> 24));
356}

361swap_bytes (Unsigned value,
362 std::integral_constant<std::size_t, 8> /* size */) LUMEX_NOEXCEPT
363{
364 return static_cast<Unsigned> (
365 (static_cast<Unsigned> (value & 0xFFULL) << 56)
366 | (static_cast<Unsigned> (value & 0xFF00ULL) << 40)
367 | (static_cast<Unsigned> (value & 0xFF0000ULL) << 24)
368 | (static_cast<Unsigned> (value & 0xFF000000ULL) << 8)
369 | (static_cast<Unsigned> (value >> 8) & 0xFF000000ULL)
370 | (static_cast<Unsigned> (value >> 24) & 0xFF0000ULL)
371 | (static_cast<Unsigned> (value >> 40) & 0xFF00ULL)
372 | static_cast<Unsigned> (value >> 56));
