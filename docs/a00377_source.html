218// 1 option. Define the exception class: LUMEX_DEFINE_EXCEPTION and
219// LUMEX_DEFINE_EXCEPTION_WITH_BODY live in LumexExceptionMacros.hpp (included
220// above) so modules that must not depend on this one can use them too.
221
222// 2 option. Throw the exception.
223// Pattern:
224// [exception_name] -> custom message
225// Example:
226// [LumexException] -> Failed to open file
227#include "lumex/core/string/LumexString"
228#define LUMEX_THROW_EXCEPTION(exception_name, msg) \
229 throw exception_name (lumex::core::string::utility::stringify ( \
230 lumDemangle (exception_name), ": ", msg));
231
232// 3. Handle the exception.
233#define LUMEX_EXCEPTION_HANDLE_BEGIN \
234 try \
235 { \
236 SET_SEH_TRANSLATOR
237
238#define LUMEX_EXCEPTION_HANDLE_END \
239 } \
240 catch (lumex_base_exception const &ex) \
241 { \
242 ex.to_stderr (); \
243 ex.to_crash_report (); \
244 } \
245 catch (std::exception const &ex) \
246 { \
247 std::cerr << "[std::exception] " << ex.what () << '\n'; \
248 lumex_base_exception (ex.what ()).to_stderr (); \
249 lumex_base_exception (ex.what ()).to_crash_report (); \
250 } \
251 catch (...) { std::cerr << "[Unknown exception]\n"; }
