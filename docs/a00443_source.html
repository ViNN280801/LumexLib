138template <std::bidirectional_iterator IteratorType,
139 std::sentinel_for<IteratorType> Sentinel, typename ValueType,
140 typename Projection = std::identity,
141 std::indirect_strict_weak_order<
142 ValueType const *, std::projected<IteratorType, Projection>>
143 Predicate
144 = std::ranges::less>
145 requires Detail::NumericConcept<ValueType>
146LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
147LUMEX_CONSTEXPR IteratorType
148 GetNearestTo (IteratorType first, Sentinel last, ValueType const &value,
149 Predicate pred = {}, Projection proj = {})
150{
151 if (first == last)
152 return last;
153
154 auto iter = std::ranges::lower_bound (first, last, value, pred, proj);
155 if (iter == last)
156 return std::ranges::prev (iter);
157 if (iter == first)
158 return iter;
159
160 auto const prev = std::ranges::prev (iter);
161
162 auto const distBetweenFoundAndSpecified
163 = Detail::distance (std::invoke (proj, *iter), value);
164 auto const distBetweenPrevAndSpecified
165 = Detail::distance (std::invoke (proj, *prev), value);
166
167 return (distBetweenFoundAndSpecified < distBetweenPrevAndSpecified) ? iter
168 : prev;
169}

174template <std::ranges::bidirectional_range RangeType, typename ValueType,
175 typename Projection = std::identity,
176 std::indirect_strict_weak_order<
177 ValueType const *,
178 std::projected<std::ranges::iterator_t<RangeType>, Projection>>
179 Predicate
180 = std::ranges::less>
181 requires Detail::NumericConcept<ValueType>
182LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
183LUMEX_CONSTEXPR std::ranges::borrowed_iterator_t<RangeType> GetNearestTo (
184 RangeType &&range, ValueType const &value, Predicate pred = {},
185 Projection proj = {})
186{
187 return GetNearestTo (std::ranges::begin (std::forward<RangeType> (range)),
188 std::ranges::end (std::forward<RangeType> (range)),
189 value, pred, proj);
190}
