1544 // Find attribute using predicate. Returns first attribute for which
1545 // predicate returned true.
1558 template <typename Predicate>
1559 XmlAttribute
1560 find_attribute (Predicate pred) const
1561 {
1562 if (!m_root)
1563 return {};
1564
1565 for (XmlAttribute attrib = first_attribute (); attrib;
1566 attrib = attrib.next_attribute ())
1567 if (pred (attrib))
1568 return attrib;
1569
1570 return {};
1571 }

1573 // Find child node using predicate. Returns first child for which predicate
1574 // returned true.
1587 template <typename Predicate>
1588 XmlNode
1589 find_child (Predicate pred) const
1590 {
1591 if (!m_root)
1592 return {};
1593
1594 for (XmlNode node = first_child (); node; node = node.next_sibling ())
1595 if (pred (node))
1596 return node;
1597
1598 return {};
1599 }

1601 // Find node from subtree using predicate. Returns first node from subtree
1602 // (depth-first), for which predicate returned true.
1618 template <typename Predicate>
1619 XmlNode
1620 find_node (Predicate pred) const
1621 {
1622 if (!m_root)
1623 return {};
1624
1625 XmlNode cur = first_child ();
1626
1627 while (cur.m_root && cur.m_root != m_root)
1628 {
1629 if (pred (cur))
1630 return cur;
1631
1632 if (cur.first_child ())
1633 cur = cur.first_child ();
1634 else if (cur.next_sibling ())
1635 cur = cur.next_sibling ();
1636 else
1637 {
1638 while (!cur.next_sibling () && cur.m_root != m_root)
1639 cur = cur.parent ();
1640
1641 if (cur.m_root != m_root)
1642 cur = cur.next_sibling ();
1643 }
1644 }
1645
1646 return {};
1647 }
