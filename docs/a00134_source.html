103inline serial_probe_result_t
104run_probe (serial_transport_t &transport,
105 std::vector<std::uint8_t> const &request,
106 std::function<bool (std::vector<std::uint8_t> const &)> const
107 &response_complete,
108 std::chrono::milliseconds deadline, std::size_t max_response_bytes)
109{
110 std::chrono::steady_clock::time_point const start
111 = std::chrono::steady_clock::now ();
112
113 auto remaining = [start, deadline] () -> std::chrono::milliseconds
114 {
115 std::chrono::milliseconds const used = elapsed_since (start);
116 return deadline > used ? deadline - used : std::chrono::milliseconds (0);
117 };
118
119 serial_probe_result_t result;
120
121 if (!transport.is_open ())
122 {
123 transport_result_t const opened = transport.open (remaining ());
124 if (opened.status != transport_status::ok)
125 {
126 result.status = opened.status == transport_status::deadline_exceeded
127 ? serial_probe_status::open_timeout
128 : serial_probe_status::open_failed;
129 result.system_error = opened.system_error;
130 result.elapsed = elapsed_since (start);
131 return result;
132 }
133 }
134
135 transport.purge ();
136
137 if (!request.empty ())
138 {
139 transport_result_t const written
140 = transport.write (request.data (), request.size (), remaining ());
141 if (written.status != transport_status::ok)
142 {
143 result.status = serial_probe_status::io_error;
144 result.system_error = written.system_error;
145 result.elapsed = elapsed_since (start);
146 return result;
147 }
148 }
149
150 std::size_t const cap = max_response_bytes < 1U ? 1U : max_response_bytes;
151 std::vector<std::uint8_t> chunk (KPROBE_READ_CHUNK_BYTES);
152
153 for (;;)
154 {
155 if (result.response.size () >= cap)
156 {
157 result.status = serial_probe_status::incomplete;
158 break;
159 }
160
161 std::size_t const space = cap - result.response.size ();
162 std::size_t const capacity
163 = space < chunk.size () ? space : chunk.size ();
164
165 transport_result_t const got
166 = transport.read (chunk.data (), capacity, remaining ());
167
168 if (got.status == transport_status::failed)
169 {
170 result.status = serial_probe_status::io_error;
171 result.system_error = got.system_error;
172 break;
173 }
174
175 // `deadline_exceeded` carries no bytes; `ok` with zero bytes means no
176 // progress. Both stop the loop instead of spinning, and the outcome
177 // depends on what has been collected so far.
178 if (got.status == transport_status::deadline_exceeded || got.bytes == 0U)
179 {
180 result.status = result.response.empty ()
181 ? serial_probe_status::no_response
182 : serial_probe_status::incomplete;
183 break;
184 }
185
186 std::size_t const count = got.bytes < capacity ? got.bytes : capacity;
187 result.response.insert (result.response.end (), chunk.begin (),
188 chunk.begin ()
189 + static_cast<std::ptrdiff_t> (count));
190
191 if (response_complete (result.response))
192 {
193 result.status = serial_probe_status::responded;
194 break;
195 }
196 }
197
198 result.elapsed = elapsed_since (start);
199 return result;
200}
