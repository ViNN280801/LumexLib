229 bool
230 try_add () LUMEX_NOEXCEPT
231 {
232 word_type current = word_.load (std::memory_order_relaxed);
233 for (;;)
234 {
235 if (current == 0)
236 return false;
237 if (word_.compare_exchange_weak (current, current + 1,
238 std::memory_order_acq_rel,
239 std::memory_order_relaxed))
240 return true;
241 }

293 long
294 use_count_settled () const LUMEX_NOEXCEPT
295 {
296 unsigned spins = 0;
297 word_type word = load ();
298 while (ext_of (word) != 0)
299 {
300 if (spins < settled_spin_limit ())
301 ++spins;
302 else
303 std::this_thread::yield ();
304 word = load ();
305 }
306 return count_of (word);
307 }

317 bool
318 drop (word_type unit) LUMEX_NOEXCEPT
319 {
320#if LUMEX_SMART_PTR_TSAN
321 return word_.fetch_sub (unit, std::memory_order_acq_rel) == unit;
322#else
323 if (word_.fetch_sub (unit, std::memory_order_release) != unit)
324 return false;
325 std::atomic_thread_fence (std::memory_order_acquire);
326 return true;
327#endif
328 }
