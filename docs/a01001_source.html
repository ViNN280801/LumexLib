272#if LUMEX_CONTRACTS_CATCH_EXCEPTIONS
273#define LUMEX_CONTRACTS_DETAIL_CHECK(tag, comment, ...) \
274 ( \
275 [&] (::lumex::core::contracts::source_location const \
276 &lumex_contracts_where) -> void \
277 { \
278 bool lumex_contracts_holds = false; \
279 try \
280 { \
281 lumex_contracts_holds = static_cast<bool> ((__VA_ARGS__)); \
282 } \
283 catch (...) \
284 { \
285 ::lumex::core::contracts::detail::tag::violated ( \
286 comment, lumex_contracts_where, \
287 ::lumex::core::contracts::detection_mode:: \
288 evaluation_exception); \
289 return; \
290 } \
291 if (!lumex_contracts_holds) \
292 ::lumex::core::contracts::detail::tag::violated ( \
293 comment, lumex_contracts_where, \
294 ::lumex::core::contracts::detection_mode::predicate_false); \
295 }) (LUMEX_CONTRACTS_DETAIL_LOCATION)
296#else
297#define LUMEX_CONTRACTS_DETAIL_CHECK(tag, comment, ...) \
298 (LUMEX_CONTRACTS_DETAIL_LIKELY (static_cast<bool> ((__VA_ARGS__))) \
299 ? static_cast<void> (0) \
300 : ::lumex::core::contracts::detail::tag::violated ( \
301 comment, LUMEX_CONTRACTS_DETAIL_LOCATION, \
302 ::lumex::core::contracts::detection_mode::predicate_false))
303#endif
