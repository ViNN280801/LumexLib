232template <typename Char, typename WriteBody>
233void
234write_with_width (format_specs_t<Char> const &parsed,
235 BasicFormatContext<Char> &ctx, WriteBody const &write_body)
236{
237 format_specs_t<Char> specs = parsed;
238 specs.width = resolve_dynamic (specs.width_ref, specs.width, ctx);
239 if (specs.width == 0)
240 {
241 write_body (ctx);
242 return;
243 }
244 std::basic_string<Char> text;
245 StringBuffer<Char> buffer (text);
246 std::locale const locale = ctx.locale ();
247 BasicFormatContext<Char> inner (buffer, ctx.args (), &locale);
248 write_body (inner);
249 write_padded (ctx.out ().buffer (), specs, Align::left, text);
250}

311template <typename Char, typename... Types> class TupleFormatter
312{
313public:
314 TupleFormatter ()
315 : _specs (), _separator (widen<Char> (", ")), _open (widen<Char> ("(")),
316 _close (widen<Char> (")"))
317 {
318 }
319
321 void
322 set_separator (BasicStringRef<Char> separator)
323 {
324 _separator.assign (separator.begin (), separator.end ());
325 }
326
328 void
329 set_brackets (BasicStringRef<Char> open, BasicStringRef<Char> close)
330 {
331 _open.assign (open.begin (), open.end ());
332 _close.assign (close.begin (), close.end ());
333 }
334
335 Char const *
336 parse (BasicFormatParseContext<Char> &ctx)
337 {
338 _specs = format_specs_t<Char> (); // brackets set by the user stay
339 Char const *it = parse_range_fill_align_width (ctx, _specs);
340 Char const *const end = ctx.end ();
341 if (it != end && *it == static_cast<Char> ('n'))
342 {
343 _open.clear ();
344 _close.clear ();
345 ++it;
346 }
347 else if (it != end && *it == static_cast<Char> ('m'))
348 {
349 if (sizeof...(Types) != 2)
350 throw FormatError ("invalid format specifier");
351 _open.clear ();
352 _close.clear ();
353 _separator = widen<Char> (": ");
354 ++it;
355 }
356 if (it != end && *it != static_cast<Char> ('}'))
357 throw FormatError ("invalid format specifier");
358 tuple_each<0, sizeof...(Types)>::template parse<Char> (_formatters);
359 return it;
360 }
361
362 template <typename Tuple>
363 BasicAppender<Char>
364 format (Tuple const &value, BasicFormatContext<Char> &ctx) const
365 {
366 TupleFormatter const &self = *this;
367 write_with_width (_specs, ctx,
368 [&self, &value] (BasicFormatContext<Char> &out)
369 {
370 out.out ().buffer ().append (self._open);
371 tuple_each<0, sizeof...(Types)>::format (
372 self._formatters, value, self._separator, out);
373 out.out ().buffer ().append (self._close);
374 });
375 return ctx.out ();
376 }
377
378private:
379 format_specs_t<Char> _specs;
380 std::basic_string<Char> _separator;
381 std::basic_string<Char> _open;
382 std::basic_string<Char> _close;
383 std::tuple<ElementFormatter<
384 lumex::core::utility::traits::meta::CleanType<Types>, Char>...>
385 _formatters;

413template <typename Range, typename Char>
414class Formatter<Range, Char,
415 typename std::enable_if<
416 Detail::is_formattable_range<Range, Char>::value>::type>
417{
418public:
419 using element_type = typename Detail::range_element<Range, Char>::type;
420
421 Formatter ()
422 : _specs (), _presentation (default_presentation ()),
423 _separator (Detail::widen<Char> (", ")), _open (), _close (),
424 _underlying ()
425 {
426 set_default_brackets ();
427 }
428
430 void
431 set_separator (Detail::BasicStringRef<Char> separator)
432 {
433 _separator.assign (separator.begin (), separator.end ());
434 }
435
437 void
438 set_brackets (Detail::BasicStringRef<Char> open,
439 Detail::BasicStringRef<Char> close)
440 {
441 _open.assign (open.begin (), open.end ());
442 _close.assign (close.begin (), close.end ());
443 }
444
446 Formatter<element_type, Char> &
447 underlying () LUMEX_NOEXCEPT
448 {
449 return _underlying;
450 }
451
452 Char const *
453 parse (BasicFormatParseContext<Char> &ctx)
454 {
455 _specs = Detail::format_specs_t<Char> (); // user brackets stay
456 Char const *it = Detail::parse_range_fill_align_width (ctx, _specs);
457 Char const *const end = ctx.end ();
458 if (it != end && *it == static_cast<Char> ('n'))
459 {
460 _open.clear ();
461 _close.clear ();
462 ++it;
463 }
464 if (it != end && *it == static_cast<Char> ('m'))
465 {
466 if (!lumex::core::utility::traits::tuple::is_pair_like<
467 element_type>::value)
468 throw FormatError ("invalid format specifier");
469 if (!_open.empty ())
470 {
471 _open = Detail::widen<Char> ("{");
472 _close = Detail::widen<Char> ("}");
473 }
474 _presentation = presentation::map;
475 ++it;
476 }
477 else if (it != end && *it == static_cast<Char> ('s'))
478 {
479 if (!std::is_same<element_type, Char>::value)
480 throw FormatError ("invalid format specifier");
481 _presentation = presentation::text;
482 ++it;
483 }
484 else if (it != end && *it == static_cast<Char> ('?') && end - it > 1
485 && it[1] == static_cast<Char> ('s'))
486 {
487 if (!std::is_same<element_type, Char>::value)
488 throw FormatError ("invalid format specifier");
489 _presentation = presentation::debug_text;
490 it += 2;
491 }
492 if (it != end && *it == static_cast<Char> (':'))
493 {
494 if (_presentation == presentation::text
495 || _presentation == presentation::debug_text)
496 throw FormatError ("invalid format specifier");
497 ctx.advance_to (it + 1);
498 it = _underlying.parse (ctx);
499 }
500 else
501 Detail::parse_default_element<Char, element_type> (_underlying);
502 if (it != end && *it != static_cast<Char> ('}'))
503 throw FormatError ("invalid format specifier");
504 if (_presentation == presentation::map)
505 Detail::use_map_element_style (_underlying, 0);
506 return it;
507 }

509 BasicAppender<Char>
510 format (Range const &range, BasicFormatContext<Char> &ctx) const

513 Detail::write_with_width (_specs, ctx,

529 static LUMEX_CONSTEXPR presentation

531 {
532 return lumex::core::utility::traits::range::has_key_type<Range>::value
533 && lumex::core::utility::traits::range::has_mapped_type<

536 element_type>::value

539 ? presentation::set

541 }
542
543 void
544 set_default_brackets ()
545 {
546 bool const braces = _presentation == presentation::map
547 || _presentation == presentation::set;
548 _open = Detail::widen<Char> (braces ? "{" : "[");
549 _close = Detail::widen<Char> (braces ? "}" : "]");
550 }
551
553 void
554 write_text (Range const &range, Detail::Buffer<Char> &buffer,
555 std::true_type) const
556 {
557 std::basic_string<Char> text;
558 for (auto it = std::begin (range); it != std::end (range); ++it)
559 text.push_back (*it);
560 if (_presentation == presentation::debug_text)
561 buffer.append (Detail::escape (text.data (), text.data () + text.size (),
562 static_cast<Char> ('"')));
563 else
564 buffer.append (text);
565 }
566
568 void
569 write_text (Range const &, Detail::Buffer<Char> &, std::false_type) const
570 {
571 }
572
573 void
574 write_body (Range const &range, BasicFormatContext<Char> &ctx) const
575 {
576 Detail::Buffer<Char> &buffer = ctx.out ().buffer ();
577 if (_presentation == presentation::text
578 || _presentation == presentation::debug_text)
579 {
580 write_text (range, buffer, std::is_same<element_type, Char> ());
581 return;
582 }
583 buffer.append (_open);
584 bool first = true;
585 for (auto it = std::begin (range); it != std::end (range); ++it)
586 {
587 if (!first)
588 buffer.append (_separator);
589 first = false;
590 _underlying.format (*it, ctx);
591 }
592 buffer.append (_close);
593 }

597 std::basic_string<Char> _separator;
598 std::basic_string<Char> _open;
599 std::basic_string<Char> _close;
600 Formatter<element_type, Char> _underlying;
601};
