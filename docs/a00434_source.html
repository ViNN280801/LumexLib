558 && std::is_integral<clean_U>::value;
560 static bool const both_floating
561 = std::is_floating_point<clean_T>::value
562 && std::is_floating_point<clean_U>::value;
564 static bool const mixed_types
565 = !same_type && !both_integral
566 && !both_floating;

570 using common_type =
571 typename std::common_type<clean_T, clean_U>::type;

886 static bool
887 compare (T current, U other) LUMEX_NOEXCEPT
888 {
889 // Check ranges to prevent overflow
890 // Signedness is taken into account so the comparison is correct
891 if ((std::is_signed<clean_T>::value && std::is_unsigned<clean_U>::value)
892 ? true
893 : false)
894 {
895 // T signed, U unsigned: if current < 0 then current >= other is never
896 // false
897 if (current < 0)
898 return false;
899 // other is always >= 0; only the upper bound is checked
900#ifdef LUMEXLUMEX_SFNC_ARCH_X86
901 using UnsignedT = typename std::make_unsigned<clean_T>::type;
902 if (static_cast<UnsignedT> (other)
903 > static_cast<UnsignedT> ((std::numeric_limits<clean_T>::max) ()))
904 return false;
905#else
906 if (other > static_cast<typename std::make_unsigned<clean_T>::type> (
907 (std::numeric_limits<clean_T>::max) ()))
908 return false;
909#endif
910 }
911 else if ((std::is_unsigned<clean_T>::value
912 && std::is_signed<clean_U>::value)
913 ? true
914 : false)
915 {
916 // T unsigned, U signed: if other < 0 then current >= other is always
917 // true true
918 if (other < 0)
919 return true;
920 // Otherwise check the upper bound
921#ifdef LUMEXLUMEX_SFNC_ARCH_X86
922 using UnsignedU = typename std::make_unsigned<clean_U>::type;
923 if (static_cast<UnsignedU> (other)
924 > static_cast<UnsignedU> ((std::numeric_limits<clean_T>::max) ()))
925 return false;
926#else
927 using UnsignedU = typename std::make_unsigned<clean_U>::type;
928 if (static_cast<UnsignedU> (other)
929 > static_cast<UnsignedU> ((std::numeric_limits<clean_T>::max) ()))
930 return false;
931#endif
932 }
933 else
934 {
935 // Both signed or both unsigned - safe comparison
936#ifdef LUMEXLUMEX_SFNC_ARCH_X86
937 // On 32-bit architectures use explicit casts to silence C4018
938 if (static_cast<CommonType> (other)
939 > static_cast<CommonType> ((std::numeric_limits<clean_T>::max) ()))
940 return false;
941 if (static_cast<CommonType> (other)
942 < static_cast<CommonType> ((std::numeric_limits<clean_T>::min) ()))
943 return true;
944#else
945 if (other > (std::numeric_limits<clean_T>::max) ())
946 return false;
947 if (other < (std::numeric_limits<clean_T>::min) ())
948 return true;
949#endif
950 }
951 return static_cast<CommonType> (current)
952 >= static_cast<CommonType> (other);
953 }
