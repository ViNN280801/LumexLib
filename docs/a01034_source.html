88struct utf8_counter
89{

93 low (value_type result,
94 std::uint32_t chr) // NOLINT(bugprone-easily-swappable-parameters)

106 }
107

113 }

165struct utf8_writer
166{
167 using value_type = std::uint8_t *;
168
169 static value_type
170 low (value_type result,
171 std::uint32_t chr) // NOLINT(bugprone-easily-swappable-parameters)
172 {
173 // U+0000..U+007F
174 if (chr < 0x80)
175 {
176 *result = static_cast<std::uint8_t> (chr);
177 return result + 1;
178 }
179 // U+0080..U+07FF
180 if (chr < 0x800)
181 {
182 result[0] = static_cast<std::uint8_t> (0xC0 | (chr >> 6));
183 result[1] = static_cast<std::uint8_t> (0x80 | (chr & 0x3F));
184 return result + 2;
185 }
186 // U+0800..U+FFFF
187 result[0] = static_cast<std::uint8_t> (0xE0 | (chr >> 12));
188 result[1] = static_cast<std::uint8_t> (0x80 | ((chr >> 6) & 0x3F));
189 result[2] = static_cast<std::uint8_t> (0x80 | (chr & 0x3F));
190 return result + 3;
191 }
192
193 static value_type
194 high (value_type result, std::uint32_t chr)
195 {
196 // U+10000..U+10FFFF
197 result[0] = static_cast<std::uint8_t> (0xF0 | (chr >> 18));
198 result[1] = static_cast<std::uint8_t> (0x80 | ((chr >> 12) & 0x3F));
199 result[2] = static_cast<std::uint8_t> (0x80 | ((chr >> 6) & 0x3F));
200 result[3] = static_cast<std::uint8_t> (0x80 | (chr & 0x3F));
201 return result + 4;
202 }
203
204 static value_type
205 any (value_type result, std::uint32_t chr)
206 {
207 return (chr < 0x10000) ? low (result, chr) : high (result, chr);
208 }
209};

316struct utf8_decoder
317{
318 using type = std::uint8_t;
319
320 template <typename Traits>
321 static typename Traits::value_type
322 process (std::uint8_t const *data, std::size_t size,
323 typename Traits::value_type result, Traits /* unused */)
324 {
325 std::uint8_t const utf8_byte_mask = 0x3f;
326
327 while (size)
328 {
329 std::uint8_t lead = *data;
330
331 // 0xxxxxxx -> U+0000..U+007F
332 if (lead < 0x80)
333 {
334 result = Traits::low (result, lead);
335 data += 1;
336 size -= 1;
337
338 // process aligned single-byte (ascii) blocks
339 if ((reinterpret_cast<std::uintptr_t> (data) & 3) == 0)
340 {
341 // round-trip through void* to silence 'cast increases required
342 // alignment of target type' warnings
343 while (
344 size >= 4
345 && (*static_cast<std::uint32_t const *> (
346 static_cast< // NOLINT(bugprone-casting-through-void)
347 void const *> (data))
348 & 0x80808080)
349 == 0)
350 {
351 result = Traits::low (result, data[0]);
352 result = Traits::low (result, data[1]);
353 result = Traits::low (result, data[2]);
354 result = Traits::low (result, data[3]);
355 data += 4;
356 size -= 4;
357 }
358 }
359 }
360 // 110xxxxx -> U+0080..U+07FF
361 else if (static_cast<unsigned int> (lead - 0xC0) < 0x20 && size >= 2
362 && (data[1] & 0xc0) == 0x80)
363 {
364 result = Traits::low (result, static_cast<std::uint32_t> (
365 ((lead & ~0xC0) << 6)
366 | (data[1] & utf8_byte_mask)));
367 data += 2;
368 size -= 2;
369 }
370 // 1110xxxx -> U+0800-U+FFFF
371 else if (static_cast<unsigned int> (lead - 0xE0) < 0x10 && size >= 3
372 && (data[1] & 0xc0) == 0x80 && (data[2] & 0xc0) == 0x80)
373 {
374 result = Traits::low (
375 result,
376 static_cast<std::uint32_t> (((lead & ~0xE0) << 12)
377 | ((data[1] & utf8_byte_mask) << 6)
378 | (data[2] & utf8_byte_mask)));
379 data += 3;
380 size -= 3;
381 }
382 // 11110xxx -> U+10000..U+10FFFF
383 else if (static_cast<unsigned int> (lead - 0xF0) < 0x08 && size >= 4
384 && (data[1] & 0xc0) == 0x80 && (data[2] & 0xc0) == 0x80
385 && (data[3] & 0xc0) == 0x80)
386 {
387 result = Traits::high (result,
388 static_cast<std::uint32_t> (
389 ((lead & ~0xF0) << 18)
390 | ((data[1] & utf8_byte_mask) << 12)
391 | ((data[2] & utf8_byte_mask) << 6)
392 | (data[3] & utf8_byte_mask)));
393 data += 4;
394 size -= 4;
395 }
396 // 10xxxxxx or 11111xxx -> invalid
397 else
398 {
399 data += 1;
400 size -= 1;
401 }
402 }
403
404 return result;
405 }

418template <bool SwapBytes> struct utf16_decoder
419{
420 using type = std::uint16_t;
421
422 template <typename Traits>
423 static typename Traits::value_type
424 process (std::uint16_t const *data, std::size_t size,
425 typename Traits::value_type result, Traits /*unused*/)
426 {
427 while (size)
428 {
429 std::uint16_t lead
430 = SwapBytes ? utility::bit::byte_swap (*data) : *data;
431
432 // U+0000..U+D7FF
433 if (lead < 0xD800)
434 { // NOLINT(bugprone-branch-clone)
435 result = Traits::low (result, lead);
436 data += 1;
437 size -= 1;
438 }
439 // U+E000..U+FFFF
440 else if (static_cast<unsigned int> (lead - 0xE000) < 0x2000)
441 {
442 result = Traits::low (result, lead);
443 data += 1;
444 size -= 1;
445 }
446 // surrogate pair lead
447 else if (static_cast<unsigned int> (lead - 0xD800) < 0x400
448 && size >= 2)
449 {
450 std::uint16_t next
451 = SwapBytes ? utility::bit::byte_swap (data[1]) : data[1];
452
453 if (static_cast<unsigned int> (next - 0xDC00) < 0x400)
454 {
455 result = Traits::high (
456 result,
457 static_cast<std::uint32_t> (
458 0x10000 + ((lead & 0x3ff) << 10) + (next & 0x3ff)));
459 data += 2;
460 size -= 2;
461 }
462 else
463 {
464 data += 1;
465 size -= 1;
466 }
467 }
468 else
469 {
470 data += 1;
471 size -= 1;
472 }
473 }
474
475 return result;
476 }

485template <bool SwapBytes> struct utf32_decoder
486{
487 using type = std::uint32_t;
488
489 template <typename Traits>
490 static typename Traits::value_type
491 process (std::uint32_t const *data, std::size_t size,
492 typename Traits::value_type result, Traits /* unused */)
493 {
494 while (size)
495 {
496 std::uint32_t lead
497 = SwapBytes ? utility::bit::byte_swap (*data) : *data;
498
499 // U+0000..U+FFFF
500 if (lead < 0x10000)
501 {
502 result = Traits::low (result, lead);
503 data += 1;
504 size -= 1;
505 }
506 // U+10000..U+10FFFF
507 else
508 {
509 result = Traits::high (result, lead);
510 data += 1;
511 size -= 1;
512 }
513 }
514
515 return result;
516 }
