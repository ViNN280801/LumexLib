247LUMEX_ATOMIC_WAIT_TABLE_VISIBILITY inline waiter_stripe_t &
248stripe_for (void const *address) LUMEX_NOEXCEPT
249{
250 alignas (
251 waiter_table_t) static unsigned char storage[sizeof (waiter_table_t)];
252 static waiter_table_t *const table
253 = ::new (static_cast<void *> (storage)) waiter_table_t ();
254 // The words are at least four-byte aligned, so the two lowest bits carry
255 // no information; the shifts mix the higher bits into the index.
256 std::uintptr_t key = reinterpret_cast<std::uintptr_t> (address) >> 2;
257 key ^= key >> 7;
258 key ^= key >> 13;
259 return table
260 ->stripes[static_cast<std::size_t> (key % waiter_table_t::stripe_count)];
261}

273inline void
274wait_until_changed (std::atomic<std::uint32_t> const &word,
275 std::uint32_t old) LUMEX_NOEXCEPT
276{
277 if (word.load (std::memory_order_relaxed) != old)
278 return;
279 waiter_stripe_t &stripe = stripe_for (&word);
280 std::unique_lock<std::mutex> guard (stripe.mutex);
281 while (word.load (std::memory_order_relaxed) == old)
282 stripe.condition.wait (guard);
283}

292inline void
293notify_all (std::atomic<std::uint32_t> &word) LUMEX_NOEXCEPT
294{
295 waiter_stripe_t &stripe = stripe_for (&word);
296 {
297 std::lock_guard<std::mutex> const guard (stripe.mutex);
298 }
299 stripe.condition.notify_all ();
300}
