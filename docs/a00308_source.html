752template <typename Char>
753LUMEX_FORMAT_CONSTEXPR std::size_t
754code_point_length (Char const *it, Char const *end) LUMEX_NOEXCEPT
755{
756 std::size_t length = 1;
757 if (sizeof (Char) == 1)
758 {
759 unsigned char const lead = static_cast<unsigned char> (*it);
760 if (lead >= 0xF0 && lead < 0xF8)
761 length = 4;
762 else if (lead >= 0xE0)
763 length = lead < 0xF0 ? 3 : 1;
764 else if (lead >= 0xC0)
765 length = 2;
766 }
767 else if (sizeof (Char) == 2)
768 {
769 unsigned const unit = static_cast<unsigned> (*it) & 0xFFFFu;
770 if (unit >= 0xD800u && unit < 0xDC00u)
771 length = 2;
772 }
773 if (static_cast<std::size_t> (end - it) < length)
774 length = 1;
775 return length;
776}

779template <typename Char>
780LUMEX_FORMAT_CONSTEXPR int
781parse_nonnegative_int (Char const *&it, Char const *end)
782{
783 unsigned long long value = 0;
784 while (it != end && is_digit (*it))
785 {
786 value
787 = value * 10 + static_cast<unsigned> (*it - static_cast<Char> ('0'));
788 if (value > static_cast<unsigned long long> (INT_MAX))
789 throw FormatError ("number is too big");
790 ++it;
791 }
792 return static_cast<int> (value);
793}

799template <typename Char>
800LUMEX_FORMAT_CONSTEXPR arg_ref_t<Char>
801parse_arg_id (Char const *&it, Char const *end)
802{
803 arg_ref_t<Char> ref;
804 if (it != end && is_digit (*it))
805 {
806 if (*it == static_cast<Char> ('0') && it + 1 != end && is_digit (it[1]))
807 throw FormatError ("invalid format string");
808 // An index above INT_MAX cannot name an argument: saturate, so the
809 // lookup reports "argument not found" rather than a number error.
810 unsigned long long value = 0;
811 while (it != end && is_digit (*it))
812 {
813 if (value <= static_cast<unsigned long long> (INT_MAX))
814 value = value * 10
815 + static_cast<unsigned> (*it - static_cast<Char> ('0'));
816 ++it;
817 }
818 ref.kind = ArgRefKind::index;
819 ref.index = value > static_cast<unsigned long long> (INT_MAX)
820 ? INT_MAX
821 : static_cast<int> (value);
822 return ref;
823 }
824 if (it != end && is_name_start (*it))
825 {
826 Char const *const start = it;
827 while (it != end && (is_name_start (*it) || is_digit (*it)))
828 ++it;
829 ref.kind = ArgRefKind::name;
830 ref.name = start;
831 ref.name_size = static_cast<std::size_t> (it - start);
832 return ref;
833 }
834 throw FormatError ("invalid format string");
835}

940template <typename Char>
941LUMEX_FORMAT_CONSTEXPR arg_ref_t<Char>
942parse_dynamic_ref (Char const *&it, Char const *end,
943 BasicFormatParseContext<Char> &ctx)
944{
945 ++it; // '{'
946 arg_ref_t<Char> ref;
947 if (it != end && *it == static_cast<Char> ('}'))
948 {
949 ref.kind = ArgRefKind::index;
950 ref.index = ctx.next_arg_id ();
951 ctx.check_dynamic_spec (ref.index);
952 }
953 else
954 {
955 ref = parse_arg_id (it, end);
956 if (ref.kind == ArgRefKind::index)
957 {
958 ctx.check_arg_id (ref.index);
959 ctx.check_dynamic_spec (ref.index);
960 }
961 else
962 ctx.check_named_arg ();
963 }
964 if (it == end || *it != static_cast<Char> ('}'))
965 throw FormatError ("invalid format string");
966 ++it;
967 return ref;
968}

994template <typename Char>
995LUMEX_FORMAT_CONSTEXPR void
996validate_specs (format_specs_t<Char> const &specs, SpecKind kind)
997{
998 Char const type = specs.type;
999 bool numeric = false;
1000 bool allows_precision = false;
1001 bool allows_localized = false;
1002 bool allows_zero = false;
1003 char const *const invalid = kind == SpecKind::character
1004 ? "invalid format specifier for char"
1005 : "invalid format specifier";
1006 switch (kind)
1007 {
1008 case SpecKind::integer:
1009 if (type != Char () && type != static_cast<Char> ('c')
1010 && !is_integer_presentation (type))
1011 throw FormatError ("invalid format specifier");
1012 numeric = type != static_cast<Char> ('c');
1013 allows_localized = numeric;
1014 allows_zero = numeric;
1015 break;
1016 case SpecKind::character:
1017 if (type != Char () && type != static_cast<Char> ('c')
1018 && type != static_cast<Char> ('?')
1019 && !is_integer_presentation (type))
1020 throw FormatError ("invalid format specifier");
1021 numeric = is_integer_presentation (type);
1022 allows_localized = numeric;
1023 allows_zero = numeric;
1024 break;
1025 case SpecKind::boolean:
1026 if (type != Char () && type != static_cast<Char> ('s')
1027 && !is_integer_presentation (type))
1028 throw FormatError ("invalid format specifier");
1029 numeric = is_integer_presentation (type);
1030 allows_localized = true;
1031 allows_zero = numeric;
1032 break;
1033 case SpecKind::floating:
1034 if (type != Char () && !is_float_presentation (type))
1035 throw FormatError ("invalid format specifier");
1036 numeric = true;
1037 allows_precision = true;
1038 allows_localized = true;
1039 allows_zero = true;
1040 break;
1041 case SpecKind::string:
1042 if (type != Char () && type != static_cast<Char> ('s')
1043 && type != static_cast<Char> ('?'))
1044 throw FormatError ("invalid format specifier");
1045 allows_precision = true;
1046 break;
1047 case SpecKind::pointer:
1048 if (type != Char () && type != static_cast<Char> ('p')
1049 && type != static_cast<Char> ('P'))
1050 throw FormatError ("invalid format specifier");
1051 // P2510 (a defect report against C++20): zeros pad after "0x".
1052 allows_zero = true;
1053 break;
1054 case SpecKind::enumeration:
1055 if (type != Char () && type != static_cast<Char> ('s')
1056 && !is_integer_presentation (type))
1057 throw FormatError ("invalid format specifier");
1058 numeric = is_integer_presentation (type);
1059 allows_precision = !numeric;
1060 allows_localized = numeric;
1061 allows_zero = numeric;
1062 break;
1063 }
1064 if ((specs.sign != Sign::none || specs.alternate) && !numeric)
1065 throw FormatError (invalid);
1066 if (specs.zero_pad && !allows_zero)
1067 throw FormatError (kind == SpecKind::character
1068 ? invalid
1069 : "format specifier requires numeric argument");
1070 if ((specs.precision >= 0 || specs.precision_ref.kind != ArgRefKind::none)
1071 && !allows_precision)
1072 throw FormatError ("invalid format specifier");
1073 if (specs.localized && !allows_localized)
1074 throw FormatError ("invalid format specifier");
1075}

1082template <typename Char>
1083LUMEX_FORMAT_CONSTEXPR Char const *
1084parse_std_specs (BasicFormatParseContext<Char> &ctx,
1085 format_specs_t<Char> &specs, SpecKind kind)
1086{
1087 Char const *it = ctx.begin ();
1088 Char const *const end = ctx.end ();
1089 if (it == end || *it == static_cast<Char> ('}'))
1090 {
1091 validate_specs (specs, kind);
1092 return it;
1093 }
1094
1095 // [[fill]align]
1096 std::size_t const fill_length = code_point_length (it, end);
1097 if (static_cast<std::size_t> (end - it) > fill_length
1098 && is_align (it[fill_length]))
1099 {
1100 if (*it == static_cast<Char> ('{'))
1101 throw FormatError ("invalid fill character '{'");
1102 if (*it == static_cast<Char> ('}'))
1103 throw FormatError ("invalid fill character '}'");
1104 for (std::size_t i = 0; i < fill_length; ++i)
1105 specs.fill[i] = it[i];
1106 specs.fill_size = static_cast<unsigned char> (fill_length);
1107 it += fill_length;
1108 }
1109 if (it != end && is_align (*it))
1110 {
1111 specs.align = *it == static_cast<Char> ('<') ? Align::left
1112 : *it == static_cast<Char> ('>') ? Align::right
1113 : Align::center;
1114 ++it;
1115 }
1116
1117 // [sign]
1118 if (it != end
1119 && (*it == static_cast<Char> ('+') || *it == static_cast<Char> ('-')
1120 || *it == static_cast<Char> (' ')))
1121 {
1122 specs.sign = *it == static_cast<Char> ('+') ? Sign::plus
1123 : *it == static_cast<Char> ('-') ? Sign::minus
1124 : Sign::space;
1125 ++it;
1126 }
1127
1128 // [#]
1129 if (it != end && *it == static_cast<Char> ('#'))
1130 {
1131 specs.alternate = true;
1132 ++it;
1133 }
1134
1135 // [0]
1136 if (it != end && *it == static_cast<Char> ('0'))
1137 {
1138 specs.zero_pad = true;
1139 ++it;
1140 // The width that may follow starts with a nonzero digit (`{:00}`
1141 // is an error, as in std::format).
1142 if (it != end && *it == static_cast<Char> ('0'))
1143 throw FormatError ("invalid format specifier");
1144 }
1145
1146 // [width]
1147 if (it != end && is_digit (*it))
1148 specs.width = parse_nonnegative_int (it, end);
1149 else if (it != end && *it == static_cast<Char> ('{'))
1150 specs.width_ref = parse_dynamic_ref (it, end, ctx);
1151
1152 // [.precision]
1153 if (it != end && *it == static_cast<Char> ('.'))
1154 {
1155 ++it;
1156 if (it != end && is_digit (*it))
1157 specs.precision = parse_nonnegative_int (it, end);
1158 else if (it != end && *it == static_cast<Char> ('{'))
1159 specs.precision_ref = parse_dynamic_ref (it, end, ctx);
1160 else if (it == end)
1161 throw FormatError ("invalid precision");
1162 else
1163 throw FormatError ("invalid format string");
1164 }
1165
1166 // [L]
1167 if (it != end && *it == static_cast<Char> ('L'))
1168 {
1169 specs.localized = true;
1170 ++it;
1171 }
1172
1173 // [type]
1174 if (it != end && *it != static_cast<Char> ('}'))
1175 {
1176 specs.type = *it;
1177 ++it;
1178 }
1179
1180 if (it != end && *it != static_cast<Char> ('}'))
1181 throw FormatError ("invalid format specifier");
1182 validate_specs (specs, kind);
1183 return it;
1184}

1187template <typename Char>
1188LUMEX_FORMAT_CONSTEXPR Char const *
1189skip_specs (Char const *it, Char const *end)
1190{
1191 int depth = 0;
1192 for (; it != end; ++it)
1193 {
1194 if (*it == static_cast<Char> ('{'))
1195 ++depth;
1196 else if (*it == static_cast<Char> ('}'))
1197 {
1198 if (depth == 0)
1199 return it;
1200 --depth;
1201 }
1202 }
1203 return it;

1211template <typename Char, typename Handler>
1212LUMEX_FORMAT_CONSTEXPR void
1213parse_format_string (Char const *begin, Char const *end, Handler &handler)
1214{
1215 Char const *text = begin;
1216 Char const *it = begin;
1217 while (it != end)
1218 {
1219 Char const value = *it;
1220 if (value == static_cast<Char> ('{'))
1221 {
1222 handler.on_text (text, it);
1223 handler.on_field_start (static_cast<std::size_t> (it - begin));
1224 ++it;
1225 if (it == end)
1226 throw FormatError ("invalid format string");
1227 if (*it == static_cast<Char> ('{'))
1228 {
1229 handler.on_text (it, it + 1);
1230 ++it;
1231 text = it;
1232 continue;
1233 }
1234 if (*it == static_cast<Char> ('}') || *it == static_cast<Char> (':'))
1235 handler.on_auto_arg ();
1236 else
1237 {
1238 arg_ref_t<Char> const ref = parse_arg_id (it, end);
1239 if (ref.kind == ArgRefKind::index)
1240 handler.on_index_arg (ref.index);
1241 else
1242 handler.on_name_arg (ref.name, ref.name_size);
1243 }
1244 if (it == end)
1245 throw FormatError ("invalid format string");
1246 if (*it == static_cast<Char> (':'))
1247 ++it;
1248 else if (*it != static_cast<Char> ('}'))
1249 throw FormatError ("invalid format string");
1250 it = handler.on_format_specs (it, end);
1251 if (it == end)
1252 throw FormatError ("missing '}' in format string");
1253 if (*it != static_cast<Char> ('}'))
1254 throw FormatError ("unknown format specifier");
1255 ++it;
1256 text = it;
1257 }
1258 else if (value == static_cast<Char> ('}'))
1259 {
1260 handler.on_text (text, it);
1261 handler.on_field_start (static_cast<std::size_t> (it - begin));
1262 ++it;
1263 if (it == end || *it != static_cast<Char> ('}'))
1264 throw FormatError ("unmatched '}' in format string");
1265 handler.on_text (it, it + 1);
1266 ++it;
1267 text = it;
1268 }
1269 else
1270 ++it;
1271 }
1272 handler.on_text (text, end);
1273}

1502template <typename Char>
1503std::uint32_t
1504decode (Char const *&it, Char const *end, bool &valid) LUMEX_NOEXCEPT
1505{
1506 valid = true;
1507 if (sizeof (Char) == 1)
1508 {
1509 unsigned char const lead = static_cast<unsigned char> (*it);
1510 std::size_t length = 0;
1511 std::uint32_t cp = 0;
1512 if (lead < 0x80)
1513 {
1514 ++it;
1515 return lead;
1516 }
1517 if (lead >= 0xC2 && lead < 0xE0)
1518 {
1519 length = 2;
1520 cp = lead & 0x1Fu;
1521 }
1522 else if (lead >= 0xE0 && lead < 0xF0)
1523 {
1524 length = 3;
1525 cp = lead & 0x0Fu;
1526 }
1527 else if (lead >= 0xF0 && lead < 0xF5)
1528 {
1529 length = 4;
1530 cp = lead & 0x07u;
1531 }
1532 if (length == 0 || static_cast<std::size_t> (end - it) < length)
1533 {
1534 valid = false;
1535 ++it;
1536 return lead;
1537 }
1538 for (std::size_t i = 1; i < length; ++i)
1539 {
1540 unsigned char const unit = static_cast<unsigned char> (it[i]);
1541 if ((unit & 0xC0u) != 0x80u)
1542 {
1543 valid = false;
1544 ++it;
1545 return lead;
1546 }
1547 cp = (cp << 6) | (unit & 0x3Fu);
1548 }
1549 it += length;
1550 return cp;
1551 }
1552 if (sizeof (Char) == 2)
1553 {
1554 std::uint32_t const unit = static_cast<std::uint32_t> (*it) & 0xFFFFu;
1555 if (unit >= 0xD800u && unit < 0xDC00u && end - it >= 2)
1556 {
1557 std::uint32_t const low
1558 = static_cast<std::uint32_t> (it[1]) & 0xFFFFu;
1559 if (low >= 0xDC00u && low < 0xE000u)
1560 {
1561 it += 2;
1562 return 0x10000u + ((unit - 0xD800u) << 10) + (low - 0xDC00u);
1563 }
1564 }
1565 if (unit >= 0xD800u && unit < 0xE000u)
1566 valid = false;
1567 ++it;
1568 return unit;
1569 }
1570 std::uint32_t const unit = static_cast<std::uint32_t> (*it);
1571 ++it;
1572 return unit;
1573}

1740template <typename Char>
1741std::size_t
1742next_grapheme (Char const *&it, Char const *end) LUMEX_NOEXCEPT
1743{
1744 if (*it == static_cast<Char> ('\r') && it + 1 != end
1745 && it[1] == static_cast<Char> ('\n'))
1746 {
1747 it += 2; // CR LF is one cluster
1748 return 1;
1749 }
1750 if (static_cast<std::uint32_t> (*it) < 0x80u
1751 && (it + 1 == end || static_cast<std::uint32_t> (it[1]) < 0x80u))
1752 {
1753 ++it; // ASCII followed by ASCII (or the end): the common fast path
1754 return 1;
1755 }
1756 bool valid = true;
1757 std::uint32_t const first = decode (it, end, valid);
1758 if (!valid)
1759 return 1;
1760 std::size_t const width = is_wide_code_point (first) ? 2 : 1;
1761 std::uint32_t previous = first;
1762 bool regional_pair_open = is_regional_indicator (first);
1763 while (it != end)
1764 {
1765 Char const *next = it;
1766 bool next_valid = true;
1767 std::uint32_t const cp = decode (next, end, next_valid);
1768 if (!next_valid)
1769 break;
1770 bool joins = is_grapheme_extend (cp) || cp == 0x200Du;
1771 if (!joins && previous == 0x200Du && is_extended_pictographic (cp))
1772 joins = true;
1773 if (!joins && regional_pair_open && is_regional_indicator (cp))
1774 {
1775 joins = true;
1776 regional_pair_open = false;
1777 }
1778 if (!joins)
1779 break;
1780 previous = cp;
1781 it = next;
1782 }
1783 return width;
1784}

2664template <typename Char, typename F>
2665void
2666write_float (Buffer<Char> &buffer, format_specs_t<Char> const &specs, F value,
2667 BasicFormatContext<Char> const &ctx)
2668{
2669 Char const type = specs.type;
2670 bool const upper
2671 = type == static_cast<Char> ('E') || type == static_cast<Char> ('F')
2672 || type == static_cast<Char> ('G') || type == static_cast<Char> ('A');
2673 bool const negative = std::signbit (value);
2674 std::basic_string<Char> const sign = sign_text<Char> (negative, specs.sign);
2675 if (!std::isfinite (value))
2676 {
2677 std::string text = std::isnan (value) ? "nan" : "inf";
2678 if (upper)
2679 for (std::size_t i = 0; i < text.size (); ++i)
2680 text[i] = static_cast<char> (text[i] - 'a' + 'A');
2681 write_number (buffer, specs, sign, widen<Char> (text), false);
2682 return;
2683 }
2684 F const magnitude = negative ? -value : value;
2685 FloatStyle style = FloatStyle::shortest;
2686 int precision = specs.precision;
2687 if (type == Char ())
2688 style = precision < 0 ? FloatStyle::shortest : FloatStyle::general;
2689 else if (type == static_cast<Char> ('a') || type == static_cast<Char> ('A'))
2690 style = FloatStyle::hex;
2691 else if (type == static_cast<Char> ('e') || type == static_cast<Char> ('E'))
2692 {
2693 style = FloatStyle::scientific;
2694 if (precision < 0)
2695 precision = 6;
2696 }
2697 else if (type == static_cast<Char> ('f') || type == static_cast<Char> ('F'))
2698 {
2699 style = FloatStyle::fixed;
2700 if (precision < 0)
2701 precision = 6;
2702 }
2703 else
2704 {
2705 style = FloatStyle::general;
2706 if (precision < 0)
2707 precision = 6;
2708 }
2709 // Fixed / scientific output grows with the precision: refuse a size that
2710 // cannot be represented rather than trying to allocate it.
2711 if ((style == FloatStyle::fixed || style == FloatStyle::scientific)
2712 && precision > INT_MAX - 1024)
2713 throw FormatError ("number is too big");
2714 // Past the number of significant digits a value can have, a larger
2715 // general precision prints the same text; cap it to bound the buffer.
2716 int const general_cap = std::numeric_limits<F>::digits
2717 - std::numeric_limits<F>::min_exponent + 16;
2718 if (style == FloatStyle::general && precision > general_cap)
2719 precision = general_cap;

2721 specs.alternate && type != Char ());
2722 if (upper)
2723 for (std::size_t i = 0; i < text.size (); ++i)
2724 if (text[i] >= 'a' && text[i] <= 'z')
2725 text[i] = static_cast<char> (text[i] - 'a' + 'A');
2726
2727 std::basic_string<Char> body = widen<Char> (text);
2728 if (specs.localized && style != FloatStyle::hex)
2729 {
2730 std::numpunct<Char> const &facet
2731 = std::use_facet<std::numpunct<Char>> (ctx.locale ());
2732 std::size_t integral = 0;
2733 while (integral < body.size () && is_digit (body[integral]))
2734 ++integral;
2735 std::basic_string<Char> rest = body.substr (integral);
2736 if (!rest.empty () && rest[0] == static_cast<Char> ('.'))
2737 rest[0] = facet.decimal_point ();
2738 body = group_digits (body.substr (0, integral), facet.grouping (),
2739 facet.thousands_sep ())
2740 + rest;
2741 }
2742 write_number (buffer, specs, sign, body, true);
2743}
