195// The `ordering_value` of a `std::` ordering as a number (-1, 0, 1, and 2 for
196// the one `std::partial_ordering` that is neither). They compare with the
197// named values, not with the literal 0 the standard classes also take: Clang
198// warns of a "zero as null pointer constant" for that literal in a template.
200LUMEX_CONSTEXPR inline signed char
201std_ordering_value (std::strong_ordering value) LUMEX_NOEXCEPT
202{
203 return value == std::strong_ordering::less ? static_cast<signed char> (-1)
204 : value == std::strong_ordering::greater
205 ? static_cast<signed char> (1)
206 : static_cast<signed char> (0);
207}
208
210LUMEX_CONSTEXPR inline signed char
211std_ordering_value (std::weak_ordering value) LUMEX_NOEXCEPT
212{
213 return value == std::weak_ordering::less ? static_cast<signed char> (-1)
214 : value == std::weak_ordering::greater ? static_cast<signed char> (1)
215 : static_cast<signed char> (0);
216}
217
219LUMEX_CONSTEXPR inline signed char
220std_ordering_value (std::partial_ordering value) LUMEX_NOEXCEPT
221{
222 return value == std::partial_ordering::less ? static_cast<signed char> (-1)
223 : value == std::partial_ordering::greater
224 ? static_cast<signed char> (1)
225 : value == std::partial_ordering::equivalent
226 ? static_cast<signed char> (0)
227 : static_cast<signed char> (2);
228}

660// The functions below are templates over the own classes, not functions of
661// `partial_ordering`: with the converting constructors of C++20 a
662// `partial_ordering` parameter would also take a `std::` ordering, and a call
663// `is_eq (a <=> b)` in a scope that has `using namespace` of this namespace
664// would be ambiguous with `std::is_eq` found through argument-dependent
665// lookup.
666
674template <typename Ordering>
675LUMEX_CONSTEXPR
676 typename std::enable_if<Detail::is_own_ordering<Ordering>::value,
677 bool>::type

679{
