213template <typename Holder>
214Holder *
215split_make_holder (::lumex::core::smart_ptr::detail::ctl_base *owner,
216 void *pointer) LUMEX_NOEXCEPT
217{
218 Holder *holder = nullptr;
219 LUMEX_SMART_PTR_TRY { holder = Holder::create (owner, pointer); }
