140 template <class Y, class = typename std::enable_if<
141 detail::is_new_compatible<Y, T>::value>::type>
142 explicit shared_ptr (Y *p) : ptr_ (p), cb_ (nullptr)
143 {
144 typedef typename detail::default_deleter_for<T, Y>::type deleter_type;
145 detail::check_std_enable_shared_from_this<Y> ();
146 deleter_type deleter;

148 enable_from_this (p, std::is_array<T> ());

487 // Creates the control block of a pointer and a deleter. The deleter is
488 // called with the pointer when the block cannot be created
489 // ([util.smartptr.shared.const]).
490 template <class P, class D, class A>
491 void
492 create_owner (P p, D &d, A const &a)
493 {
494 typedef detail::ctl_ptr<P, D, A> block_type;
495 LUMEX_SMART_PTR_TRY { cb_ = block_type::create (p, std::move (d), a); }
496 LUMEX_SMART_PTR_CATCH_ALL
497 {
498 d (p);
499 LUMEX_SMART_PTR_RETHROW;
500 }
501 cb_->set_anchor (reinterpret_cast<std::uintptr_t> (ptr_));
502 }

517 // unique_ptr with a deleter object: the deleter moves into the block.
518 template <class Y, class D>
519 void
520 adopt_unique (std::unique_ptr<Y, D> &r, std::false_type)
521 {
522 typename std::unique_ptr<Y, D>::pointer const p = r.get ();
523 if (p == nullptr)
524 return;
525 cb_ = detail::ctl_ptr<typename std::unique_ptr<Y, D>::pointer, D,
526 detail::block_allocator<void>>::
527 create (p, std::move (r.get_deleter ()),
528 detail::block_allocator<void> ());
529 ptr_ = p;
530 cb_->set_anchor (reinterpret_cast<std::uintptr_t> (ptr_));
531 r.release ();
532 enable_from_this (static_cast<element_type *> (ptr_), std::is_array<T> ());
533 }
534
535 // unique_ptr with a deleter reference: the block refers to the deleter.
536 template <class Y, class D>
537 void
538 adopt_unique (std::unique_ptr<Y, D> &r, std::true_type)
539 {
540 typedef typename std::remove_reference<D>::type deleter_object;
541 typedef std::reference_wrapper<deleter_object> deleter_ref;
542 typename std::unique_ptr<Y, D>::pointer const p = r.get ();
543 if (p == nullptr)
544 return;
545 cb_ = detail::ctl_ptr<typename std::unique_ptr<Y, D>::pointer, deleter_ref,
546 detail::block_allocator<void>>::
547 create (p, deleter_ref (r.get_deleter ()),
548 detail::block_allocator<void> ());
549 ptr_ = p;
550 cb_->set_anchor (reinterpret_cast<std::uintptr_t> (ptr_));
551 r.release ();
552 enable_from_this (static_cast<element_type *> (ptr_), std::is_array<T> ());
553 }
554
555 element_type *ptr_;
556 detail::ctl_base *cb_;
557};
