81inline long long
82floor_div (long long value, long long divisor) LUMEX_NOEXCEPT
83{
84 long long quotient = value / divisor;
85 if ((value % divisor != 0) && ((value < 0) != (divisor < 0)))
86 --quotient;
87 return quotient;
88}

91inline long long
92floor_mod (long long value, long long divisor) LUMEX_NOEXCEPT
93{
94 return value - floor_div (value, divisor) * divisor;
95}

98inline long long
99days_from_civil (long long year_value, int month_value,
100 int day_value) LUMEX_NOEXCEPT
101{
102 year_value -= month_value <= 2 ? 1 : 0;
103 long long const era = floor_div (year_value, 400);
104 long long const year_of_era = year_value - era * 400;
105 long long const day_of_year
106 = (153 * (month_value > 2 ? month_value - 3 : month_value + 9) + 2) / 5
107 + day_value - 1;
108 long long const day_of_era
109 = year_of_era * 365 + year_of_era / 4 - year_of_era / 100 + day_of_year;
110 return era * 146097 + day_of_era - 719468;
111}

132inline void
133civil_from_days (long long day_count, civil_time_t &time) LUMEX_NOEXCEPT
134{
135 long long const shifted = day_count + 719468;
136 long long const era = floor_div (shifted, 146097);
137 long long const day_of_era = shifted - era * 146097;
138 long long const year_of_era = (day_of_era - day_of_era / 1460
139 + day_of_era / 36524 - day_of_era / 146096)
140 / 365;
141 long long const day_of_year
142 = day_of_era - (365 * year_of_era + year_of_era / 4 - year_of_era / 100);
143 long long const month_index = (5 * day_of_year + 2) / 153;
144 time.day = static_cast<int> (day_of_year - (153 * month_index + 2) / 5 + 1);
145 time.month = static_cast<int> (month_index < 10 ? month_index + 3
146 : month_index - 9);
147 time.year = year_of_era + era * 400 + (time.month <= 2 ? 1 : 0);
148 time.weekday = static_cast<int> (floor_mod (day_count + 4, 7));
149 time.year_day
150 = static_cast<int> (day_count - days_from_civil (time.year, 1, 1));
151}

158inline long long
159power_of_ten (int exponent) LUMEX_NOEXCEPT
160{
161 long long value = 1;
162 for (int i = 0; i < exponent; ++i)
163 value *= 10;
164 return value;
165}

171template <typename Period>
172int
173fractional_digits () LUMEX_NOEXCEPT
174{
175 for (int digits = 0; digits <= 18; ++digits)
176 if (power_of_ten (digits) % Period::den == 0)
177 return digits;
178 return 6;

185template <typename Rep, typename Period>
186void
187split_seconds (std::chrono::duration<Rep, Period> const &value, int digits,
188 long long &whole_seconds, long long &ticks)
189{
190 std::chrono::seconds const whole
191 = std::chrono::duration_cast<std::chrono::seconds> (value);
192 whole_seconds = static_cast<long long> (whole.count ());
193 ticks = 0;
194 if (digits == 0)
195 return;
196 typedef typename std::common_type<std::chrono::duration<Rep, Period>,
197 std::chrono::seconds>::type common_type;
198 common_type const rest = common_type (value) - common_type (whole);
199 typedef typename common_type::period period;
200 long long const scale = power_of_ten (digits);
201 if (!std::is_floating_point<Rep>::value && scale % period::den == 0)
202 {
203 // Exact in integers: rest < 1 s, so the product stays below 10^18.
204 ticks = static_cast<long long> (rest.count ()) * period::num
205 * (scale / period::den);
206 return;
207 }
208 // Floating-point counts, or a period without an exact decimal fraction
209 // (6 digits, truncated).
210 typedef std::chrono::duration<double> double_seconds;
211 double const fraction
212 = std::chrono::duration_cast<double_seconds> (rest).count ();
213 ticks = static_cast<long long> (fraction * static_cast<double> (scale));
214 if (ticks >= scale)
215 ticks = scale - 1;
216 else if (ticks <= -scale)
217 ticks = 1 - scale;
218}

277template <typename Char>
278bool
279is_chrono_conversion (Char conversion, Char modifier,
280 bool is_duration) LUMEX_NOEXCEPT
281{
282 char const value = static_cast<char> (conversion);
283 if (static_cast<Char> (value) != conversion || value == '\0')
284 return false;
285 // Time of day and literals work for both; `%Q %q` only for durations;
286 // calendar conversions only for time points.
287 static char const common_ok[] = "HMSTRIprjnt%";
288 static char const duration_ok[] = "Qq";
289 static char const date_ok[] = "YCymdeFDaAbBhuwUWVGgcxXZz";
290 std::string const allowed
291 = std::string (common_ok) + (is_duration ? duration_ok : date_ok);
292 if (allowed.find (value) == std::string::npos)
293 return false;
294 if (modifier == static_cast<Char> ('E'))
295 {
296 static char const e_ok[] = "cCxXyYz";
297 for (char const *it = e_ok; *it != '\0'; ++it)
298 if (*it == value)
299 return !is_duration;
300 return false;
301 }
302 if (modifier == static_cast<Char> ('O'))
303 {
304 static char const o_ok[] = "deHImMSuUVwWyz";
305 for (char const *it = o_ok; *it != '\0'; ++it)
306 if (*it == value)
307 return !is_duration || value == 'H' || value == 'I' || value == 'M'
308 || value == 'S';
309 return false;
310 }
311 return true;
312}

318template <typename Char>
319void
320check_chrono_specs (Char const *begin, Char const *end, bool is_duration)
321{
322 if (begin != end && *begin != static_cast<Char> ('%'))
323 throw FormatError ("invalid format specifier");
324 for (Char const *it = begin; it != end; ++it)
325 {
326 if (*it == static_cast<Char> ('{'))
327 throw FormatError ("invalid format string");
328 if (*it != static_cast<Char> ('%'))
329 continue;
330 ++it;
331 if (it == end)
332 throw FormatError ("invalid format string");
333 Char modifier = Char ();
334 if (*it == static_cast<Char> ('E') || *it == static_cast<Char> ('O'))
335 {
336 modifier = *it;
337 ++it;
338 if (it == end)
339 throw FormatError ("invalid format string");
340 }
341 if (!is_chrono_conversion (*it, modifier, is_duration))
342 throw FormatError ("invalid format specifier");
343 }
344}

355template <typename Char>
356void
357append_number (std::basic_string<Char> &out, long long value, int width)
358{
359 if (value < 0)
360 {
361 out.push_back (static_cast<Char> ('-'));
362 value = -value;
363 }
364 std::string digits = std::to_string (value);
365 if (static_cast<int> (digits.size ()) < width)
366 digits.insert (0, static_cast<std::size_t> (width) - digits.size (), '0');
367 append_ascii (out, digits);
368}

383template <typename Char>
384void
385append_localized (std::basic_string<Char> &out, civil_time_t const &time,
386 char conversion, char modifier, std::locale const &locale)
387{
388 std::tm tm_value = std::tm ();
389 tm_value.tm_year = static_cast<int> (time.year - 1900);
390 tm_value.tm_mon = time.month - 1;
391 tm_value.tm_mday = time.day;
392 tm_value.tm_hour = static_cast<int> (time.hours % 24);
393 tm_value.tm_min = time.minutes;
394 tm_value.tm_sec = time.seconds;
395 tm_value.tm_wday = time.weekday;
396 tm_value.tm_yday = time.year_day;
397 std::basic_ostringstream<Char> stream;
398 stream.imbue (locale);
399 std::use_facet<std::time_put<Char>> (locale).put (
400 std::ostreambuf_iterator<Char> (stream), stream, stream.fill (),
401 &tm_value, conversion, modifier);
402 out += stream.str ();
403}

430inline void
431iso_week (civil_time_t const &time, long long &iso_year,
432 int &week) LUMEX_NOEXCEPT
433{
434 long long const day_count
435 = days_from_civil (time.year, time.month, time.day);
436 int const iso_weekday = time.weekday == 0 ? 7 : time.weekday;
437 long long const thursday = day_count + 4 - iso_weekday;
438 civil_time_t thursday_time = civil_time_t ();
439 civil_from_days (thursday, thursday_time);
440 iso_year = thursday_time.year;
441 week = thursday_time.year_day / 7 + 1;
442}

445template <typename Char>
446void
447write_chrono (std::basic_string<Char> &out, Char const *it, Char const *end,
448 civil_time_t const &time, std::basic_string<Char> const &count,
449 std::string const &unit, bool localized,
450 std::locale const &locale)
451{
452 if (time.negative)
453 out.push_back (static_cast<Char> ('-'));
454 for (; it != end; ++it)
455 {
456 if (*it != static_cast<Char> ('%'))
457 {
458 out.push_back (*it);
459 continue;
460 }
461 ++it;
462 char modifier = '\0';
463 if (*it == static_cast<Char> ('E') || *it == static_cast<Char> ('O'))
464 {
465 modifier = static_cast<char> (*it);
466 ++it;
467 }
468 char const conversion = static_cast<char> (*it);
469 long long const hour12 = time.hours % 12 == 0 ? 12 : time.hours % 12;
470 bool const names = conversion == 'a' || conversion == 'A'
471 || conversion == 'b' || conversion == 'B'
472 || conversion == 'h' || conversion == 'p'
473 || conversion == 'c' || conversion == 'x'
474 || conversion == 'X' || conversion == 'r';
475 if (localized && names)
476 {
477 append_localized (out, time, conversion, modifier, locale);
478 continue;
479 }
480 switch (conversion)
481 {
482 case 'n':
483 out.push_back (static_cast<Char> ('\n'));
484 break;
485 case 't':
486 out.push_back (static_cast<Char> ('\t'));
487 break;
488 case '%':
489 out.push_back (static_cast<Char> ('%'));
490 break;
491 case 'Q':
492 out += count;
493 break;
494 case 'q':
495 append_ascii (out, unit);
496 break;
497 case 'H':
498 append_number (out, time.hours, 2);
499 break;
500 case 'I':
501 append_number (out, hour12, 2);
502 break;
503 case 'M':
504 append_number (out, time.minutes, 2);
505 break;
506 case 'S':
507 append_seconds (out, time);
508 break;
509 case 'p':
510 append_ascii (out, time.hours % 24 < 12 ? "AM" : "PM");
511 break;
512 case 'R':
513 append_number (out, time.hours, 2);
514 out.push_back (static_cast<Char> (':'));
515 append_number (out, time.minutes, 2);
516 break;
517 case 'T':
518 append_number (out, time.hours, 2);
519 out.push_back (static_cast<Char> (':'));
520 append_number (out, time.minutes, 2);
521 out.push_back (static_cast<Char> (':'));
522 append_seconds (out, time);
523 break;
524 case 'r':
525 append_number (out, hour12, 2);
526 out.push_back (static_cast<Char> (':'));
527 append_number (out, time.minutes, 2);
528 out.push_back (static_cast<Char> (':'));
529 append_number (out, time.seconds, 2);
530 append_ascii (out, time.hours % 24 < 12 ? " AM" : " PM");
531 break;
532 case 'j':
533 if (time.has_date)
534 append_number (out, time.year_day + 1, 3);
535 else
536 append_number (out, time.days, 0);
537 break;
538 case 'Y':
539 append_number (out, time.year, 4);
540 break;
541 case 'C':
542 append_number (out, floor_div (time.year, 100), 2);
543 break;
544 case 'y':
545 append_number (out, floor_mod (time.year, 100), 2);
546 break;
547 case 'm':
548 append_number (out, time.month, 2);
549 break;
550 case 'd':
551 append_number (out, time.day, 2);
552 break;
553 case 'e':
554 if (time.day < 10)
555 out.push_back (static_cast<Char> (' '));
556 append_number (out, time.day, 1);
557 break;
558 case 'F':
559 append_number (out, time.year, 4);
560 out.push_back (static_cast<Char> ('-'));
561 append_number (out, time.month, 2);
562 out.push_back (static_cast<Char> ('-'));
563 append_number (out, time.day, 2);
564 break;
565 case 'D':
566 case 'x':
567 append_number (out, time.month, 2);
568 out.push_back (static_cast<Char> ('/'));
569 append_number (out, time.day, 2);
570 out.push_back (static_cast<Char> ('/'));
571 append_number (out, floor_mod (time.year, 100), 2);
572 break;
573 case 'X':
574 append_number (out, time.hours, 2);
575 out.push_back (static_cast<Char> (':'));
576 append_number (out, time.minutes, 2);
577 out.push_back (static_cast<Char> (':'));
578 append_number (out, time.seconds, 2);
579 break;
580 case 'c':
581 append_ascii (out, weekday_name (time.weekday, false));
582 out.push_back (static_cast<Char> (' '));
583 append_ascii (out, month_name (time.month, false));
584 out.push_back (static_cast<Char> (' '));
585 if (time.day < 10)
586 out.push_back (static_cast<Char> (' '));
587 append_number (out, time.day, 1);
588 out.push_back (static_cast<Char> (' '));
589 append_number (out, time.hours, 2);
590 out.push_back (static_cast<Char> (':'));
591 append_number (out, time.minutes, 2);
592 out.push_back (static_cast<Char> (':'));
593 append_number (out, time.seconds, 2);
594 out.push_back (static_cast<Char> (' '));
595 append_number (out, time.year, 4);
596 break;
597 case 'a':
598 case 'A':
599 append_ascii (out, weekday_name (time.weekday, conversion == 'A'));
600 break;
601 case 'b':
602 case 'h':
603 case 'B':
604 append_ascii (out, month_name (time.month, conversion == 'B'));
605 break;
606 case 'u':
607 append_number (out, time.weekday == 0 ? 7 : time.weekday, 1);
608 break;
609 case 'w':
610 append_number (out, time.weekday, 1);
611 break;
612 case 'U':
613 append_number (out, (time.year_day + 7 - time.weekday) / 7, 2);
614 break;
615 case 'W':
616 append_number (out, (time.year_day + 7 - (time.weekday + 6) % 7) / 7,
617 2);
618 break;
619 case 'V':
620 case 'G':
621 case 'g':
622 {
623 long long iso_year = 0;
624 int week = 0;
625 iso_week (time, iso_year, week);
626 if (conversion == 'V')
627 append_number (out, week, 2);
628 else if (conversion == 'G')
629 append_number (out, iso_year, 4);
630 else
631 append_number (out, floor_mod (iso_year, 100), 2);
632 }
633 break;
634 case 'Z':
635 append_ascii (out, "UTC");
636 break;
637 case 'z':
638 append_ascii (out, modifier != '\0' ? "+00:00" : "+0000");
639 break;
640 default:
641 throw FormatError ("invalid format specifier");
642 }
643 }
644}

651template <typename Char> class ChronoFormatter
652{
653public:
654 ChronoFormatter () : _specs (), _chrono () {}
655
656protected:
657 Char const *
658 parse_chrono (BasicFormatParseContext<Char> &ctx, bool is_duration,
659 bool allows_precision)
660 {
661 _specs = format_specs_t<Char> (); // a second parse starts from scratch
662 _chrono.clear ();
663 Char const *it = parse_fill_align_width (ctx, _specs, true);
664 Char const *const end = ctx.end ();
665 if (it != end && *it == static_cast<Char> ('.'))
666 {
667 if (!allows_precision)
668 throw FormatError ("invalid format specifier");
669 ++it;
670 if (it != end && is_digit (*it))
671 _specs.precision = parse_nonnegative_int (it, end);
672 else if (it != end && *it == static_cast<Char> ('{'))
673 _specs.precision_ref = parse_dynamic_ref (it, end, ctx);
674 else
675 throw FormatError ("invalid precision");
676 }
677 if (it != end && *it == static_cast<Char> ('L'))
678 {
679 _specs.localized = true;
680 ++it;
681 }
682 Char const *const specs_begin = it;
683 while (it != end && *it != static_cast<Char> ('}'))
684 ++it;
685 check_chrono_specs (specs_begin, it, is_duration);
686 _chrono.assign (specs_begin, it);
687 return it;
688 }
689
691 int
692 precision (BasicFormatContext<Char> const &ctx) const
693 {
694 return resolve_dynamic (_specs.precision_ref, _specs.precision, ctx);
695 }
696
697 void
698 write (BasicFormatContext<Char> &ctx,
699 std::basic_string<Char> const &text) const
700 {
701 format_specs_t<Char> specs = _specs;
702 specs.width = resolve_dynamic (specs.width_ref, specs.width, ctx);
703 write_padded (ctx.out ().buffer (), specs, Align::left, text);
704 }
705
706 bool
707 has_chrono_specs () const LUMEX_NOEXCEPT
708 {
709 return !_chrono.empty ();
710 }
711
712 std::basic_string<Char> const &
713 chrono_specs () const LUMEX_NOEXCEPT
714 {
715 return _chrono;
716 }
717
718 bool
719 localized () const LUMEX_NOEXCEPT
720 {
721 return _specs.localized;
722 }
723
724private:
725 format_specs_t<Char> _specs;
726 std::basic_string<Char> _chrono;
727};

730template <typename Char, typename Rep>
731std::basic_string<Char>
732duration_count (Rep count, int precision)
733{
734 std::basic_string<Char> text;
735 StringBuffer<Char> buffer (text);
736 std::basic_string<Char> spec;
737 if (std::is_floating_point<Rep>::value && precision >= 0)
738 {
739 spec.push_back (static_cast<Char> ('.'));
740 append_number (spec, precision, 1);
741 spec.push_back (static_cast<Char> ('f'));
742 }
743 spec.push_back (static_cast<Char> ('}'));
744 Formatter<Rep, Char> formatter;
745 BasicFormatParseContext<Char> parse_ctx (
746 BasicStringRef<Char> (spec.data (), spec.size ()));

760template <typename Rep, typename Period, typename Char>
761class Formatter<
762 std::chrono::duration<Rep, Period>, Char,
763 typename std::enable_if<Detail::has_formatter<Char, Rep> ()
764 && std::is_arithmetic<Rep>::value>::type>
765 : public Detail::ChronoFormatter<Char>
766{
767public:
768 Char const *
769 parse (BasicFormatParseContext<Char> &ctx)
770 {
771 return this->parse_chrono (ctx, true, std::is_floating_point<Rep>::value);
772 }
773
774 BasicAppender<Char>
775 format (std::chrono::duration<Rep, Period> const &value,
776 BasicFormatContext<Char> &ctx) const
777 {
778 int const precision = this->precision (ctx);
779 std::basic_string<Char> text;
780 if (!this->has_chrono_specs ())
781 {
782 text = Detail::duration_count<Char> (value.count (), precision);
783 Detail::append_ascii (text, Detail::unit_suffix<Period> ());
784 }
785 else
786 {
787 typedef std::chrono::duration<Rep, Period> duration_type;
788 bool const negative = value < duration_type::zero ();
789 if (std::is_floating_point<Rep>::value)
790 {
791 // Whole seconds must fit a long long; inf / nan have no fields.
792 double const seconds
793 = std::chrono::duration_cast<std::chrono::duration<double>> (
794 value)
795 .count ();
796 if (!(std::fabs (seconds) < 9.2e18))
797 throw FormatError ("duration is out of range for chrono-specs");
798 }
799 Detail::civil_time_t time = Detail::civil_time_t ();
800 time.negative = negative;
801 time.digits = Detail::fractional_digits<Period> ();
802 // Split the signed value (both parts truncate toward zero), then take
803 // the magnitude of the parts: negating the value itself would
804 // overflow for duration::min ().
805 long long signed_seconds = 0;
806 long long signed_ticks = 0;
807 Detail::split_seconds (value, time.digits, signed_seconds,
808 signed_ticks);
809 unsigned long long const total_seconds
810 = signed_seconds < 0
811 ? 0ull - static_cast<unsigned long long> (signed_seconds)
812 : static_cast<unsigned long long> (signed_seconds);
813 time.ticks = signed_ticks < 0 ? -signed_ticks : signed_ticks;
814 time.hours = static_cast<long long> (total_seconds / 3600);
815 time.minutes = static_cast<int> (total_seconds % 3600 / 60);
816 time.seconds = static_cast<int> (total_seconds % 60);
817 time.days = static_cast<long long> (total_seconds / 86400);
818 std::basic_string<Char> count
819 = Detail::duration_count<Char> (value.count (), precision);
820 if (!count.empty () && count[0] == static_cast<Char> ('-'))
821 count.erase (0, 1); // the sign is printed once, in front
822 std::basic_string<Char> const &specs = this->chrono_specs ();
823 Detail::write_chrono (
824 text, specs.data (), specs.data () + specs.size (), time, count,
825 Detail::unit_suffix<Period> (), this->localized (), ctx.locale ());
826 }
827 this->write (ctx, text);
828 return ctx.out ();
829 }
830};

836template <typename Duration, typename Char>
837class Formatter<std::chrono::time_point<std::chrono::system_clock, Duration>,
838 Char,
839 typename std::enable_if<!std::chrono::treat_as_floating_point<
840 typename Duration::rep>::value>::type>
841 : public Detail::ChronoFormatter<Char>
842{
843public:
844 Char const *
845 parse (BasicFormatParseContext<Char> &ctx)
846 {
847 return this->parse_chrono (ctx, false, false);
848 }
849
850 BasicAppender<Char>
851 format (std::chrono::time_point<std::chrono::system_clock, Duration> const
852 &value,
853 BasicFormatContext<Char> &ctx) const
854 {
855 typedef std::chrono::duration<long long, std::ratio<86400>> days_type;
856 typedef typename std::common_type<Duration, days_type,
857 std::chrono::seconds>::type common_type;
858 common_type const since_epoch (value.time_since_epoch ());
859 // Whole seconds rounded down (duration_cast truncates toward zero),
860 // then whole days rounded down, so times before 1970 stay correct.
861 std::chrono::seconds whole
862 = std::chrono::duration_cast<std::chrono::seconds> (since_epoch);
863 if (common_type (whole) > since_epoch)
864 whole -= std::chrono::seconds (1);
865 long long const day_count
866 = Detail::floor_div (static_cast<long long> (whole.count ()), 86400);
867 common_type const time_of_day
868 = since_epoch - common_type (days_type (day_count));
869 Detail::civil_time_t time = Detail::civil_time_t ();
870 time.has_date = true;
871 Detail::civil_from_days (day_count, time);
872 time.digits = Detail::fractional_digits<typename Duration::period> ();
873 long long total_seconds = 0;
874 Detail::split_seconds (time_of_day, time.digits, total_seconds,
875 time.ticks);
876 time.hours = total_seconds / 3600;
877 time.minutes = static_cast<int> (total_seconds % 3600 / 60);
878 time.seconds = static_cast<int> (total_seconds % 60);
879
880 static Char const date_time[] = { '%', 'F', ' ', '%', 'T' };
881 static Char const date_only[] = { '%', 'F' };
882 bool const whole_days = std::ratio_greater_equal<typename Duration::period,
883 std::ratio<86400>>::value;
884 std::basic_string<Char> const default_specs
885 = whole_days ? std::basic_string<Char> (date_only, 2)
886 : std::basic_string<Char> (date_time, 5);
887 std::basic_string<Char> const &specs
888 = this->has_chrono_specs () ? this->chrono_specs () : default_specs;
889 std::basic_string<Char> text;
890 Detail::write_chrono (text, specs.data (), specs.data () + specs.size (),
891 time, std::basic_string<Char> (), std::string (),
892 this->localized (), ctx.locale ());
893 this->write (ctx, text);
894 return ctx.out ();
895 }
