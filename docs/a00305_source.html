129// 116444736000000000ULL - count of intervals between 1601 and 1970
130// If subtract this number from ulInt.QuadPart (which contains the number of
131// 100-nanosecond intervals since 1601) we get the number of 100-nanosecond
132// intervals since 1970.
133
134// 10000000ULL - count of 100-nanosecond interval in 1 second
135// Since std::time_t is measured in seconds, and ulInt.QuadPart (after
136// subtraction) contains the number of 100-nanosecond intervals, dividing by
137// 10000000ULL converts these 100-nanosecond intervals into seconds.
140LUMEX_CONSTEXPR unsigned long long
141 KDEFAULT_WINDOWS_FILETIME_TO_UNIX_EPOCH_INTERVALS
142 = 116444736000000000ULL;
145LUMEX_CONSTEXPR unsigned long long KDEFAULT_HUNDRED_NANOSECONDS_PER_SECOND
146 = 10000000ULL;
