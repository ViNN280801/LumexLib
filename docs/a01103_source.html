172template <typename T, typename std::enable_if<
173 traits::meta::is_extractible<T>::value, int>::type
174 = 0>
175LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
176optional_t<T> as (void const *data, std::size_t size) LUMEX_NOEXCEPT
177{
178 if ((data == nullptr) || (size < sizeof (T)))
179 return optional_t<T> ();
180
181 T res{};
182 std::memcpy (std::addressof (res), data, sizeof (T));
183 return optional_t<T> (res);
184}

193template <
194 typename T, typename TSource,
195 typename std::enable_if<traits::meta::is_extractible<T>::value
196 && Detail::is_data_source<TSource>::value,
197 int>::type
198 = 0>
199LUMEX_ATTRIBUTE_NODISCARD ("return value must be used")
200optional_t<T> as (TSource const &source) LUMEX_NOEXCEPT
201{
202 int const rawSize{ source.get_data_size () };
203 if (rawSize < 0)
204 return optional_t<T> ();
205 return as<T> (source.get_data (), static_cast<std::size_t> (rawSize));
206}
