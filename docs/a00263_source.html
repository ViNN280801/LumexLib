214// 1 option. Define the exception class: LUMEX_DEFINE_EXCEPTION and
215// LUMEX_DEFINE_EXCEPTION_WITH_BODY live in LumexExceptionMacros.hpp (included
216// above) so modules that must not depend on this one can use them too.
217
218// 2 option. Throw the exception.
219// Pattern:
220// [exception_name] -> custom message
221// Example:
222// [LumexException] -> Failed to open file
223#include "lumex/core/string/LumexString"
224#define LUMEX_THROW_EXCEPTION(exception_name, msg) \
225 throw exception_name (lumex::core::string::utility::stringify ( \
226 lumDemangle (exception_name), ": ", msg));
227
228// 3. Handle the exception.
229#define LUMEX_EXCEPTION_HANDLE_BEGIN \
230 try \
231 { \
232 SET_SEH_TRANSLATOR
233
234#define LUMEX_EXCEPTION_HANDLE_END \
235 } \
236 catch (LumexBaseException const &ex) \
237 { \
238 ex.to_stderr (); \
239 ex.to_crash_report (); \
240 } \
241 catch (std::exception const &ex) \
242 { \
243 std::cerr << "[std::exception] " << ex.what () << '\n'; \
244 LumexBaseException (ex.what ()).to_stderr (); \
245 LumexBaseException (ex.what ()).to_crash_report (); \
246 } \
247 catch (...) { std::cerr << "[Unknown exception]\n"; }
