353// The scans: the first (last) index of [lo, hi) that satisfies pred, else
354// no_pos. From C++14 they are loops. In C++11 a constexpr function is a
355// single return statement, so constant evaluation uses a recursion over
356// halves (the depth is log2 of the length) and run time a plain loop.
357#if __cplusplus >= 201402L
358template <class Pred>
359LUMEX_CONSTEXPR std::size_t
360scan_fwd (Pred const &pred, std::size_t lo, std::size_t hi) LUMEX_NOEXCEPT
361{
362 for (; lo < hi; ++lo)
363 if (pred (lo))
364 return lo;
365 return no_pos;
366}

368template <class Pred>
369LUMEX_CONSTEXPR std::size_t
370scan_bwd (Pred const &pred, std::size_t lo, std::size_t hi) LUMEX_NOEXCEPT
371{
372 for (; hi > lo; --hi)
373 if (pred (hi - 1))
374 return hi - 1;
375 return no_pos;
376}
377#else
378template <class Pred>
379LUMEX_CONSTEXPR std::size_t scan_fwd_rec (Pred const &pred, std::size_t lo,
380 std::size_t hi) LUMEX_NOEXCEPT;

383LUMEX_CONSTEXPR std::size_t
384scan_fwd_pick (std::size_t found, Pred const &pred, std::size_t mid,

386{
387 return found != no_pos ? found : scan_fwd_rec (pred, mid, hi);

400}

824inline std::uint64_t
825hash_round (std::uint64_t acc, std::uint64_t block) LUMEX_NOEXCEPT
826{
827 block *= 0xC2B2AE3D27D4EB4FULL;
828 block = (block << 31) | (block >> 33);
829 block *= 0x9E3779B185EBCA87ULL;
830 acc ^= block;
831 acc = ((acc << 27) | (acc >> 37)) * 0x9E3779B185EBCA87ULL
832 + 0x85EBCA77C2B2AE63ULL;
833 return acc;
834}
835
836inline std::uint64_t
837hash_final (std::uint64_t acc) LUMEX_NOEXCEPT
838{
839 acc ^= acc >> 33;
840 acc *= 0xFF51AFD7ED558CCDULL;
841 acc ^= acc >> 33;
842 acc *= 0xC4CEB9FE1A85EC53ULL;
843 acc ^= acc >> 33;
844 return acc;
845}

848inline std::uint64_t
849load_le64 (unsigned char const *bytes) LUMEX_NOEXCEPT
850{
851 std::uint64_t value;
852 std::memcpy (&value, bytes, sizeof value);
853#if defined(__BYTE_ORDER__) && defined(__ORDER_BIG_ENDIAN__) \
854 && __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
855 value = ((value & 0x00000000000000FFULL) << 56)
856 | ((value & 0x000000000000FF00ULL) << 40)
857 | ((value & 0x0000000000FF0000ULL) << 24)
858 | ((value & 0x00000000FF000000ULL) << 8)
859 | ((value & 0x000000FF00000000ULL) >> 8)
860 | ((value & 0x0000FF0000000000ULL) >> 24)
861 | ((value & 0x00FF000000000000ULL) >> 40)
862 | ((value & 0xFF00000000000000ULL) >> 56);
863#endif
864 return value;
865}

867template <class CharT>
868std::uint64_t
869block_hash (CharT const *data, std::size_t size, std::true_type) LUMEX_NOEXCEPT
870{
871 // One-byte characters: the bytes, eight per round.
872 unsigned char const *bytes = reinterpret_cast<unsigned char const *> (data);
873 std::uint64_t acc = 0x27D4EB2F165667C5ULL ^ (size * 0x9E3779B97F4A7C15ULL);
874 std::size_t at = 0;
875 for (; at + 8 <= size; at += 8)
876 acc = hash_round (acc, load_le64 (bytes + at));
877 if (at < size)
878 {
879 std::uint64_t tail = 0;
880 for (std::size_t i = 0; at + i < size; ++i)
881 tail |= static_cast<std::uint64_t> (bytes[at + i]) << (8 * i);
882 acc = hash_round (acc, tail);
883 }
884 return hash_final (acc);
885}

887template <class CharT>
888std::uint64_t
889block_hash (CharT const *data, std::size_t size,
890 std::false_type) LUMEX_NOEXCEPT
891{
892 // Wider characters: the code unit values, two per round.
893 typedef typename std::make_unsigned<CharT>::type unit_t;
894 std::uint64_t acc = 0x27D4EB2F165667C5ULL ^ (size * 0x9E3779B97F4A7C15ULL);
895 std::size_t at = 0;
896 for (; at + 2 <= size; at += 2)
897 {
898 std::uint64_t const low
899 = static_cast<std::uint32_t> (static_cast<unit_t> (data[at]));
900 std::uint64_t const high
901 = static_cast<std::uint32_t> (static_cast<unit_t> (data[at + 1]));
902 acc = hash_round (acc, low | (high << 32));
903 }
904 if (at < size)
905 acc = hash_round (
906 acc, static_cast<std::uint32_t> (static_cast<unit_t> (data[at])));
907 return hash_final (acc);
908}
