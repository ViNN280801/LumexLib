373// The scans: the first (last) index of [lo, hi) that satisfies pred, else
374// no_pos. From C++14 they are loops. In C++11 a constexpr function is a
375// single return statement, so constant evaluation uses a recursion over
376// halves (the depth is log2 of the length) and run time a plain loop.
377#if __cplusplus >= 201402L
378template <class Pred>
379LUMEX_CONSTEXPR std::size_t
380scan_fwd (Pred const &pred, std::size_t lo, std::size_t hi) LUMEX_NOEXCEPT
381{
382 for (; lo < hi; ++lo)
383 if (pred (lo))
384 return lo;
385 return no_pos;
386}
387
388template <class Pred>
389LUMEX_CONSTEXPR std::size_t
390scan_bwd (Pred const &pred, std::size_t lo, std::size_t hi) LUMEX_NOEXCEPT
391{
392 for (; hi > lo; --hi)
393 if (pred (hi - 1))
394 return hi - 1;
395 return no_pos;
396}
397#else
398template <class Pred>
399LUMEX_CONSTEXPR std::size_t scan_fwd_rec (Pred const &pred, std::size_t lo,
400 std::size_t hi) LUMEX_NOEXCEPT;

404scan_fwd_pick (std::size_t found, Pred const &pred, std::size_t mid,
405 std::size_t hi) LUMEX_NOEXCEPT
406{
407 return found != no_pos ? found : scan_fwd_rec (pred, mid, hi);

844inline std::uint64_t
845hash_round (std::uint64_t acc, std::uint64_t block) LUMEX_NOEXCEPT
846{
847 block *= 0xC2B2AE3D27D4EB4FULL;
848 block = (block << 31) | (block >> 33);
849 block *= 0x9E3779B185EBCA87ULL;
850 acc ^= block;
851 acc = ((acc << 27) | (acc >> 37)) * 0x9E3779B185EBCA87ULL
852 + 0x85EBCA77C2B2AE63ULL;
853 return acc;
854}
855
856inline std::uint64_t
857hash_final (std::uint64_t acc) LUMEX_NOEXCEPT
858{
859 acc ^= acc >> 33;
860 acc *= 0xFF51AFD7ED558CCDULL;
861 acc ^= acc >> 33;
862 acc *= 0xC4CEB9FE1A85EC53ULL;
863 acc ^= acc >> 33;
864 return acc;
865}

868inline std::uint64_t
869load_le64 (unsigned char const *bytes) LUMEX_NOEXCEPT
870{
871 std::uint64_t value;
872 std::memcpy (&value, bytes, sizeof value);
873#if defined(__BYTE_ORDER__) && defined(__ORDER_BIG_ENDIAN__) \
874 && __BYTE_ORDER__ == __ORDER_BIG_ENDIAN__
875 value = ((value & 0x00000000000000FFULL) << 56)
876 | ((value & 0x000000000000FF00ULL) << 40)
877 | ((value & 0x0000000000FF0000ULL) << 24)
878 | ((value & 0x00000000FF000000ULL) << 8)
879 | ((value & 0x000000FF00000000ULL) >> 8)
880 | ((value & 0x0000FF0000000000ULL) >> 24)
881 | ((value & 0x00FF000000000000ULL) >> 40)
882 | ((value & 0xFF00000000000000ULL) >> 56);
883#endif
884 return value;
885}

887template <class CharT>
888std::uint64_t
889block_hash (CharT const *data, std::size_t size, std::true_type) LUMEX_NOEXCEPT
890{
891 // One-byte characters: the bytes, eight per round.
892 unsigned char const *bytes = reinterpret_cast<unsigned char const *> (data);
893 std::uint64_t acc = 0x27D4EB2F165667C5ULL ^ (size * 0x9E3779B97F4A7C15ULL);
894 std::size_t at = 0;
895 for (; at + 8 <= size; at += 8)
896 acc = hash_round (acc, load_le64 (bytes + at));
897 if (at < size)
898 {
899 std::uint64_t tail = 0;
900 for (std::size_t i = 0; at + i < size; ++i)
901 tail |= static_cast<std::uint64_t> (bytes[at + i]) << (8 * i);
902 acc = hash_round (acc, tail);
903 }
904 return hash_final (acc);
905}
906
907template <class CharT>
908std::uint64_t
909block_hash (CharT const *data, std::size_t size,
910 std::false_type) LUMEX_NOEXCEPT
911{
912 // Wider characters: the code unit values, two per round.
913 typedef typename std::make_unsigned<CharT>::type unit_t;
914 std::uint64_t acc = 0x27D4EB2F165667C5ULL ^ (size * 0x9E3779B97F4A7C15ULL);
915 std::size_t at = 0;
916 for (; at + 2 <= size; at += 2)
917 {
918 std::uint64_t const low
919 = static_cast<std::uint32_t> (static_cast<unit_t> (data[at]));
920 std::uint64_t const high
921 = static_cast<std::uint32_t> (static_cast<unit_t> (data[at + 1]));
922 acc = hash_round (acc, low | (high << 32));
923 }
924 if (at < size)
925 acc = hash_round (
926 acc, static_cast<std::uint32_t> (static_cast<unit_t> (data[at])));
927 return hash_final (acc);
928}

1024 std::streamsize const width = ostr.width ();

1033 good = buf->sputn (data, size) == size;
1034 if (good && left)
1035 good = put_fill (buf, ostr.fill (), pad);
