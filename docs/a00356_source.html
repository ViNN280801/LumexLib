93template <typename NumericType>
94typename std::enable_if<
95 std::is_integral<NumericType>::value
96 && std::is_unsigned<NumericType>::value
97 && !std::is_same<bool, NumericType>::value
98 && (sizeof (NumericType) == 1 || sizeof (NumericType) == 2
99 || sizeof (NumericType) == 4 || sizeof (NumericType) == 8),
100 std::uint8_t>::type
101count_leading_zeros (NumericType value) LUMEX_NOEXCEPT
102{
103 // __builtin_clz(0) is undefined, and _BitScanReverse(0) reports no bit.
104 if (value == 0)
105 return sizeof (NumericType) * 8;
106
107#if defined(_MSC_VER)
108 // clang-cl defines __clang__ as well as _MSC_VER. This branch is first
109 // so this compiler keeps the MSVC intrinsic.
110 unsigned long index = 0;
111
112 if (sizeof (NumericType) <= 4)
113 {
114 _BitScanReverse (std::addressof (index),
115 static_cast<std::uint32_t> (value));
116 // Index of the highest set bit (0 is the least significant bit).
117 return (sizeof (NumericType) * 8) - 1 - index;
118 }
119 else
120 {
121#if defined(_M_X64) || defined(_M_ARM64)
122 _BitScanReverse64 (std::addressof (index),
123 static_cast<std::uint64_t> (value));
124 return 64 - 1 - index;
125#else
126 // 32-bit MSVC has no _BitScanReverse64. Split the value in half.
127 std::uint32_t high = static_cast<std::uint32_t> (value >> 32);
128 if (high != 0)
129 {
130 _BitScanReverse (std::addressof (index), high);
131 return 32 - 1 - index;
132 }
133 else
134 {
135 _BitScanReverse (std::addressof (index),
136 static_cast<std::uint32_t> (value));
137 return 64 - 1 - index;
138 }
139#endif
140 }
141
142#elif defined(__GNUC__) || defined(__clang__)
143 if (sizeof (NumericType) <= 4)
144 {
145 // __builtin_clz counts zeros of a 32-bit unsigned int.
146 return __builtin_clz (static_cast<std::uint32_t> (value))
147 - (32 - sizeof (NumericType) * 8);
148 }
149 else
150 {
151 return __builtin_clzll (static_cast<std::uint64_t> (value));
152 }
153
154#else
155 std::uint8_t bits = sizeof (NumericType) * 8;
156 std::uint8_t count = 0;
157
158 // Halve the window: 32, then 16, 8, 4, 2, 1 for a 64-bit value.
159 for (int shift = bits >> 1; shift > 0; shift >>= 1)
160 {
161 if ((value >> shift) != 0)
162 {
163 value >>= shift;
164 }
165 else
166 {
167 count += shift;
168 }
169 }
170 return count;
171#endif
172}

210template <typename T>
211 requires std::is_integral_v<T> && std::has_unique_object_representations_v<T>
212LUMEX_CONSTEXPR T
213byte_swap (T value) LUMEX_NOEXCEPT
214{
215 using U = std::make_unsigned_t<T>;
216 U u = static_cast<U> (value);
217
218 LUMEX_CONSTEXPR_IF (sizeof (T) == 1) { return value; }
219 else
220 {
221 if (std::is_constant_evaluated ())
222 {
223 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
224 std::ranges::reverse (bytes);
225 U swapped = std::bit_cast<U> (bytes);
226 return static_cast<T> (swapped);
227 }
228
229#if defined(__clang__) || defined(__GNUC__)
230 LUMEX_CONSTEXPR_IF (sizeof (T) == 2)
231 {
232 u = static_cast<U> (__builtin_bswap16 (static_cast<std::uint16_t> (u)));
233 }
234 else LUMEX_CONSTEXPR_IF (sizeof (T) == 4)
235 {
236 u = static_cast<U> (__builtin_bswap32 (static_cast<std::uint32_t> (u)));
237 }
238 else LUMEX_CONSTEXPR_IF (sizeof (T) == 8)
239 {
240 u = static_cast<U> (__builtin_bswap64 (static_cast<std::uint64_t> (u)));
241 }
242 else
243 {
244 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
245 std::ranges::reverse (bytes);
246 u = std::bit_cast<U> (bytes);
247 }
248 return static_cast<T> (u);
249#elif defined(_MSC_VER)
250 LUMEX_CONSTEXPR_IF (sizeof (T) == 2)
251 {
252 u = static_cast<U> (_byteswap_ushort (static_cast<std::uint16_t> (u)));
253 }
254 else LUMEX_CONSTEXPR_IF (sizeof (T) == 4)
255 {
256 u = static_cast<U> (_byteswap_ulong (static_cast<std::uint32_t> (u)));
257 }
258 else LUMEX_CONSTEXPR_IF (sizeof (T) == 8)
259 {
260 u = static_cast<U> (_byteswap_uint64 (static_cast<std::uint64_t> (u)));
261 }
262 else
263 {
264 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
265 std::ranges::reverse (bytes);
266 u = std::bit_cast<U> (bytes);
267 }
268 return static_cast<T> (u);
269#else
270 // Portable fallback for compilers without a recognized bswap builtin.
271 auto bytes = std::bit_cast<std::array<std::byte, sizeof (T)>> (u);
272 std::ranges::reverse (bytes);
273 u = std::bit_cast<U> (bytes);
274 return static_cast<T> (u);
275#endif
276 }
