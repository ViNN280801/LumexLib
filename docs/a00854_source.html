431template <typename T>
432LUMEX_API inline T *
433new_xpath_variable (char_t const *name)
434{
435 std::size_t length = utility::strlength (name);
436 if (length == 0)
437 return nullptr; // empty variable names are invalid
438
439 // $$ we can't use offsetof(T, name) because T is non-POD, so we just
440 // allocate additional length characters
441 void *memory // NOLINT(cppcoreguidelines-owning-memory)
442 = malloc (sizeof (T) // NOLINT(cppcoreguidelines-no-malloc)
443 + (length * sizeof (char_t)));
444 if (!memory)
445 return nullptr;
446
447 T *result = new (memory) T (); // NOLINT(cppcoreguidelines-owning-memory)
448
449 std::memcpy (
450 result
451 ->name, // NOLINT(cppcoreguidelines-pro-bounds-array-to-pointer-decay)
452 name, (length + 1) * sizeof (char_t));
453
454 return result;
455}
