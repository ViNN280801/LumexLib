116template <typename Pointer> struct smart_ptr_traits_t;

122template <typename T> struct smart_ptr_traits_t<std::shared_ptr<T>>
123{
125 struct probe_t
126 {
127 };
128
129 static probe_t
130 probe (std::shared_ptr<T> const &) LUMEX_NOEXCEPT
131 {
132 return probe_t ();
133 }
134
135 static bool
136 equivalent (std::shared_ptr<T> const &current,
137 std::shared_ptr<T> const &expected, probe_t const &,
138 probe_t &) LUMEX_NOEXCEPT
139 {
140 return current.get () == expected.get ()
141 && !current.owner_before (expected)
142 && !expected.owner_before (current);
143 }
144};

150template <typename T> struct smart_ptr_traits_t<std::weak_ptr<T>>
151{
153 using probe_t = std::shared_ptr<T>;
154
155 static probe_t
156 probe (std::weak_ptr<T> const &pointer) LUMEX_NOEXCEPT
157 {
158 return pointer.lock ();
159 }
160
161 static bool
162 equivalent (std::weak_ptr<T> const &current,
163 std::weak_ptr<T> const &expected, probe_t const &expected_probe,
164 probe_t &current_probe) LUMEX_NOEXCEPT
165 {
166 if (current.owner_before (expected) || expected.owner_before (current))
167 return false;
168 // Same owner. An empty probe means that the object was gone when it was
169 // taken (or that both pointers are empty): the stored pointers can no
170 // longer be observed. use_count () rather than operator bool, because a
171 // live owner may store a null pointer.
172 if (expected_probe.use_count () == 0)
173 return true;
174 // The probe keeps the object alive, so the current pointer, which has
175 // the same owner, locks as well.
176 current_probe = current.lock ();
177 return current_probe.get () == expected_probe.get ();
178 }
179};

185template <typename Pointer> class LockBasedCell
186{
187 using traits_type = smart_ptr_traits_t<Pointer>;
188 using probe_type = typename traits_type::probe_t;
189 using lock_guard_type = sync::Detail::BitLockGuard;
190
191public:
192 static LUMEX_CONSTEXPR bool is_always_lock_free = false;
193
194 LUMEX_CONSTEXPR
195 LockBasedCell () LUMEX_NOEXCEPT : lock_ (), epoch_ (0u), value_ () {}
196
197 explicit LockBasedCell (Pointer desired) LUMEX_NOEXCEPT
198 : lock_ (),
199 epoch_ (0u),
200 value_ (std::move (desired))
201 {
202 }
203
204 LockBasedCell (LockBasedCell const &) = delete;
205 LockBasedCell &operator= (LockBasedCell const &) = delete;
206
207 bool
208 is_lock_free () const LUMEX_NOEXCEPT
209 {
210 return false;
211 }
212
213 Pointer
214 load (std::memory_order) const LUMEX_NOEXCEPT
215 {
216 lock_guard_type const guard (lock_);
217 return value_;
218 }
219
220 void
221 store (Pointer desired, std::memory_order) LUMEX_NOEXCEPT
222 {
223 lock_guard_type const guard (lock_);
224 // desired takes the previous value and releases it when the caller's
225 // argument is destroyed, after the lock.
226 value_.swap (desired);
227 }
228
229 Pointer
230 exchange (Pointer desired, std::memory_order) LUMEX_NOEXCEPT
231 {
232 {
233 lock_guard_type const guard (lock_);
234 value_.swap (desired);
235 }
236 return desired;
237 }
238
239 bool
240 compare_exchange_strong (Pointer &expected, Pointer desired,
241 std::memory_order, std::memory_order) LUMEX_NOEXCEPT
242 {
243 return compare_exchange (expected, std::move (desired));
244 }
245
246 bool
247 compare_exchange_strong (Pointer &expected, Pointer desired,
248 std::memory_order) LUMEX_NOEXCEPT
249 {
250 return compare_exchange (expected, std::move (desired));
251 }
252
254 bool
255 compare_exchange_weak (Pointer &expected, Pointer desired, std::memory_order,
256 std::memory_order) LUMEX_NOEXCEPT
257 {
258 return compare_exchange (expected, std::move (desired));
259 }
260
262 bool
263 compare_exchange_weak (Pointer &expected, Pointer desired,
264 std::memory_order) LUMEX_NOEXCEPT
265 {
266 return compare_exchange (expected, std::move (desired));
267 }
268
269 void
270 wait (Pointer const &old, std::memory_order) const LUMEX_NOEXCEPT
271 {
272 for (;;)
273 {
274 // Read the counter before the value: a notification that follows a
275 // change made after this comparison then changes the counter that
276 // this thread sleeps on.
277 std::uint32_t const epoch = epoch_.load (std::memory_order_acquire);
278 if (!holds_equivalent (old))
279 return;
280 sync::Detail::wait_until_changed (epoch_, epoch);
281 }
282 }
283
284 void
285 notify_one () LUMEX_NOEXCEPT
286 {
287 epoch_.fetch_add (1u, std::memory_order_release);
288 sync::Detail::notify_one (epoch_);
289 }
290
291 void
292 notify_all () LUMEX_NOEXCEPT
293 {
294 epoch_.fetch_add (1u, std::memory_order_release);
295 sync::Detail::notify_all (epoch_);
296 }
297
298private:
299 bool
300 compare_exchange (Pointer &expected, Pointer desired) LUMEX_NOEXCEPT
301 {
302 // Declared before the guard, so they are released after the lock.
303 probe_type const expected_probe = traits_type::probe (expected);
304 probe_type current_probe;
305 Pointer current;
306 bool exchanged = false;
307 {
308 lock_guard_type const guard (lock_);
309 if (traits_type::equivalent (value_, expected, expected_probe,
310 current_probe))
311 {
312 value_.swap (desired);
313 exchanged = true;
314 }
315 else
316 current = value_;
317 }
318 if (!exchanged)
319 expected = std::move (current);
320 return exchanged;
321 }
322
323 bool
324 holds_equivalent (Pointer const &old) const LUMEX_NOEXCEPT
325 {
326 probe_type const old_probe = traits_type::probe (old);
327 probe_type current_probe;
328 lock_guard_type const guard (lock_);
329 return traits_type::equivalent (value_, old, old_probe, current_probe);
330 }
331
332 sync::Detail::BitLock lock_;
333 std::atomic<std::uint32_t> epoch_;
334 Pointer value_;
335};
