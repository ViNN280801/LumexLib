# lumex/tests/core/math/constants/CMakeLists.txt

# The tests of lumex/core/math/constants (the LUMEX_MATH_CONSTANTS_* macros).
# One suite per standard of the module (lumex/tests/LumexTestStandards.cmake):
#   LumexMathConstantsCxx11Tests  Digits (the text of every macro against
#                                 digits computed outside the library),
#                                 Values (float, double, long double and the
#                                 constant expressions), Relations
#                                 (identities, ulps) and Macros (the macro
#                                 forms), all .cxx11
#   LumexMathConstantsCxx17Tests  the same + the special functions of <cmath>
#                                 (std::riemann_zeta, std::comp_ellint_1,
#                                 std::beta) where the library has them
#   LumexMathConstantsCxx20Tests  + std::numbers and consteval / constinit
# LumexMathConstantsReference.hpp holds the reference table, generated by
# LumexMathConstantsReference.py from the python decimal module.
lumex_add_standard_suites(MathConstants
    MODULE math
    LINK
        lumex::math
    SOURCES
        LumexMathConstantsReference.hpp
        LumexMathConstantsSupport.hpp
)
